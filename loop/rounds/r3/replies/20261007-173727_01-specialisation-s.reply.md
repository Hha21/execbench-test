### Rationale

Parent B200 times on the smallest inputs are 4.8 µs at 12.6 MB and 6.8 µs at 25.2 MB, against floors of 1.57 and 3.15 µs.

In the parent, each SM runs about 10 CTAs and 41 warps at B·S = 128. At B·S = 256 it asks for 3,072 CTAs, but only 2,368 fit at once, so a partial second wave runs. Every 8-row CTA also refetches its 4 KB weight tile from L2, and the first touch of each weight line misses to DRAM. That doubles the requests to L1/L2.

The specialist covers B·S ≤ 320 (four workloads). It uses one-shot programs that each own one (stream, 16-head group) for NT tokens, so the tile is a 3-D `[NT,16,128]` block:

- The weight tile is loaded once per program and broadcast over tokens.
- Each thread issues 8 independent x loads up front.
- 384–396 CTAs are all resident in one wave, about 2.6 per SM and 4× fewer warps.

B·S = 586 and everything larger use the parent's exact kernel and configuration.

```python file=kernel.py
"""r3-tokblk-s-disp: SOL-ExecBench #38, per-head RMSNorm of Q and K, fp32, [B, S, 48, 128].

Size dispatch (shapes only):
  tokens = B*S <= 320 : _qk_tokblk  (new small-input specialist)
  otherwise           : _qk_oneshot (parent g2-os-r8w4, identical kernel and config: ROWS=8, 4 warps, evict hints)

_qk_tokblk: program (c, g, s) handles stream s (0=Q, 1=K), head group g (HB heads) and tokens c*NT .. c*NT+NT-1
as one [NT, HB, 128] block. Each tile row is 512 contiguous bytes, and token blocks are H*D*4 bytes apart.
- The weight tile [HB, 128] is loaded once per program and broadcast over the NT tokens.
- All x loads of a thread are issued together; there is no loop and no shared memory.
- The grid is small (about 2.6 CTAs per SM on the smallest inputs), so every CTA is resident in a single wave.
A token mask is compiled in only when NT does not divide B*S (the 1x131 workload).
"""

import triton
import triton.language as tl

D = 128
H = 48

# (max tokens, meta) for the small-input specialist; checked in order.
SMALL = [
    (160, {'NT': 2, 'HB': 16, 'num_warps': 4}),   # B*S = 128, 131 -> 384 / 396 CTAs, 16 KB of x each
    (320, {'NT': 4, 'HB': 16, 'num_warps': 8}),   # B*S = 256      -> 384 CTAs, 32 KB of x each
]
# Parent g2-os-r8w4 configuration, used unchanged for every larger size.
PARENT = {'ROWS': 8, 'num_warps': 4, 'EVICT': True}


@triton.jit
def _qk_tokblk(q_ptr, k_ptr, wq_ptr, wk_ptr, qo_ptr, ko_ptr, n_tok, eps,
               NT: tl.constexpr, HB: tl.constexpr, D: tl.constexpr, H: tl.constexpr,
               MASKED: tl.constexpr):
    c = tl.program_id(0)
    g = tl.program_id(1)
    is_k = tl.program_id(2) == 1
    x_ptr = tl.where(is_k, k_ptr, q_ptr)
    w_ptr = tl.where(is_k, wk_ptr, wq_ptr)
    y_ptr = tl.where(is_k, ko_ptr, qo_ptr)

    tok = c * NT + tl.arange(0, NT)                      # [NT]
    heads = g * HB + tl.arange(0, HB)                    # [HB]
    cols = tl.arange(0, D)                               # [D]
    hc = heads[:, None] * D + cols[None, :]              # [HB, D] offsets inside one token (and into w)
    offs = (tok * (H * D))[:, None, None] + hc[None, :, :]   # [NT, HB, D]

    # Issue the DRAM stream first, then the (L2-resident) weight tile, once per program.
    if MASKED:
        m = tl.broadcast_to((tok < n_tok)[:, None, None], [NT, HB, D])
        x = tl.load(x_ptr + offs, mask=m, other=0.0, eviction_policy="evict_first")
    else:
        x = tl.load(x_ptr + offs, eviction_policy="evict_first")
    w = tl.load(w_ptr + hc, eviction_policy="evict_last")

    inv = tl.math.rsqrt(tl.sum(x * x, axis=2) * (1.0 / D) + eps)    # [NT, HB]; * (1/D) is exact
    y = (x * inv[:, :, None]) * w[None, :, :]

    if MASKED:
        tl.store(y_ptr + offs, y, mask=m, eviction_policy="evict_first")
    else:
        tl.store(y_ptr + offs, y, eviction_policy="evict_first")


@triton.jit
def _qk_oneshot(q_ptr, k_ptr, wq_ptr, wk_ptr, qo_ptr, ko_ptr, n_rows, eps,
                ROWS: tl.constexpr, D: tl.constexpr, H: tl.constexpr, EVICT: tl.constexpr):
    # Parent g2-os-r8w4 kernel, unchanged.
    pid = tl.program_id(0)
    n_tiles = n_rows // ROWS
    is_k = pid >= n_tiles
    tile = tl.where(is_k, pid - n_tiles, pid)
    x_ptr = tl.where(is_k, k_ptr, q_ptr)
    w_ptr = tl.where(is_k, wk_ptr, wq_ptr)
    y_ptr = tl.where(is_k, ko_ptr, qo_ptr)
    rows = tile * ROWS + tl.arange(0, ROWS)
    cols = tl.arange(0, D)
    offs = rows[:, None] * D + cols[None, :]
    w_offs = (rows % H)[:, None] * D + cols[None, :]
    if EVICT:
        x = tl.load(x_ptr + offs, eviction_policy="evict_first")
        w = tl.load(w_ptr + w_offs, eviction_policy="evict_last")
    else:
        x = tl.load(x_ptr + offs)
        w = tl.load(w_ptr + w_offs)
    inv = tl.math.rsqrt(tl.sum(x * x, axis=1) * (1.0 / D) + eps)
    y = (x * inv[:, None]) * w
    if EVICT:
        tl.store(y_ptr + offs, y, eviction_policy="evict_first")
    else:
        tl.store(y_ptr + offs, y)


def run(query, key, weight_q, weight_k, eps, query_norm, key_norm):
    n_rows = query.numel() // D
    n_tok = n_rows // H
    eps = float(eps)

    for limit, meta in SMALL:
        if n_tok <= limit:
            nt = meta['NT']
            hb = meta['HB']
            grid = (triton.cdiv(n_tok, nt), H // hb, 2)
            _qk_tokblk[grid](query, key, weight_q, weight_k, query_norm, key_norm, n_tok, eps,
                             NT=nt, HB=hb, D=D, H=H, MASKED=(n_tok % nt != 0),
                             num_warps=meta['num_warps'])
            return

    rows = PARENT['ROWS']
    _qk_oneshot[(2 * (n_rows // rows),)](query, key, weight_q, weight_k, query_norm, key_norm, n_rows, eps,
                                         ROWS=rows, D=D, H=H, EVICT=PARENT['EVICT'],
                                         num_warps=PARENT['num_warps'])
```
