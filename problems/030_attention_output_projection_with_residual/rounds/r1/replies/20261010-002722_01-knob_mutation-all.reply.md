The forced-algorithm table passes 16/16 workloads. It moves the L band by about 6% and two S sizes by about 8%. The M band cannot be improved by picking a different cuBLAS algorithm, so it misses the M=4096 target, and the H10 success bar is not met.

**What explains the 7–15% gap to plain matmul (attribution at M=2048 / 4096 / 8192, µs):**

| | plain matmul | fused, C≠D | C=D (diagnostic) | beta=0 |
|---|---|---|---|---|
| 2048 | 20.6 | 23.2 | 21.2 | 20.5 |
| 4096 | 35.7 | 38.5 | 36.4 | 35.6 |
| 8192 | 70.6 | 76.5 | 71.5 | 70.4 |

- **Reading the residual causes most of the gap.** With the same kernel, beta=0 runs at plain-matmul speed, and C=D costs only about 1 µs.
- **Algorithm choice explains part of it at L only.** The default 128x256 2x2 kernel handles the separate residual read badly. The 128x192 kernel with a 2x1 cluster gets M=8192 to 72.2 µs, within about 2.5% of plain matmul.
- **cuBLAS versions:** the extension uses torch's own cuBLASLt (`nvidia/cu13/lib`, `cublasLtGetVersion` = 130000), not the toolkit's 13.2.1. Torch loads it first, so the portal should behave the same way.

**Search:** I timed every heuristic result for all 10 sizes, plus 154 configs collected from proxy shapes at 4 sizes. Hand-built configs were 3–8% slower than the heuristic's own configs, so the table picks from the heuristic list. Split-K is about 2× slower at M=256, which refutes H12.

**Test results (rented B200):** 71.4–71.9 µs at L (5.6–5.9% below r0), 9.6 µs at M=512 (−8.4%), 11.4 µs at M=586 (−7.7%), M band unchanged. The predicted portal score is 0.412 ± 0.006, so the ≥ 0.43 target will most likely not be reached either. I have not checked that cuBLAS keeps the same choices at the portal's locked clock.

### Rationale
E2 measured where the cost of the fused cuBLASLt call comes from, then enumerated and timed algorithms per size. The candidate is r0 with a table keyed by M only: if the actual shape's heuristic list contains the chosen nvjet config (and AlgoCheck confirms it), that config is used; otherwise heuristic 0. It is still one kernel with no workspace, so every call launches the same activity sequence. Tested 16/16.

```cuda file=kernel.cu
#include <cuda_runtime.h>
#include <cublasLt.h>
#include <mutex>

// D[M,N] = A[M,K] * W[N,K]^T + R[M,N], all bf16 row-major, fp32 accumulate, one cublasLt matmul
// with C = residual and D = output (C != D is allowed by cublasLt, so the residual add is fused).
// cuBLAS is column-major: D^T[N,M] = op_T(W_col)[N,K] * op_N(A_col)[K,M] + R^T.
//
// Algorithm choice is keyed by M only. Where an nvjet config measurably beats heuristic 0 (rented
// B200, harness timing, interleaved repeats), we take the first heuristic result of the actual
// shape whose (TILE_ID, CLUSTER_SHAPE_ID) matches the table; otherwise heuristic 0.

namespace {
cublasLtHandle_t get_handle() {
    static cublasLtHandle_t h = nullptr;
    static std::once_flag once;
    std::call_once(once, [] { cublasLtCreate(&h); });
    return h;
}

// Returns the wanted (tile id, cluster shape id) for this M, or tile = 0 for heuristic 0.
void pick(int M, unsigned& tile, unsigned& cluster) {
    tile = 0; cluster = 0;
    if (M >= 6144) { tile = 32; cluster = 3; }        // nvjet 128x192 2cta, cluster 2x1: -8% at 7976/8192
    else if (M == 512) { tile = 183; cluster = 5; }   // nvjet 144x64 2cta, cluster 1x2: -8%
    else if (M == 586) { tile = 401; cluster = 3; }   // -4%
}
}  // namespace

extern "C" int oproj_launch(const void* A, const void* R, const void* W, void* D,
                             int M, int N, int K, void* ws, size_t ws_bytes, cudaStream_t stream) {
    cublasLtHandle_t handle = get_handle();
    cublasLtMatmulDesc_t op = nullptr;
    cublasLtMatrixLayout_t la = nullptr, lb = nullptr, lc = nullptr, ld = nullptr;
    cublasLtMatmulPreference_t pref = nullptr;
    cublasStatus_t st;
#define LT(x) st = (x); if (st != CUBLAS_STATUS_SUCCESS) goto done;
    LT(cublasLtMatmulDescCreate(&op, CUBLAS_COMPUTE_32F, CUDA_R_32F));
    {
        cublasOperation_t tA = CUBLAS_OP_T, tB = CUBLAS_OP_N;
        LT(cublasLtMatmulDescSetAttribute(op, CUBLASLT_MATMUL_DESC_TRANSA, &tA, sizeof(tA)));
        LT(cublasLtMatmulDescSetAttribute(op, CUBLASLT_MATMUL_DESC_TRANSB, &tB, sizeof(tB)));
    }
    // "A" operand = W viewed column-major: K rows x N cols, ld = K; transposed by the op.
    LT(cublasLtMatrixLayoutCreate(&la, CUDA_R_16BF, K, N, K));
    // "B" operand = attn viewed column-major: K rows x M cols, ld = K.
    LT(cublasLtMatrixLayoutCreate(&lb, CUDA_R_16BF, K, M, K));
    // C (residual) and D (output) viewed column-major: N rows x M cols, ld = N.
    LT(cublasLtMatrixLayoutCreate(&lc, CUDA_R_16BF, N, M, N));
    LT(cublasLtMatrixLayoutCreate(&ld, CUDA_R_16BF, N, M, N));
    LT(cublasLtMatmulPreferenceCreate(&pref));
    LT(cublasLtMatmulPreferenceSetAttribute(pref, CUBLASLT_MATMUL_PREF_MAX_WORKSPACE_BYTES, &ws_bytes, sizeof(ws_bytes)));
    {
        cublasLtMatmulHeuristicResult_t heur[32];
        int nret = 0;
        LT(cublasLtMatmulAlgoGetHeuristic(handle, op, la, lb, lc, ld, pref, 32, heur, &nret));
        if (nret == 0) { st = CUBLAS_STATUS_NOT_SUPPORTED; goto done; }
        int sel = 0;
        unsigned want_tile, want_cluster;
        pick(M, want_tile, want_cluster);
        if (want_tile) {
            for (int i = 0; i < nret; ++i) {
                uint32_t tile = 0, sk = 0; uint16_t cl = 0; size_t wr = 0;
                cublasLtMatmulAlgoConfigGetAttribute(&heur[i].algo, CUBLASLT_ALGO_CONFIG_TILE_ID, &tile, sizeof(tile), &wr);
                cublasLtMatmulAlgoConfigGetAttribute(&heur[i].algo, CUBLASLT_ALGO_CONFIG_CLUSTER_SHAPE_ID, &cl, sizeof(cl), &wr);
                cublasLtMatmulAlgoConfigGetAttribute(&heur[i].algo, CUBLASLT_ALGO_CONFIG_SPLITK_NUM, &sk, sizeof(sk), &wr);
                if (tile != want_tile || cl != want_cluster || sk > 1) continue;
                cublasLtMatmulHeuristicResult_t chk{};
                if (cublasLtMatmulAlgoCheck(handle, op, la, lb, lc, ld, &heur[i].algo, &chk) == CUBLAS_STATUS_SUCCESS &&
                    chk.workspaceSize <= ws_bytes) { sel = i; break; }
            }
        }
        const float alpha = 1.0f, beta = 1.0f;
        LT(cublasLtMatmul(handle, op, &alpha, W, la, A, lb, &beta, R, lc, D, ld,
                          &heur[sel].algo, ws, ws_bytes, stream));
    }
done:
    if (pref) cublasLtMatmulPreferenceDestroy(pref);
    if (ld) cublasLtMatrixLayoutDestroy(ld);
    if (lc) cublasLtMatrixLayoutDestroy(lc);
    if (lb) cublasLtMatrixLayoutDestroy(lb);
    if (la) cublasLtMatrixLayoutDestroy(la);
    if (op) cublasLtMatmulDescDestroy(op);
    return (int)st;
#undef LT
}
```

```cpp file=binding.cpp
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <c10/cuda/CUDAGuard.h>
#include <cuda_runtime.h>

extern "C" int oproj_launch(const void* A, const void* R, const void* W, void* D,
                             int M, int N, int K, void* ws, size_t ws_bytes, cudaStream_t stream);

namespace {
constexpr int64_t kWorkspace = 32ll << 20;  // 32 MB, the size cuBLAS recommends on Hopper/Blackwell
}

void run(torch::Tensor attn_output, torch::Tensor residual, torch::Tensor o_proj_weight, torch::Tensor output) {
    TORCH_CHECK(attn_output.is_cuda() && residual.is_cuda() && o_proj_weight.is_cuda() && output.is_cuda());
    TORCH_CHECK(attn_output.scalar_type() == at::kBFloat16 && residual.scalar_type() == at::kBFloat16 &&
                o_proj_weight.scalar_type() == at::kBFloat16 && output.scalar_type() == at::kBFloat16);
    TORCH_CHECK(attn_output.is_contiguous() && residual.is_contiguous() && o_proj_weight.is_contiguous() && output.is_contiguous());
    const int64_t K = attn_output.size(-1);
    const int64_t N = o_proj_weight.size(0);
    const int64_t M = attn_output.numel() / K;
    TORCH_CHECK(o_proj_weight.size(1) == K && residual.numel() == M * N && output.numel() == M * N);
    TORCH_CHECK(M <= INT32_MAX && (K % 8) == 0 && (N % 8) == 0);

    const c10::cuda::OptionalCUDAGuard guard(attn_output.device());
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    // Workspace from the caching allocator: no kernel is launched by this.
    auto ws = at::empty({kWorkspace}, attn_output.options().dtype(at::kByte));
    int st = oproj_launch(attn_output.data_ptr(), residual.data_ptr(), o_proj_weight.data_ptr(), output.data_ptr(),
                          (int)M, (int)N, (int)K, ws.data_ptr(), (size_t)kWorkspace, stream);
    TORCH_CHECK(st == 0, "cublasLt matmul failed with status ", st);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("run", &run, "o_proj + residual via cublasLt (bf16 in, fp32 accumulate), shape-keyed algorithm");
}
```
