### Rationale
I built H30 (lineage E) and it is refuted.
- **What works:** CUTLASS 4.4.1 sm100 builder GEMMs with an EVT epilogue (LN fold, exact-erf GELU, per-image GRN sum of squares as a non-final `Sm90RowReduction` with a custom square-add reducer) compile in 37 s. They are numerically right: TF32 operands read straight from fp32 memory round to nearest (H14 confirmed through CUTLASS).
- **The kill line fires:** cold G1 takes 17.0 µs at 8,768 and 20.0 µs at 8,384, against kill lines of 14 and 12 µs. The plain batched-per-image TF32 mainloop alone already takes 12.7 / 11.2 µs. Two reasons: per-image batching re-reads W1 once per image, and TF32 doubles the operand bytes. On top of that, the erf GELU node adds 2.4–8.5 µs.
- **Full pipeline (K1, G1, K3, G2, Kred):** 0 elements beyond tolerance over 10 seeded rounds on all six rows, max |err| 1.67–1.94e-3. It beats lineage A only at 2,768 (40.1 vs 48.5 µs), and D already holds that row at 32.4 µs.
- **run_tests:** correctness passed on all 16 rows in both runs, but 2 rows per run failed the harness's timing match, so this is not shippable as is.

```cuda file=ekern.cu
// Lineage E custom launches for the wide-C small-image rows (7x7 C=768, 14x14 C=384):
//  K1   bulk-plane depthwise conv + LayerNorm stats per token, d' = d*lnw token-major [b][hw][C], -mean and rstd per
//       token; spare warps fold u = W1.lnw, v = W1.lnb + b1, b2' = W2.gb + b2 while the planes land
//  G1   (g1.cu) CUTLASS TF32 GEMM with the LN-fold + GELU + per-image sum-of-squares EVT epilogue
//  K3   GRN scale in place: s = 1 + gw*gf/(mean_k gf + eps), one round of loads
//  G2   (g2.cu) CUTLASS TF32 split-K partials
//  Kred sum of partials + b2' + residual, transposed to NCHW
#include <cuda_runtime.h>
#include <stdint.h>
#include <stddef.h>

extern "C" int e_g1_launch(const float*, const float*, float*, const float*, const float*, const float*, const float*,
                           float*, int, int, int, int, int, void*, size_t, cudaStream_t);
extern "C" int e_g2_launch(const float*, const float*, float*, int, int, int, int, void*, size_t, cudaStream_t);

#define E_WSB (1u << 20)
static inline size_t e_al(size_t v) { return (v + 255) & ~(size_t)255; }
static int e_pick_ks(int B, int HW) {
  if (HW == 49) return B <= 2 ? 16 : (B <= 8 ? 6 : (B <= 16 ? 3 : 2));
  return B <= 8 ? 4 : 2;
}
struct ELayout { size_t d, negmu, rstd, u, v, b2p, hid, gf2p, P, ws1, ws2, total; int mstride, MT, KS; };
static ELayout e_layout(int B, int C, int H, int W) {
  ELayout L; const int HW = H * W, M = B * HW, N = 4 * C;
  L.mstride = (HW + 7) / 8 * 8; L.MT = (HW + 63) / 64; L.KS = e_pick_ks(B, HW);
  size_t o = 0;
  L.d = o; o = e_al(o + (size_t)M * C * 4);
  L.negmu = o; o = e_al(o + ((size_t)B * L.mstride + 128) * 4);
  L.rstd = o; o = e_al(o + ((size_t)B * L.mstride + 128) * 4);
  L.u = o; o = e_al(o + (size_t)N * 4);
  L.v = o; o = e_al(o + (size_t)N * 4);
  L.b2p = o; o = e_al(o + (size_t)C * 4);
  L.hid = o; o = e_al(o + (size_t)M * N * 4);
  L.gf2p = o; o = e_al(o + (size_t)B * L.MT * N * 4);
  L.P = o; o = e_al(o + (size_t)L.KS * M * C * 4);
  L.ws1 = o; o += E_WSB; L.ws2 = o; o += E_WSB;
  L.total = o;
  return L;
}
extern "C" int e_supported(int B, int C, int H, int W) {
  return (B >= 1 && B <= 32 && ((H == 7 && W == 7 && C == 768) || (H == 14 && W == 14 && C == 384))) ? 1 : 0;
}
extern "C" size_t e_ws_bytes(int B, int C, int H, int W) { return e_layout(B, C, H, W).total; }

struct E1P {
  const float *x, *dww, *dwb, *lnw, *lnb, *w1, *b1, *w2, *gb, *b2;
  float *d, *negmu, *rstd, *u, *v, *b2p;
  int C, H, HW, mstride, nconv, nbands, fold_all;
  float ln_eps;
};

__device__ __forceinline__ float e_wsum(float v) {
#pragma unroll
  for (int o = 16; o > 0; o >>= 1) v += __shfl_xor_sync(0xffffffffu, v, o);
  return v;
}
__device__ __forceinline__ uint32_t e_sa(const void* p) { return (uint32_t)__cvta_generic_to_shared(p); }
__device__ __forceinline__ void e_bulk(void* dst, const void* src, uint32_t bytes, uint32_t bar) {
  asm volatile("cp.async.bulk.shared::cluster.global.mbarrier::complete_tx::bytes [%0], [%1], %2, [%3];"
               :: "r"(e_sa(dst)), "l"(src), "r"(bytes), "r"(bar) : "memory");
}
__device__ __forceinline__ void e_wait(uint32_t bar, uint32_t ph) {
  asm volatile("{\n .reg .pred p;\n EWAIT:\n mbarrier.try_wait.parity.shared::cta.b64 p, [%0], %1;\n @!p bra EWAIT;\n}"
               :: "r"(bar), "r"(ph) : "memory");
}

// fold vectors, 2C units: unit < C is a b2' row (4C long, two halves), else 4 consecutive W1 rows (u and v).
// Every unit issues all of its loads before the first use (compile-time trip counts).
template <int CQ>
__device__ __forceinline__ void e_fold(const E1P& p, int gwarp, int nw, int lane) {
  constexpr int C = CQ * 128, C4 = 4 * C;
  for (int u = gwarp; u < 2 * C; u += nw) {
    if (u < C) {
      const float4* wr = reinterpret_cast<const float4*>(p.w2 + (size_t)u * C4);
      const float4* g4 = reinterpret_cast<const float4*>(p.gb);
      float s = 0.f;
#pragma unroll
      for (int hf = 0; hf < 2; hf++) {
        float4 a[2 * CQ], g[2 * CQ];
#pragma unroll
        for (int i = 0; i < 2 * CQ; i++) {
          a[i] = __ldg(wr + hf * 64 * CQ + lane + 32 * i);
          g[i] = __ldg(g4 + hf * 64 * CQ + lane + 32 * i);
        }
#pragma unroll
        for (int i = 0; i < 2 * CQ; i++) s += a[i].x * g[i].x + a[i].y * g[i].y + a[i].z * g[i].z + a[i].w * g[i].w;
      }
      s = e_wsum(s);
      if (lane == 0) p.b2p[u] = s + __ldg(p.b2 + u);
    } else {
      const int n0 = (u - C) * 4;
      const float4* l4 = reinterpret_cast<const float4*>(p.lnw);
      const float4* b4 = reinterpret_cast<const float4*>(p.lnb);
      float4 a[4][CQ], l[CQ], bb[CQ];
#pragma unroll
      for (int i = 0; i < CQ; i++) { l[i] = __ldg(l4 + lane + 32 * i); bb[i] = __ldg(b4 + lane + 32 * i); }
#pragma unroll
      for (int r = 0; r < 4; r++)
#pragma unroll
        for (int i = 0; i < CQ; i++) a[r][i] = __ldg(reinterpret_cast<const float4*>(p.w1 + (size_t)(n0 + r) * C) + lane + 32 * i);
#pragma unroll
      for (int r = 0; r < 4; r++) {
        float s1 = 0.f, s2 = 0.f;
#pragma unroll
        for (int i = 0; i < CQ; i++) {
          s1 += a[r][i].x * l[i].x + a[r][i].y * l[i].y + a[r][i].z * l[i].z + a[r][i].w * l[i].w;
          s2 += a[r][i].x * bb[i].x + a[r][i].y * bb[i].y + a[r][i].z * bb[i].z + a[r][i].w * bb[i].w;
        }
        s1 = e_wsum(s1); s2 = e_wsum(s2);
        if (lane == 0) { p.u[n0 + r] = s1; p.v[n0 + r] = s2 + __ldg(p.b1 + n0 + r); }
      }
    }
  }
}

// W: image width (= H); NRW: plane rows staged per channel; SLOT: floats per channel slot (odd multiple of 4 for 14x14
// to spread banks); R: output rows per CTA; CQ = C / 128
template <int W, int NRW, int SLOT, int R, int CQ>
__global__ void __launch_bounds__(256, 1) e_k1(E1P p) {
  extern __shared__ float4 e_sm4[];
  __shared__ __align__(8) unsigned long long e_bar;
  constexpr int C = CQ * 128, CP = C + 4;
  float* planes = reinterpret_cast<float*>(e_sm4);
  const int tid = threadIdx.x, warp = tid >> 5, lane = tid & 31;
  const int H = p.H, HW = p.HW;
  const bool conv = (int)blockIdx.x < p.nconv;
  const int b = conv ? (int)blockIdx.x / p.nbands : 0;
  const int h0 = conv ? ((int)blockIdx.x % p.nbands) * R : 0;
  int hlo = 0;
  if (NRW < W) { hlo = (h0 - 3) & ~1; hlo = hlo < 0 ? 0 : hlo; hlo = hlo > H - NRW ? H - NRW : hlo; }
  const uint32_t bar = e_sa(&e_bar);
  if (conv) {
    if (tid == 0) {
      asm volatile("mbarrier.init.shared::cta.b64 [%0], 1;" :: "r"(bar));
      asm volatile("fence.mbarrier_init.release.cluster;" ::: "memory");
    }
    __syncthreads();
    if (warp == 0) {
      const uint32_t total = (NRW == W) ? (uint32_t)C * HW * 4u : (uint32_t)C * NRW * W * 4u;
      if (lane == 0) asm volatile("mbarrier.arrive.expect_tx.shared::cta.b64 _, [%0], %1;" :: "r"(bar), "r"(total) : "memory");
      __syncwarp();
      if (NRW == W) {
        constexpr int cpl = C / 32;
        e_bulk(planes + (size_t)lane * cpl * SLOT, p.x + ((size_t)b * C + lane * cpl) * HW, (uint32_t)cpl * HW * 4u, bar);
      } else {
        for (int c = lane; c < C; c += 32)
          e_bulk(planes + (size_t)c * SLOT, p.x + ((size_t)b * C + c) * HW + hlo * W, (uint32_t)NRW * W * 4u, bar);
      }
    }
  }
  if (p.fold_all) e_fold<CQ>(p, (int)blockIdx.x * 8 + warp, (int)gridDim.x * 8, lane);
  else if (!conv) e_fold<CQ>(p, ((int)blockIdx.x - p.nconv) * 8 + warp, ((int)gridDim.x - p.nconv) * 8, lane);
  if (!conv) return;
  float wt[49], wn[49];
#pragma unroll
  for (int i = 0; i < 49; i++) wt[i] = __ldg(p.dww + (size_t)tid * 49 + i);
  e_wait(bar, 0);
  float* outs = planes + (size_t)C * SLOT;
  for (int c = tid; c < C; c += 256) {
    const int cn = c + 256;
    if (cn < C) {
#pragma unroll
      for (int i = 0; i < 49; i++) wn[i] = __ldg(p.dww + (size_t)cn * 49 + i);
    }
    const float bias = __ldg(p.dwb + c);
    const float* pl = planes + (size_t)c * SLOT;
#pragma unroll
    for (int r = 0; r < R; r++) {
      const int h = h0 + r;
      if (h < H) {
        float acc[W];
#pragma unroll
        for (int w = 0; w < W; w++) acc[w] = 0.f;
#pragma unroll
        for (int i = 0; i < 7; i++) {
          const int hh = h - 3 + i;
          if (hh >= 0 && hh < H) {
            const float* rp = pl + (hh - hlo) * W;
            float vv[W + 6];
#pragma unroll
            for (int j = 0; j < 3; j++) { vv[j] = 0.f; vv[W + 3 + j] = 0.f; }
#pragma unroll
            for (int j = 0; j < W; j++) vv[j + 3] = rp[j];
#pragma unroll
            for (int w = 0; w < W; w++)
#pragma unroll
              for (int j = 0; j < 7; j++) acc[w] = fmaf(vv[w + j], wt[i * 7 + j], acc[w]);
          }
        }
        float* o = outs + (size_t)(r * W) * CP + c;
#pragma unroll
        for (int w = 0; w < W; w++) o[(size_t)w * CP] = acc[w] + bias;
      }
    }
    if (cn < C) {
#pragma unroll
      for (int i = 0; i < 49; i++) wt[i] = wn[i];
    }
  }
  __syncthreads();
  const float invC = 1.f / (float)C;
  for (int t = warp; t < R * W; t += 8) {
    const int r = t / W, h = h0 + r;
    if (h >= H) continue;
    const int tok = h * W + (t - r * W);
    const float4* row = reinterpret_cast<const float4*>(outs + (size_t)t * CP);
    float4 vr[CQ], lw[CQ];
    float s = 0.f, ss = 0.f;
#pragma unroll
    for (int i = 0; i < CQ; i++) {
      vr[i] = row[lane + 32 * i];
      lw[i] = __ldg(reinterpret_cast<const float4*>(p.lnw) + lane + 32 * i);
      s += (vr[i].x + vr[i].y) + (vr[i].z + vr[i].w);
      ss += vr[i].x * vr[i].x + vr[i].y * vr[i].y + vr[i].z * vr[i].z + vr[i].w * vr[i].w;
    }
    s = e_wsum(s); ss = e_wsum(ss);
    const float mean = s * invC;
    const float var = fmaxf(ss * invC - mean * mean, 0.f);
    float4* dst = reinterpret_cast<float4*>(p.d + ((size_t)b * HW + tok) * C);
#pragma unroll
    for (int i = 0; i < CQ; i++)
      dst[lane + 32 * i] = make_float4(vr[i].x * lw[i].x, vr[i].y * lw[i].y, vr[i].z * lw[i].z, vr[i].w * lw[i].w);
    if (lane == 0) { p.negmu[b * p.mstride + tok] = -mean; p.rstd[b * p.mstride + tok] = rsqrtf(var + p.ln_eps); }
  }
}

// NTT: token slots per thread (16 token lanes), MT: m-tile partials per image, NQ: float4 groups per thread for the mean
template <int NTT, int MT, int NQ>
__global__ void __launch_bounds__(256) e_k3(const float* __restrict__ gf2p, const float* __restrict__ gw, float* hid, int HW,
                                            int N, float eps) {
  __shared__ float red[8];
  const int tid = threadIdx.x, kq = tid & 15, tl = tid >> 4;
  const int b = blockIdx.y, k0 = blockIdx.x * 64 + kq * 4;
  const float4* G = reinterpret_cast<const float4*>(gf2p + (size_t)b * MT * N);
  const int N4 = N / 4;
  float4 own[MT], all[NQ][MT], hv[NTT];
#pragma unroll
  for (int m = 0; m < MT; m++) own[m] = __ldg(G + m * N4 + k0 / 4);
#pragma unroll
  for (int q = 0; q < NQ; q++) {
    const int idx = tid + 256 * q;
#pragma unroll
    for (int m = 0; m < MT; m++) all[q][m] = (idx < N4) ? __ldg(G + m * N4 + idx) : make_float4(0.f, 0.f, 0.f, 0.f);
  }
  float* hb = hid + (size_t)b * HW * N + k0;
#pragma unroll
  for (int i = 0; i < NTT; i++) {
    const int t = tl + 16 * i;
    if (t < HW) hv[i] = *reinterpret_cast<const float4*>(hb + (size_t)t * N);
  }
  const float4 g4 = __ldg(reinterpret_cast<const float4*>(gw + k0));
  float s = 0.f;
#pragma unroll
  for (int q = 0; q < NQ; q++) {
    float4 a = all[q][0];
#pragma unroll
    for (int m = 1; m < MT; m++) { a.x += all[q][m].x; a.y += all[q][m].y; a.z += all[q][m].z; a.w += all[q][m].w; }
    s += (sqrtf(a.x) + sqrtf(a.y)) + (sqrtf(a.z) + sqrtf(a.w));
  }
  s = e_wsum(s);
  if ((tid & 31) == 0) red[tid >> 5] = s;
  __syncthreads();
  float tot = 0.f;
#pragma unroll
  for (int w = 0; w < 8; w++) tot += red[w];
  const float inv = 1.f / (tot / (float)N + eps);
  float4 o = own[0];
#pragma unroll
  for (int m = 1; m < MT; m++) { o.x += own[m].x; o.y += own[m].y; o.z += own[m].z; o.w += own[m].w; }
  const float s0 = fmaf(g4.x * inv, sqrtf(o.x), 1.f), s1 = fmaf(g4.y * inv, sqrtf(o.y), 1.f);
  const float s2 = fmaf(g4.z * inv, sqrtf(o.z), 1.f), s3 = fmaf(g4.w * inv, sqrtf(o.w), 1.f);
#pragma unroll
  for (int i = 0; i < NTT; i++) {
    const int t = tl + 16 * i;
    if (t < HW)
      *reinterpret_cast<float4*>(hb + (size_t)t * N) = make_float4(hv[i].x * s0, hv[i].y * s1, hv[i].z * s2, hv[i].w * s3);
  }
}

// out[b][c][hw] = x + b2'[c] + sum_ks P[ks][b*HW+hw][c]; CTA = (32 channels, image); all loads issued before use
template <int KS, int NTT2, int NX>
__global__ void __launch_bounds__(256) e_kred(const float* __restrict__ P, const float* __restrict__ x,
                                              const float* __restrict__ b2p, float* __restrict__ out, int C, int HW, int M) {
  __shared__ float st[32 * 197];
  const int tid = threadIdx.x, cq = tid & 7, tl = tid >> 3;
  const int c0 = blockIdx.x * 32, b = blockIdx.y;
  const size_t base = ((size_t)b * C + c0) * HW;
  const int tot = 32 * HW, HP = HW + 1;
  float xv[NX], bv[NX];
#pragma unroll
  for (int i = 0; i < NX; i++) {
    const int j = tid + 256 * i;
    xv[i] = (j < tot) ? __ldg(x + base + j) : 0.f;
    bv[i] = (j < tot) ? __ldg(b2p + c0 + j / HW) : 0.f;
  }
  float4 pv[KS][NTT2];
#pragma unroll
  for (int ks = 0; ks < KS; ks++)
#pragma unroll
    for (int i = 0; i < NTT2; i++) {
      const int t = tl + 32 * i;
      pv[ks][i] = (t < HW) ? __ldg(reinterpret_cast<const float4*>(P + ((size_t)ks * M + (size_t)b * HW + t) * C + c0 + cq * 4))
                           : make_float4(0.f, 0.f, 0.f, 0.f);
    }
#pragma unroll
  for (int i = 0; i < NTT2; i++) {
    float4 a = pv[0][i];
#pragma unroll
    for (int ks = 1; ks < KS; ks++) { a.x += pv[ks][i].x; a.y += pv[ks][i].y; a.z += pv[ks][i].z; a.w += pv[ks][i].w; }
    const int t = tl + 32 * i;
    if (t < HW) {
      st[(cq * 4 + 0) * HP + t] = a.x; st[(cq * 4 + 1) * HP + t] = a.y;
      st[(cq * 4 + 2) * HP + t] = a.z; st[(cq * 4 + 3) * HP + t] = a.w;
    }
  }
  __syncthreads();
#pragma unroll
  for (int i = 0; i < NX; i++) {
    const int j = tid + 256 * i;
    if (j < tot) {
      const int c = j / HW, hw = j - c * HW;
      out[base + j] = xv[i] + bv[i] + st[c * HP + hw];
    }
  }
}

  if (cudaFuncSetAttribute(kern, cudaFuncAttributeMaxDynamicSharedMemorySize, 200 * 1024) != cudaSuccess) return "smem attr";

}

  return nullptr;
}
```

```cuda file=kernel.cu
// ConvNeXtV2 block with GRN: one cooperative launch, three phases, fp32 SIMT maths.
//  P1: depthwise 7x7 conv (+bias) -> xn [C][Mp] (K-major), per-(token, 32-channel chunk) LN partial sums;
//      zero GRN sums; transpose weights to [K][N]
//  P2: hidden[4C][Mp] = gelu(LN(xn) @ W1^T + b1); sumsq[b][k] += hidden^2 (per image)
//  P3: out = x + (hidden * (1 + gw*nf) + gb) @ W2^T + b2, written NCHW (split-K with atomics at small sizes)
#include <cuda_runtime.h>
#include <cooperative_groups.h>
namespace cg = cooperative_groups;

#define NT 256
#define MAXIMG 5
#define GEMM_SMEM_FLOATS (128 * 132)
#define SMEM_BYTES (GEMM_SMEM_FLOATS * 4 + MAXIMG * 128 * 4 + 64)
#define P1_SLOTS 512

struct Params {
  const float* x; const float* dww; const float* dwb; const float* lnw; const float* lnb;
  const float* w1; const float* b1; const float* gw; const float* gb; const float* w2; const float* b2;
  float* out;
  float* xn; float* part; float* hidden; float* sumsq; float* w1t; float* w2t;
  int B, C, H, W, HW, M, Mp, C4, nch;
  float eps, ln_eps;
  int p1_mode, p1_twr, p1_tiles, p1_lpr, p1_rg, p1_hb, p1_wb;
  int cfg2, cfg3, split3;
};

__device__ __forceinline__ float gelu_f(float v) { return 0.5f * v * (1.f + erff(v * 0.70710678118654752f)); }

__device__ __forceinline__ float warp_sum(float v) {
#pragma unroll
  for (int o = 16; o > 0; o >>= 1) v += __shfl_xor_sync(0xffffffffu, v, o);
  return v;
}

// ---------------------------------------------------------------- phase 1
__device__ void transpose_mat(const float* __restrict__ src, int R, int Cc, float* __restrict__ dst, float* tile) {
  const int rt = (R + 31) / 32, ct = (Cc + 31) / 32;
  const int tid = threadIdx.x, j = tid & 31, i0 = tid >> 5;
  for (int t = blockIdx.x; t < rt * ct; t += gridDim.x) {
    const int tr = t / ct, tc = t % ct;
#pragma unroll
    for (int s = 0; s < 4; s++) {
      const int i = i0 + 8 * s, r = tr * 32 + i, c = tc * 32 + j;
      tile[i * 33 + j] = (r < R && c < Cc) ? src[(size_t)r * Cc + c] : 0.f;
    }
    __syncthreads();
#pragma unroll
    for (int s = 0; s < 4; s++) {
      const int i = i0 + 8 * s, c = tc * 32 + i, r = tr * 32 + j;
      if (r < R && c < Cc) dst[(size_t)c * R + r] = tile[j * 33 + i];
    }
    __syncthreads();
  }
}

// 4 rows x TW cols output patch of one channel; wt = 49 weights (registers)
template <int TW>
__device__ __forceinline__ void conv_patch(const float* __restrict__ xc, const float* wt, int H, int W, int h0, int w0,
                                           float (&acc)[4][TW]) {
#pragma unroll
  for (int r = 0; r < 4; r++)
#pragma unroll
    for (int cc = 0; cc < TW; cc++) acc[r][cc] = 0.f;
#pragma unroll
  for (int i = 0; i < 10; i++) {
    const int hh = h0 - 3 + i;
    const bool rowok = (hh >= 0) && (hh < H);
    float v[TW + 6];
    if (TW == 4) {
      const float* rp = xc + hh * W;
      float4 L = make_float4(0.f, 0.f, 0.f, 0.f), Cq = L, R = L;
      if (rowok) {
        if (w0 >= 4) L = __ldg(reinterpret_cast<const float4*>(rp + w0 - 4));
        Cq = __ldg(reinterpret_cast<const float4*>(rp + w0));
        if (w0 + 4 < W) R = __ldg(reinterpret_cast<const float4*>(rp + w0 + 4));
      }
      v[0] = L.y; v[1] = L.z; v[2] = L.w; v[3] = Cq.x; v[4] = Cq.y; v[5] = Cq.z; v[6] = Cq.w;
      v[7] = R.x; v[8] = R.y; v[9] = R.z;
    } else {
#pragma unroll
      for (int jj = 0; jj < TW + 6; jj++) {
        const int ww = w0 - 3 + jj;
        v[jj] = (rowok && ww >= 0 && ww < W) ? __ldg(xc + hh * W + ww) : 0.f;
      }
    }
#pragma unroll
    for (int r = 0; r < 4; r++) {
      const int ki = i - r;
      if (ki >= 0 && ki < 7) {
#pragma unroll
        for (int cc = 0; cc < TW; cc++)
#pragma unroll
          for (int j = 0; j < 7; j++) acc[r][cc] = fmaf(v[cc + j], wt[ki * 7 + j], acc[r][cc]);
      }
    }
  }
}

// MODE 1: patch tiles (TWR = TW in {4,1}); MODE 0: flattened token tiles (TWR = tokens per lane in {1,2,4})
template <int MODE, int TWR>
__device__ void phase1_conv(const Params& p, float* smem) {
  const int tid = threadIdx.x, warp = tid >> 5, lane = tid & 31;
  const int C = p.C, H = p.H, W = p.W, HW = p.HW, M = p.M, Mp = p.Mp, nch = p.nch;
  float* st_s = smem;                     // [8][P1_SLOTS][2]
  float* wsm = smem + 8 * P1_SLOTS * 2;   // [32][49]
  const int lpr = p.p1_lpr, rg = p.p1_rg, hb = p.p1_hb, wb = p.p1_wb;
  const int rows_per_tile = 4 * rg;
  const int rpt = lpr * TWR;  // slots per row (mode 1)

  for (int tile = blockIdx.x; tile < p.p1_tiles; tile += gridDim.x) {
    int b = 0, cch = 0, hbi = 0, wbi = 0, tok0 = 0;
    if (MODE == 1) {
      const int per = nch * hb * wb;
      b = tile / per; int rem = tile - b * per;
      cch = rem / (hb * wb); rem -= cch * (hb * wb);
      hbi = rem / wb; wbi = rem - hbi * wb;
    } else {
      cch = tile % nch; tok0 = (tile / nch) * (32 * TWR);
    }
    for (int i = tid; i < 8 * P1_SLOTS * 2; i += NT) st_s[i] = 0.f;
    for (int i = tid; i < 32 * 49; i += NT) wsm[i] = __ldg(p.dww + (size_t)cch * 32 * 49 + i);
    __syncthreads();

    int rgi = 0, cgi = 0, h0 = 0, w0 = 0; bool lane_ok = true;
    if (MODE == 1) {
      rgi = lane / lpr; cgi = lane - rgi * lpr;
      lane_ok = rgi < rg;
      h0 = hbi * rows_per_tile + rgi * 4;
      w0 = (wbi * 32 + cgi) * TWR;
      lane_ok = lane_ok && (w0 < W);
    }
    int tk[MODE == 0 ? TWR : 1], tb[MODE == 0 ? TWR : 1], th[MODE == 0 ? TWR : 1], tw_[MODE == 0 ? TWR : 1];
    bool tv[MODE == 0 ? TWR : 1];
    if (MODE == 0) {
#pragma unroll
      for (int r = 0; r < TWR; r++) {
        const int t = tok0 + r * 32 + lane;
        tv[r] = t < M;
        const int tt = tv[r] ? t : 0;
        const int bb = tt / HW; const int rem = tt - bb * HW; const int hh = rem / W;
        tk[r] = t; tb[r] = bb; th[r] = hh; tw_[r] = rem - hh * W;
      }
    }

#pragma unroll 1
    for (int cl = 0; cl < 4; cl++) {
      const int clc = warp * 4 + cl;
      const int c = cch * 32 + clc;
      float wt[49];
#pragma unroll
      for (int i = 0; i < 49; i++) wt[i] = wsm[clc * 49 + i];
      const float bias = __ldg(p.dwb + c);
      float* xrow = p.xn + (size_t)c * Mp;
      float* sts = st_s + warp * P1_SLOTS * 2;
      if (MODE == 1) {
        if (lane_ok) {
          float acc[4][TWR];
          conv_patch<TWR>(p.x + ((size_t)b * C + c) * HW, wt, H, W, h0, w0, acc);
#pragma unroll
          for (int r = 0; r < 4; r++) {
            const int h = h0 + r;
            if (h < H) {
              const int t = b * HW + h * W + w0;
              const int slot = (rgi * 4 + r) * rpt + cgi * TWR;
              if (TWR == 4) {
                float4 o = make_float4(acc[r][0] + bias, acc[r][1] + bias, acc[r][2] + bias, acc[r][3] + bias);
                *reinterpret_cast<float4*>(xrow + t) = o;
                sts[(slot + 0) * 2] += o.x; sts[(slot + 0) * 2 + 1] += o.x * o.x;
                sts[(slot + 1) * 2] += o.y; sts[(slot + 1) * 2 + 1] += o.y * o.y;
                sts[(slot + 2) * 2] += o.z; sts[(slot + 2) * 2 + 1] += o.z * o.z;
                sts[(slot + 3) * 2] += o.w; sts[(slot + 3) * 2 + 1] += o.w * o.w;
              } else {
#pragma unroll
                for (int cc = 0; cc < TWR; cc++) {
                  const float o = acc[r][cc] + bias;
                  xrow[t + cc] = o;
                  sts[(slot + cc) * 2] += o; sts[(slot + cc) * 2 + 1] += o * o;
                }
              }
            }
          }
        }
      } else {
        float acc[TWR];
#pragma unroll
        for (int r = 0; r < TWR; r++) acc[r] = 0.f;
#pragma unroll
        for (int i = 0; i < 7; i++)
#pragma unroll
          for (int j = 0; j < 7; j++) {
            const float wv = wt[i * 7 + j];
#pragma unroll
            for (int r = 0; r < TWR; r++) {
              const int hh = th[r] - 3 + i, ww = tw_[r] - 3 + j;
              if (tv[r] && hh >= 0 && hh < H && ww >= 0 && ww < W)
                acc[r] = fmaf(__ldg(p.x + ((size_t)tb[r] * C + c) * HW + hh * W + ww), wv, acc[r]);
            }
          }
#pragma unroll
        for (int r = 0; r < TWR; r++) {
          if (tv[r]) {
            const float o = acc[r] + bias;
            xrow[tk[r]] = o;
            const int slot = r * 32 + lane;
            sts[slot * 2] += o; sts[slot * 2 + 1] += o * o;
          }
        }
      }
    }
    __syncthreads();
    const int nslots = (MODE == 1) ? rows_per_tile * rpt : 32 * TWR;
    for (int slot = tid; slot < nslots; slot += NT) {
      int t; bool valid;
      if (MODE == 1) {
        const int row = slot / rpt, col = slot - row * rpt;
        const int h = hbi * rows_per_tile + row, w = wbi * 32 * TWR + col;
        valid = (h < H) && (w < W);
        t = b * HW + h * W + w;
      } else {
        t = tok0 + slot; valid = t < M;
      }
      if (valid) {
        float S1 = 0.f, S2 = 0.f;
#pragma unroll
        for (int ws = 0; ws < 8; ws++) { S1 += st_s[(ws * P1_SLOTS + slot) * 2]; S2 += st_s[(ws * P1_SLOTS + slot) * 2 + 1]; }
        float* pp = p.part + ((size_t)cch * Mp + t) * 2;
        pp[0] = S1; pp[1] = S2;
      }
    }
    __syncthreads();
  }
}

__device__ void phase1(const Params& p, float* smem) {
  const int tid = threadIdx.x;
  for (int i = blockIdx.x * NT + tid; i < p.B * p.C4; i += gridDim.x * NT) p.sumsq[i] = 0.f;
  transpose_mat(p.w1, p.C4, p.C, p.w1t, smem);
  transpose_mat(p.w2, p.C, p.C4, p.w2t, smem);
  if (p.p1_mode == 1) {
    if (p.p1_twr == 4) phase1_conv<1, 4>(p, smem); else phase1_conv<1, 1>(p, smem);
  } else {
    if (p.p1_twr == 4) phase1_conv<0, 4>(p, smem);
    else if (p.p1_twr == 2) phase1_conv<0, 2>(p, smem);
    else phase1_conv<0, 1>(p, smem);
  }
}

// ---------------------------------------------------------------- GEMM phases
// A = act [K][Mp] (K-major), B = Wt [K][N]. 8 warps = WM x WN spatial warps x KS K-groups. Lane grid 8 x 4.
template <int BM, int BN, int WM, int WN, int KS, int BKc, int PHASE>
__device__ void gemm_phase(const Params& p, float* smem, float* sq_s, float* inv_s, float* red_s, int split) {
  constexpr int WS = WM * WN;
  constexpr int GT = WS * 32;
  constexpr int TM = BM / (WM * 8), TN = BN / (WN * 4);
  constexpr int AM4 = BM / 4, BN4 = BN / 4;
  constexpr int AQ = (BKc * AM4) / GT, BQ = (BKc * BN4) / GT;
  constexpr int STAGE = BKc * (BM + BN);
  constexpr int CS_LD = BM + 4;
  constexpr int NSTEP = NT / BM, NPT = BN / NSTEP;
  static_assert(WS * KS == 8, "8 warps");
  static_assert((BKc * AM4) % GT == 0 && (BKc * BN4) % GT == 0, "quad mapping");
  static_assert(GT % AM4 == 0, "token quad constant per thread");
  static_assert(KS * 2 * STAGE <= GEMM_SMEM_FLOATS && KS * BN * CS_LD <= GEMM_SMEM_FLOATS, "smem");
  static_assert(TM == 8 || TM == 4, "TM");
  static_assert(TN == 8 || TN == 6, "TN");

  const int tid = threadIdx.x, warp = tid >> 5, lane = tid & 31;
  const int kg = warp / WS, ws = warp - kg * WS, wm = ws / WN, wn = ws - wm * WN;
  const int lm = lane >> 2, ln = lane & 3;
  const int gt = ws * 32 + lane;
  const int M = p.M, Mp = p.Mp, HW = p.HW;
  const int N = (PHASE == 2) ? p.C4 : p.C;
  const int K = (PHASE == 2) ? p.C : p.C4;
  const float* __restrict__ Ag = (PHASE == 2) ? p.xn : p.hidden;
  const float* __restrict__ Bg = (PHASE == 2) ? p.w1t : p.w2t;
  const int mt = (M + BM - 1) / BM, nt = (N + BN - 1) / BN;
  const int ntiles = mt * nt * split, nk = K / BKc;
  float* As = smem + kg * 2 * STAGE;
  float* Bs = As + 2 * BKc * BM;
  const int row_m = wm * (BM / WM) + lm * TM;
  const int col_n = wn * (BN / WN) + ln * TN;
  const int a_m4 = (gt % AM4) * 4;
  const int a_kq = gt / AM4;
  constexpr int KSTEP_A = GT / AM4;

  for (int tile = blockIdx.x; tile < ntiles; tile += gridDim.x) {
    int rem = tile;
    const int ni = rem % nt; rem /= nt;
    const int mi = rem % mt; const int s = rem / mt;
    const int m0 = mi * BM, n0 = ni * BN;
    const int c0 = (int)(((long)nk * s) / split), c1 = (int)(((long)nk * (s + 1)) / split);
    const int b_first = m0 / HW;
    const int nimg = min(MAXIMG, (min(m0 + BM, M) - 1) / HW - b_first + 1);

    // per-thread token quad constants
    float tc0[4], tc1[4]; int timg[4]; bool tval[4];
#pragma unroll
    for (int e = 0; e < 4; e++) {
      const int t = m0 + a_m4 + e;
      tval[e] = t < M;
      const int tt = tval[e] ? t : 0;
      if (PHASE == 2) {
        float S1 = 0.f, S2 = 0.f;
        for (int ch = 0; ch < p.nch; ch++) {
          const float2 pp = __ldg(reinterpret_cast<const float2*>(p.part + ((size_t)ch * Mp + tt) * 2));
          S1 += pp.x; S2 += pp.y;
        }
        const float mean = S1 / (float)p.C;
        float var = S2 / (float)p.C - mean * mean; var = fmaxf(var, 0.f);
        tc0[e] = mean; tc1[e] = rsqrtf(var + p.ln_eps); timg[e] = 0;
      } else {
        timg[e] = tt / HW - b_first; tc0[e] = 0.f; tc1[e] = 0.f;
      }
    }
    if (PHASE == 3) {
      for (int img = 0; img < nimg; img++) {
        const float* sq = p.sumsq + (size_t)(b_first + img) * K;
        float sacc = 0.f;
        for (int k = tid; k < K; k += NT) sacc += sqrtf(sq[k]);
        sacc = warp_sum(sacc);
        if (lane == 0) red_s[warp] = sacc;
        __syncthreads();
        if (tid == 0) {
          float tot = 0.f;
#pragma unroll
          for (int w2 = 0; w2 < 8; w2++) tot += red_s[w2];
          inv_s[img] = 1.f / (tot / (float)K + p.eps);
        }
        __syncthreads();
      }
#pragma unroll
      for (int e = 0; e < 4; e++) tc1[e] = inv_s[timg[e] < MAXIMG ? timg[e] : 0];
    }

    float acc[TM][TN];
#pragma unroll
    for (int i = 0; i < TM; i++)
#pragma unroll
      for (int j = 0; j < TN; j++) acc[i][j] = 0.f;

    float4 ra[AQ], rb[BQ];
    auto load_chunk = [&](int kc) {
      const int k0 = kc * BKc;
#pragma unroll
      for (int q = 0; q < AQ; q++) {
        const int ka = k0 + q * KSTEP_A + a_kq;
        float4 v = __ldg(reinterpret_cast<const float4*>(Ag + (size_t)ka * Mp + m0 + a_m4));
        float vv[4] = {v.x, v.y, v.z, v.w};
        if (PHASE == 2) {
          const float lw = __ldg(p.lnw + ka), lb = __ldg(p.lnb + ka);
#pragma unroll
          for (int e = 0; e < 4; e++) vv[e] = tval[e] ? fmaf((vv[e] - tc0[e]) * tc1[e], lw, lb) : 0.f;
        } else {
          const float gwk = __ldg(p.gw + ka), gbk = __ldg(p.gb + ka);
#pragma unroll
          for (int e = 0; e < 4; e++) {
            const float sq = __ldg(p.sumsq + (size_t)(b_first + timg[e]) * K + ka);
            vv[e] = tval[e] ? fmaf(vv[e], fmaf(gwk * tc1[e], sqrtf(sq), 1.f), gbk) : 0.f;
          }
        }
        ra[q] = make_float4(vv[0], vv[1], vv[2], vv[3]);
      }
#pragma unroll
      for (int q = 0; q < BQ; q++) {
        const int id = q * GT + gt, kb = id / BN4, n4 = (id - kb * BN4) * 4;
        const int col = n0 + n4;
        rb[q] = (col < N) ? __ldg(reinterpret_cast<const float4*>(Bg + (size_t)(k0 + kb) * N + col))
                          : make_float4(0.f, 0.f, 0.f, 0.f);
      }
    };
    auto store_regs = [&](int buf) {
#pragma unroll
      for (int q = 0; q < AQ; q++)
        *reinterpret_cast<float4*>(As + (buf * BKc + q * KSTEP_A + a_kq) * BM + a_m4) = ra[q];
#pragma unroll
      for (int q = 0; q < BQ; q++) {
        const int id = q * GT + gt, kb = id / BN4, n4 = (id - kb * BN4) * 4;
        *reinterpret_cast<float4*>(Bs + (buf * BKc + kb) * BN + n4) = rb[q];
      }
    };
    auto gsync = [&]() {
      if (KS == 1) __syncthreads();
      else if (GT == 32) __syncwarp();
      else asm volatile("bar.sync %0, %1;" ::"r"(kg + 1), "r"(GT));
    };

    int kc = c0 + kg;
    if (kc < c1) { load_chunk(kc); store_regs(0); }
    gsync();
    for (int it = 0; kc < c1; it++, kc += KS) {
      const int buf = it & 1;
      const bool more = (kc + KS) < c1;
      if (more) load_chunk(kc + KS);
      const float* Ab = As + buf * BKc * BM + row_m;
      const float* Bb = Bs + buf * BKc * BN + col_n;
#pragma unroll 4
      for (int k = 0; k < BKc; k++) {
        float a[TM], b[TN];
        {
          const float4 a0 = *reinterpret_cast<const float4*>(Ab + k * BM);
          a[0] = a0.x; a[1] = a0.y; a[2] = a0.z; a[3] = a0.w;
          if (TM == 8) {
            const float4 a1 = *reinterpret_cast<const float4*>(Ab + k * BM + 4);
            a[4] = a1.x; a[5] = a1.y; a[6] = a1.z; a[7] = a1.w;
          }
        }
        if (TN == 8) {
          const float4 b0 = *reinterpret_cast<const float4*>(Bb + k * BN);
          const float4 b1 = *reinterpret_cast<const float4*>(Bb + k * BN + 4);
          b[0] = b0.x; b[1] = b0.y; b[2] = b0.z; b[3] = b0.w; b[4] = b1.x; b[5] = b1.y; b[6] = b1.z; b[7] = b1.w;
        } else {
          const float2 b0 = *reinterpret_cast<const float2*>(Bb + k * BN);
          const float2 b1 = *reinterpret_cast<const float2*>(Bb + k * BN + 2);
          const float2 b2 = *reinterpret_cast<const float2*>(Bb + k * BN + 4);
          b[0] = b0.x; b[1] = b0.y; b[2] = b1.x; b[3] = b1.y; b[4] = b2.x; b[5] = b2.y;
        }
#pragma unroll
        for (int i = 0; i < TM; i++)
#pragma unroll
          for (int j = 0; j < TN; j++) acc[i][j] = fmaf(a[i], b[j], acc[i][j]);
      }
      if (more) store_regs(buf ^ 1);
      gsync();
    }

    // ---- epilogue: partial tiles -> smem [KS][BN][CS_LD], then one token per thread
    __syncthreads();
    float* Cs = smem;
#pragma unroll
    for (int j = 0; j < TN; j++) {
      float* dst = Cs + (kg * BN + col_n + j) * CS_LD + row_m;
      *reinterpret_cast<float4*>(dst) = make_float4(acc[0][j], acc[1][j], acc[2][j], acc[3][j]);
      if (TM == 8) *reinterpret_cast<float4*>(dst + 4) = make_float4(acc[4][j], acc[5][j], acc[6][j], acc[7][j]);
    }
    __syncthreads();
    const int m_l = tid % BM, n_l0 = tid / BM;
    const int t = m0 + m_l;
    const bool tv = t < M;
    const int img = tv ? t / HW : b_first;
    const int hw = t - img * HW;
    if (PHASE == 2) {
      const int img0 = __shfl_sync(0xffffffffu, img, 0);
      const bool uniform = __all_sync(0xffffffffu, (img == img0) && tv);
      const int il0 = img0 - b_first;
#pragma unroll 4
      for (int i = 0; i < NPT; i++) {
        const int n_l = n_l0 + NSTEP * i, col = n0 + n_l;
        float v = 0.f;
#pragma unroll
        for (int g = 0; g < KS; g++) v += Cs[(g * BN + n_l) * CS_LD + m_l];
        const float gl = gelu_f(v + __ldg(p.b1 + col));
        if (tv) p.hidden[(size_t)col * Mp + t] = gl;
        float g2 = tv ? gl * gl : 0.f;
        if (uniform) {
          g2 = warp_sum(g2);
          if (lane == 0) {
            if (il0 < MAXIMG) atomicAdd(&sq_s[il0 * BN + n_l], g2);
            else atomicAdd(p.sumsq + (size_t)img0 * N + col, g2);
          }
        } else if (tv) {
          const int il = img - b_first;
          if (il < MAXIMG) atomicAdd(&sq_s[il * BN + n_l], g2);
          else atomicAdd(p.sumsq + (size_t)img * N + col, g2);
        }
      }
      __syncthreads();
      for (int idx = tid; idx < nimg * BN; idx += NT) {
        const int im = idx / BN, n = idx - im * BN;
        const float v = sq_s[idx]; sq_s[idx] = 0.f;
        atomicAdd(p.sumsq + (size_t)(b_first + im) * N + n0 + n, v);
      }
      __syncthreads();
    } else {
#pragma unroll 4
      for (int i = 0; i < NPT; i++) {
        const int n_l = n_l0 + NSTEP * i, col = n0 + n_l;
        if (col < N && tv) {
          float v = 0.f;
#pragma unroll
          for (int g = 0; g < KS; g++) v += Cs[(g * BN + n_l) * CS_LD + m_l];
          const size_t idx = ((size_t)img * N + col) * HW + hw;
          if (split == 1) p.out[idx] = v + __ldg(p.b2 + col) + __ldg(p.x + idx);
          else atomicAdd(p.out + idx, v);
        }
      }
      __syncthreads();
    }
  }
}

template <int PHASE>
__device__ __forceinline__ void run_gemm(const Params& p, int cfg, int split, float* smem, float* sq_s, float* inv_s, float* red_s) {
  switch (cfg) {
    case 0: gemm_phase<128, 128, 2, 4, 1, 32, PHASE>(p, smem, sq_s, inv_s, red_s, split); break;
    case 1: gemm_phase<128, 96, 2, 4, 1, 32, PHASE>(p, smem, sq_s, inv_s, red_s, split); break;
    case 2: gemm_phase<64, 64, 2, 2, 2, 32, PHASE>(p, smem, sq_s, inv_s, red_s, split); break;
    default: gemm_phase<32, 32, 1, 1, 8, 16, PHASE>(p, smem, sq_s, inv_s, red_s, split); break;
  }
}

__global__ void __launch_bounds__(NT, 2) convnext_fused(Params p) {
  extern __shared__ float4 smem4[];
  float* smem = reinterpret_cast<float*>(smem4);
  float* sq_s = smem + GEMM_SMEM_FLOATS;
  float* inv_s = sq_s + MAXIMG * 128;
  float* red_s = inv_s + 8;
  for (int i = threadIdx.x; i < MAXIMG * 128; i += NT) sq_s[i] = 0.f;
  __syncthreads();
  cg::grid_group grid = cg::this_grid();

  phase1(p, smem);
  grid.sync();
  if (p.split3 > 1) {  // initialise the output with residual + bias for the split-K atomics
    const long total = (long)p.M * p.C;
    for (long i = (long)blockIdx.x * NT + threadIdx.x; i < total; i += (long)gridDim.x * NT) {
      const int c = (int)((i / p.HW) % p.C);
      p.out[i] = p.x[i] + p.b2[c];
    }
  }
  run_gemm<2>(p, p.cfg2, 1, smem, sq_s, inv_s, red_s);
  grid.sync();
  run_gemm<3>(p, p.cfg3, p.split3, smem, sq_s, inv_s, red_s);
}

struct Pick { int cfg; int split; };
static Pick pick_cfg(long M, long N, long K, int grid, bool allow_split) {
  struct CfgD { int id, BM, BN, BK; double eff; };
  const CfgD cfgs[4] = {{0, 128, 128, 32, 0.80}, {1, 128, 96, 32, 0.78}, {2, 64, 64, 32, 0.60}, {3, 32, 32, 16, 0.40}};
  double best = 1e300; Pick r{0, 1};
  for (int ci = 0; ci < 4; ci++) {
    const CfgD& c = cfgs[ci];
    if (c.id == 1 && (N % 96) != 0) continue;
    const long tiles = ((M + c.BM - 1) / c.BM) * ((N + c.BN - 1) / c.BN);
    const long nk = K / c.BK;
    for (int S = 1; S <= (allow_split ? 16 : 1); S *= 2) {
      if (S > 1 && nk / S < 2) break;
      const long total = tiles * S;
      const long waves = (total + grid - 1) / grid;
      double cost = (double)c.BM * c.BN * ((double)K / S) / c.eff * (double)waves;
      if (S > 1) cost *= 1.08 + 0.03 * S;
      if (cost < best) { best = cost; r.cfg = c.id; r.split = S; }
    }
  }
  return r;
}

extern "C" const char* convnext_launch(Params p, void* stream_v) {
  static int grid_size = 0;
  cudaError_t e;
  if (grid_size == 0) {
    e = cudaFuncSetAttribute(convnext_fused, cudaFuncAttributeMaxDynamicSharedMemorySize, SMEM_BYTES);
    if (e != cudaSuccess) return cudaGetErrorString(e);
    int dev = 0; cudaGetDevice(&dev);
    int sms = 0; cudaDeviceGetAttribute(&sms, cudaDevAttrMultiProcessorCount, dev);
    int bps = 0;
    e = cudaOccupancyMaxActiveBlocksPerMultiprocessor(&bps, convnext_fused, NT, SMEM_BYTES);
    if (e != cudaSuccess) return cudaGetErrorString(e);
    if (bps < 1) bps = 1;
    grid_size = sms * bps;
  }
  const Pick k2 = pick_cfg(p.M, p.C4, p.C, grid_size, false);
  const Pick k3 = pick_cfg(p.M, p.C, p.C4, grid_size, true);
  p.cfg2 = k2.cfg; p.cfg3 = k3.cfg; p.split3 = k3.split;
  // phase-1 geometry
  {
    const int tw = (p.W % 4 == 0) ? 4 : 1;
    const int cgt = p.W / tw;
    int lpr, rg, wb;
    if (cgt >= 32) { lpr = 32; rg = 1; wb = (cgt + 31) / 32; }
    else { lpr = cgt; rg = 32 / cgt; wb = 1; }
    if (rg > 8) rg = 8;
    const int rows = 4 * rg;
    if (rows <= p.H) {
      p.p1_mode = 1; p.p1_twr = tw; p.p1_lpr = lpr; p.p1_rg = rg; p.p1_wb = wb;
      p.p1_hb = (p.H + rows - 1) / rows;
      p.p1_tiles = p.B * p.nch * p.p1_hb * p.p1_wb;
    } else {
      p.p1_mode = 0; p.p1_lpr = 0; p.p1_rg = 0; p.p1_wb = 0; p.p1_hb = 0;
      int r0 = 4;
      while (r0 > 1 && ((p.M + 32 * r0 - 1) / (32 * r0)) * p.nch < grid_size) r0 >>= 1;
      p.p1_twr = r0;
      p.p1_tiles = ((p.M + 32 * r0 - 1) / (32 * r0)) * p.nch;
    }
  }
  void* args[] = {&p};
  e = cudaLaunchCooperativeKernel((const void*)convnext_fused, dim3(grid_size), dim3(NT), args, (size_t)SMEM_BYTES,
                                  (cudaStream_t)stream_v);
  if (e != cudaSuccess) return cudaGetErrorString(e);
  return nullptr;
}
```
