I built and tested lineage F. The pipeline has 4 launches on the 28×28 rows and 5 on the 7×7 and 14×14 rows. It passes 16/16 workloads (portal-order re-runs included) and the 10-round fp64 precision log is clean on all 9 rows it handles. H32's success line is not met: same container, it beats r5-merge-b by 1–3 µs on 8,768, 8,384 and 16,768, and by 4.2 µs on 1,192.

- **cuBLASLt GELU epilogue: rejected on precision, not on speed.** It costs 1.3–3.9 µs over the plain bias epilogue (the kill gate did not fire). But it uses the tanh form of GELU (up to 4.7e-4 off exact erf), and in emulation that takes 1,192 and 4,192 to about 100% of their atol. Exact-erf GELU now runs in the GRN kernel, which gives the best margins any lineage has logged (max error at 59–76% of each row's atol).
- **Swap-AB NCHW GEMM2:** adopted only for 28×28 images (2–4 µs faster). For 7×7 and 14×14 it is 20–190 µs slower, so the transpose-add kernel stays there.

### Rationale
The kernel uses four or five launches. K1 runs in clusters of 8: a bulk-plane conv, with LayerNorm finished through a DSMEM exchange of per-token sums, plus role CTAs that convert the weights and fold b2' into hi/lo fp16 columns of W2. G1 is cuBLASLt fp16 with a bias epilogue and fp32 output. K3 does exact-erf GELU and the GRN statistics through a self-resetting 64-bit cell, then rescales into fp16. G2 is cuBLASLt: NCHW swap-AB with x as the C operand for 28×28 images, and channels-last plus a transpose-add kernel otherwise.

```cuda file=kernel.cu
// Lineage F (r6): library-lean pipeline for the C>=192, HW<=784 rows (7x7 / 14x14 / 28x28 images).
//  K1 (clusters of 8 CTAs along channels): bulk-plane depthwise conv + LayerNorm finished in-CTA (per-token sum/sumsq
//      exchanged through DSMEM) -> a16 [M][C] fp16. Role CTAs: W1 -> fp16, W2 -> fp16 [C][4C+8] with b2' = W2.gb + b2
//      as hi/lo fp16 columns (K augmentation), fp32 weights L2-prefetched at entry.
//  G1 (cuBLASLt, binding.cpp): z = a16 W1^T + b1, fp32 out (BIAS epilogue).
//  K3 (one wave): exact-erf GELU, per-image GRN statistics through a self-resetting 64-bit cell,
//      h' = fp16(gelu(z) * (1 + gw*nf)) [M][4C+8] with the ones columns for the b2' augmentation.
//  G2 (cuBLASLt): NCHW swap-AB batched (out_b = h'_b W2^T + x_b, beta 1) or channels-last + K4 transpose-add.
// Other shapes run the r0 fp32 cooperative kernel (namespace r0ns, unchanged).
#include <cuda_runtime.h>
#include <cuda_fp16.h>
#include <cooperative_groups.h>
#include <stdint.h>

#define F_NT 512
#define F_MAXB 64

struct K1P {
  const float* x; const float* dww; const float* dwb; const float* lnw; const float* lnb;
  const float* w1; const float* w2; const float* gb; const float* b2;
  void* a16; void* w1h; void* w2h;
  int B, C, H, W, HW, CPC, NS, RS, PS, T, K2, nrole, rows_per_role;
  long long w1chunk;
  float ln_eps;
};
struct K3P {
  const float* z; const float* gw; void* hp;
  int B, C4, HW, KS, NSL, K2;
  float eps;
};

__device__ __forceinline__ uint32_t s_addr(const void* p) { return (uint32_t)__cvta_generic_to_shared(p); }
__device__ __forceinline__ void mbar_init(uint64_t* m, unsigned n) {
  asm volatile("mbarrier.init.shared::cta.b64 [%0], %1;" ::"r"(s_addr(m)), "r"(n) : "memory");
  asm volatile("fence.mbarrier_init.release.cluster;" ::: "memory");
}
__device__ __forceinline__ void mbar_expect(uint64_t* m, unsigned bytes) {
  asm volatile("mbarrier.arrive.expect_tx.shared::cta.b64 _, [%0], %1;" ::"r"(s_addr(m)), "r"(bytes) : "memory");
}
__device__ __forceinline__ void mbar_wait0(uint64_t* m) {
  const uint32_t a = s_addr(m);
  uint32_t ok = 0;
  while (!ok) {
    asm volatile("{\n\t.reg .pred p;\n\tmbarrier.try_wait.parity.shared::cta.b64 p, [%1], 0;\n\tselp.u32 %0, 1, 0, p;\n\t}"
                 : "=r"(ok) : "r"(a) : "memory");
  }
}
__device__ __forceinline__ void bulk_g2s(void* dst, const void* src, unsigned bytes, uint64_t* m) {
  asm volatile("cp.async.bulk.shared::cluster.global.mbarrier::complete_tx::bytes [%0], [%1], %2, [%3];"
               ::"r"(s_addr(dst)), "l"(src), "r"(bytes), "r"(s_addr(m)) : "memory");
}
__device__ __forceinline__ void prefetch_l2(const void* src, unsigned bytes) {
  asm volatile("cp.async.bulk.prefetch.L2.global [%0], %1;" ::"l"(src), "r"(bytes) : "memory");
}
__device__ __forceinline__ void cl_arrive() { asm volatile("barrier.cluster.arrive.release.aligned;" ::: "memory"); }
__device__ __forceinline__ void cl_wait() { asm volatile("barrier.cluster.wait.acquire.aligned;" ::: "memory"); }
__device__ __forceinline__ float2 ld_peer_f2(const float2* p, unsigned rank) {
  uint32_t r;
  asm volatile("mapa.shared::cluster.u32 %0, %1, %2;" : "=r"(r) : "r"(s_addr(p)), "r"(rank));
  float2 v;
  asm volatile("ld.shared::cluster.v2.f32 {%0, %1}, [%2];" : "=f"(v.x), "=f"(v.y) : "r"(r) : "memory");
  return v;
}
__device__ __forceinline__ float f_wsum(float v) {
#pragma unroll
  for (int o = 16; o > 0; o >>= 1) v += __shfl_xor_sync(0xffffffffu, v, o);
  return v;
}
__device__ __forceinline__ uint32_t pack_h2(float a, float b) {
  __half2 h = __floats2half2_rn(a, b);
  return *reinterpret_cast<uint32_t*>(&h);
}

// ---------------------------------------------------------------- K1 role CTAs: weight conversion
// Each role CTA converts up to 2 rows of W2 (fp16 copy + fp32 dot with gb -> b2' hi/lo) and one chunk of W1;
// the fp32 sources are L2-prefetched at entry and the first round of loads is issued before any use.
__device__ __forceinline__ void k1_role(const K1P& p, float* red) {
  const int tid = threadIdx.x, lane = tid & 31, warp = tid >> 5;
  const int role = ((int)(blockIdx.z - p.B) * p.NS + (int)blockIdx.y) * 8 + (int)blockIdx.x;
  const int C = p.C, C4 = 4 * C;
  const long long U1 = (long long)C * C;  // float4 units of W1 [4C][C]
  const long long u0 = (long long)role * p.w1chunk;
  const long long u1 = (u0 + p.w1chunk < U1) ? u0 + p.w1chunk : U1;
  const int rA = role * p.rows_per_role;
  const int rB = min(C, rA + p.rows_per_role);
  const int nr = rB > rA ? rB - rA : 0;
  if (tid == 0) {
    if (nr > 0) prefetch_l2(p.w2 + (size_t)rA * C4, (unsigned)(nr * C4 * 4));
    if (u1 > u0) prefetch_l2(p.w1 + u0 * 4, (unsigned)((u1 - u0) * 16));
  }
  const float4 z4 = make_float4(0.f, 0.f, 0.f, 0.f);
  const float4* w1v = reinterpret_cast<const float4*>(p.w1);
  const float4* w2v = reinterpret_cast<const float4*>(p.w2);
  const float4* gbv = reinterpret_cast<const float4*>(p.gb);
  __half* w1h = reinterpret_cast<__half*>(p.w1h);
  __half* w2h = reinterpret_cast<__half*>(p.w2h);
  float4 g[2], v[2][2], w[8];
#pragma unroll
  for (int i = 0; i < 2; i++) { const int j = tid + i * F_NT; g[i] = (nr > 0 && j < C) ? __ldg(gbv + j) : z4; }
#pragma unroll
  for (int rr = 0; rr < 2; rr++)
#pragma unroll
    for (int i = 0; i < 2; i++) {
      const int j = tid + i * F_NT;
      v[rr][i] = (rr < nr && j < C) ? __ldg(w2v + (size_t)(rA + rr) * C + j) : z4;
    }
#pragma unroll
  for (int i = 0; i < 8; i++) { const long long u = u0 + i * F_NT + tid; w[i] = (u < u1) ? __ldg(w1v + u) : z4; }
#pragma unroll
  for (int i = 0; i < 8; i++) {
    const long long u = u0 + i * F_NT + tid;
    if (u < u1) *reinterpret_cast<uint2*>(w1h + u * 4) = make_uint2(pack_h2(w[i].x, w[i].y), pack_h2(w[i].z, w[i].w));
  }
  if (nr > 0) {
#pragma unroll
    for (int rr = 0; rr < 2; rr++) {
      float s = 0.f;
#pragma unroll
      for (int i = 0; i < 2; i++) {
        const int j = tid + i * F_NT;
        const float4 a = v[rr][i];
        s = fmaf(a.x, g[i].x, s); s = fmaf(a.y, g[i].y, s); s = fmaf(a.z, g[i].z, s); s = fmaf(a.w, g[i].w, s);
        if (rr < nr && j < C)
          *reinterpret_cast<uint2*>(w2h + (size_t)(rA + rr) * p.K2 + 4 * j) = make_uint2(pack_h2(a.x, a.y), pack_h2(a.z, a.w));
      }
      s = f_wsum(s);
      if (lane == 0) red[warp * 2 + rr] = s;
    }
    __syncthreads();
    if (tid < nr) {
      float s = 0.f;
#pragma unroll
      for (int w2 = 0; w2 < F_NT / 32; w2++) s += red[w2 * 2 + tid];
      const float bv = s + __ldg(p.b2 + rA + tid);
      const __half hi = __float2half_rn(bv);
      const __half lo = __float2half_rn(bv - __half2float(hi));
      const uint4 q = make_uint4((uint32_t)__half_as_ushort(hi) | ((uint32_t)__half_as_ushort(lo) << 16), 0u, 0u, 0u);
      *reinterpret_cast<uint4*>(w2h + (size_t)(rA + tid) * p.K2 + C4) = q;
    }
  }
  for (long long base = u0 + 8LL * F_NT; base < u1; base += 8LL * F_NT) {  // only when the W1 chunk is large
#pragma unroll
    for (int i = 0; i < 8; i++) { const long long u = base + i * F_NT + tid; w[i] = (u < u1) ? __ldg(w1v + u) : z4; }
#pragma unroll
    for (int i = 0; i < 8; i++) {
      const long long u = base + i * F_NT + tid;
      if (u < u1) *reinterpret_cast<uint2*>(w1h + u * 4) = make_uint2(pack_h2(w[i].x, w[i].y), pack_h2(w[i].z, w[i].w));
    }
  }
}

// ---------------------------------------------------------------- K1: conv + LayerNorm (cluster of 8 along C)
__global__ void __launch_bounds__(F_NT, 2) f_k1(K1P p) {
  extern __shared__ __align__(128) float smf[];
  if ((int)blockIdx.z >= p.B) { k1_role(p, smf); return; }
  const int tid = threadIdx.x;
  const int b = blockIdx.z, s = blockIdx.y, rk = blockIdx.x;
  const int C = p.C, H = p.H, W = p.W, HW = p.HW, CPC = p.CPC, RS = p.RS, PS = p.PS, T = p.T;
  const int c0 = rk * CPC, r0 = s * RS;
  const int lo = max(0, r0 - 3), hi = min(H, r0 + RS + 3);
  float* xs = smf;
  float* dS = xs + CPC * PS;
  float* wS = dS + CPC * T;
  float* bS = wS + CPC * 49;
  float* lwS = bS + CPC;
  float* lbS = lwS + CPC;
  float2* stS = reinterpret_cast<float2*>(lbS + CPC);
  float2* mrS = stS + T;
  uint64_t* mbar = reinterpret_cast<uint64_t*>(mrS + T);
  if (tid == 0) {
    mbar_init(mbar, 1);
    if (p.NS == 1) {  // the CTA's channel planes are one contiguous run of NCHW
      const unsigned bytes = (unsigned)(CPC * HW * 4);
      mbar_expect(mbar, bytes);
      bulk_g2s(xs, p.x + ((size_t)b * C + c0) * HW, bytes, mbar);
    } else {          // row strip with halo, one copy per channel
      const unsigned rb = (unsigned)((hi - lo) * W * 4);
      mbar_expect(mbar, rb * CPC);
      for (int ch = 0; ch < CPC; ch++)
        bulk_g2s(xs + ch * PS, p.x + (((size_t)b * C + c0 + ch) * H + lo) * W, rb, mbar);
    }
  }
  for (int i = tid; i < CPC * 49; i += F_NT) wS[i] = __ldg(p.dww + (size_t)c0 * 49 + i);
  for (int i = tid; i < CPC; i += F_NT) {
    bS[i] = __ldg(p.dwb + c0 + i); lwS[i] = __ldg(p.lnw + c0 + i); lbS[i] = __ldg(p.lnb + c0 + i);
  }
  __syncthreads();
  mbar_wait0(mbar);
  // one thread per (channel, output row, 7-wide column segment)
  const int G = W / 7, per_ch = RS * G, items = CPC * per_ch;
  for (int it = tid; it < items; it += F_NT) {
    const int ch = it / per_ch;
    const int rem = it - ch * per_ch;
    const int orow = rem / G;
    const int sg = rem - orow * G;
    const int oh = r0 + orow;
    if (oh >= H) continue;
    const float* xp = xs + ch * PS - lo * W;
    const float* wp = wS + ch * 49;
    float acc[7];
    const float bias = bS[ch];
#pragma unroll
    for (int o = 0; o < 7; o++) acc[o] = bias;
    const int wc0 = sg * 7 - 3;
#pragma unroll
    for (int i = 0; i < 7; i++) {
      const int ih = oh - 3 + i;
      const bool rv = (ih >= 0) && (ih < H);
      float v[13];
#pragma unroll
      for (int jj = 0; jj < 13; jj++) {
        const int iw = wc0 + jj;
        v[jj] = (rv && iw >= 0 && iw < W) ? xp[ih * W + iw] : 0.f;
      }
      float wr[7];
#pragma unroll
      for (int j = 0; j < 7; j++) wr[j] = wp[i * 7 + j];
#pragma unroll
      for (int o = 0; o < 7; o++)
#pragma unroll
        for (int j = 0; j < 7; j++) acc[o] = fmaf(v[o + j], wr[j], acc[o]);
    }
    float* dp = dS + ch * T + orow * W + sg * 7;
#pragma unroll
    for (int o = 0; o < 7; o++) dp[o] = acc[o];
  }
  __syncthreads();
  for (int t = tid; t < T; t += F_NT) {
    float s1 = 0.f, s2 = 0.f;
    for (int ch = 0; ch < CPC; ch++) { const float v = dS[ch * T + t]; s1 += v; s2 = fmaf(v, v, s2); }
    stS[t] = make_float2(s1, s2);
  }
  __syncwarp();
  cl_arrive();
  cl_wait();
  const float invC = 1.f / (float)C;
  for (int t = tid; t < T; t += F_NT) {
    float S1 = 0.f, S2 = 0.f;
#pragma unroll
    for (int r = 0; r < 8; r++) { const float2 v = ld_peer_f2(stS + t, r); S1 += v.x; S2 += v.y; }
    const float mean = S1 * invC;
    const float var = fmaxf(S2 * invC - mean * mean, 0.f);
    mrS[t] = make_float2(mean, rsqrtf(var + p.ln_eps));
  }
  __syncwarp();
  cl_arrive();  // our reads of the peers' statistics are done
  __syncthreads();
  const int G8 = CPC >> 3, tb = r0 * W;
  __half* a16 = reinterpret_cast<__half*>(p.a16);
  for (int it = tid; it < T * G8; it += F_NT) {
    const int t = it / G8, g = it - t * G8;
    if (tb + t >= HW) continue;
    const float2 mr = mrS[t];
    uint32_t q[4];
#pragma unroll
    for (int k = 0; k < 4; k++) {
      const int c = g * 8 + 2 * k;
      const float v0 = fmaf((dS[c * T + t] - mr.x) * mr.y, lwS[c], lbS[c]);
      const float v1 = fmaf((dS[(c + 1) * T + t] - mr.x) * mr.y, lwS[c + 1], lbS[c + 1]);
      q[k] = pack_h2(v0, v1);
    }
    *reinterpret_cast<uint4*>(a16 + ((size_t)b * HW + tb + t) * C + c0 + g * 8) = make_uint4(q[0], q[1], q[2], q[3]);
  }
  __syncwarp();
  cl_wait();  // peers may still read our statistics
}

// ---------------------------------------------------------------- K3: GELU + GRN statistics + scale, one wave
// CTA (k-slice, image): cp.async its [HW][KS] fp32 slice of z into smem, exact erf GELU, column sum of squares,
// then one 64-bit red.add per CTA into the image cell (arrival count in bits 52..63, sum of gf in 2^-20 fixed point
// below); every CTA polls the cell until all slices of the image have arrived. The last CTA to leave resets the cell.
// The cells live in module memory (zero at load) and are back to zero at the end of every call.
__device__ unsigned long long f_cell[F_MAXB];
__device__ unsigned int f_dep[F_MAXB];
__global__ void __launch_bounds__(F_NT) f_k3(K3P p) {
  extern __shared__ __align__(128) float smf[];
  const int tid = threadIdx.x, lane = tid & 31, warp = tid >> 5;
  const int sl = blockIdx.x, b = blockIdx.y;
  const int KS = p.KS, HW = p.HW, C4 = p.C4, K2 = p.K2, k0 = sl * KS;
  float* zs = smf;
  float* red = zs + HW * KS;
  float* gwS = red + F_NT;
  float* scS = gwS + KS;
  float* misc = scS + KS;
  const float* zb = p.z + (size_t)b * HW * C4 + k0;
  {
    const int q4 = KS >> 2, nq = HW * q4;
    for (int i = tid; i < nq; i += F_NT) {
      const int t = i / q4, q = i - t * q4;
      asm volatile("cp.async.cg.shared.global [%0], [%1], 16;" ::"r"(s_addr(zs + t * KS + q * 4)), "l"(zb + (size_t)t * C4 + q * 4)
                   : "memory");
    }
    asm volatile("cp.async.commit_group;" ::: "memory");
  }
  for (int i = tid; i < KS; i += F_NT) gwS[i] = __ldg(p.gw + k0 + i);
  __half* hp = reinterpret_cast<__half*>(p.hp);
  if (sl == 0) {  // ones columns that carry b2' (hi, lo) through GEMM2
    const uint4 one = make_uint4(0x3C003C00u, 0u, 0u, 0u);
    for (int t = tid; t < HW; t += F_NT) *reinterpret_cast<uint4*>(hp + ((size_t)b * HW + t) * K2 + C4) = one;
  }
  asm volatile("cp.async.wait_group 0;" ::: "memory");
  __syncthreads();
  const int TG = F_NT / KS, kk = tid % KS, tg = tid / KS;
  float acc0 = 0.f, acc1 = 0.f;
  int t = tg;
  for (; t + TG < HW; t += 2 * TG) {
    const float v0 = zs[t * KS + kk], v1 = zs[(t + TG) * KS + kk];
    const float g0 = 0.5f * v0 * (1.f + erff(v0 * 0.70710678118654752f));
    const float g1 = 0.5f * v1 * (1.f + erff(v1 * 0.70710678118654752f));
    zs[t * KS + kk] = g0; zs[(t + TG) * KS + kk] = g1;
    acc0 = fmaf(g0, g0, acc0); acc1 = fmaf(g1, g1, acc1);
  }
  if (t < HW) {
    const float v0 = zs[t * KS + kk];
    const float g0 = 0.5f * v0 * (1.f + erff(v0 * 0.70710678118654752f));
    zs[t * KS + kk] = g0; acc0 = fmaf(g0, g0, acc0);
  }
  red[tid] = acc0 + acc1;
  __syncthreads();
  float gf = 0.f;
  if (tid < KS) {
    float s = 0.f;
    for (int q = 0; q < TG; q++) s += red[q * KS + tid];
    gf = sqrtf(s);
    scS[tid] = gf;
  }
  const float wsum = f_wsum(gf);
  __syncthreads();
  if (lane == 0) red[warp] = wsum;
  __syncthreads();
  if (tid == 0) {
    float part = 0.f;
#pragma unroll
    for (int w = 0; w < F_NT / 32; w++) part += red[w];
    const unsigned long long add = (1ull << 52) + __double2ull_rn((double)part * 1048576.0);
    asm volatile("red.relaxed.gpu.global.add.u64 [%0], %1;" ::"l"(f_cell + b), "l"(add) : "memory");
    unsigned long long v;
    do {
      asm volatile("ld.relaxed.gpu.global.u64 %0, [%1];" : "=l"(v) : "l"(f_cell + b) : "memory");
    } while ((unsigned)(v >> 52) < (unsigned)p.NSL);
    const double tot = (double)(v & ((1ull << 52) - 1ull)) * (1.0 / 1048576.0);
    misc[0] = (float)(1.0 / (tot / (double)C4 + (double)p.eps));
  }
  __syncthreads();
  if (tid < KS) scS[tid] = fmaf(gwS[tid] * scS[tid], misc[0], 1.f);
  __syncthreads();
  const int G8 = KS >> 3;
  for (int it = tid; it < HW * G8; it += F_NT) {
    const int tt = it / G8, gg = it - tt * G8;
    const float4 a = *reinterpret_cast<const float4*>(zs + tt * KS + gg * 8);
    const float4 c = *reinterpret_cast<const float4*>(zs + tt * KS + gg * 8 + 4);
    const float4 s0 = *reinterpret_cast<const float4*>(scS + gg * 8);
    const float4 s1 = *reinterpret_cast<const float4*>(scS + gg * 8 + 4);
    const uint4 q = make_uint4(pack_h2(a.x * s0.x, a.y * s0.y), pack_h2(a.z * s0.z, a.w * s0.w),
                               pack_h2(c.x * s1.x, c.y * s1.y), pack_h2(c.z * s1.z, c.w * s1.w));
    *reinterpret_cast<uint4*>(hp + ((size_t)b * HW + tt) * K2 + k0 + gg * 8) = q;
  }
  if (tid == 0) {  // departure: the last CTA of the image resets its cell for the next call
    const unsigned old = atomicAdd(&f_dep[b], 1u);
    if (old == (unsigned)(p.NSL - 1)) { atomicExch(f_cell + b, 0ull); atomicExch(&f_dep[b], 0u); }
  }
}

// ---------------------------------------------------------------- K4: channels-last y -> NCHW out (+ x)
__global__ void __launch_bounds__(256) f_k4(const float* __restrict__ y, const float* __restrict__ x, float* __restrict__ out,
                                            int C, int HW) {
  __shared__ float tl[32][33];
  const int b = blockIdx.z, c0 = blockIdx.y * 32, h0 = blockIdx.x * 32, tx = threadIdx.x & 31, ty = threadIdx.x >> 5;
  float xv[4];
#pragma unroll
  for (int i = 0; i < 4; i++) {
    const int c = c0 + ty + 8 * i, hw = h0 + tx;
    xv[i] = (hw < HW) ? __ldg(x + ((size_t)b * C + c) * HW + hw) : 0.f;
  }
#pragma unroll
  for (int i = 0; i < 4; i++) {
    const int hw = h0 + ty + 8 * i;
    if (hw < HW) tl[ty + 8 * i][tx] = __ldg(y + ((size_t)b * HW + hw) * C + c0 + tx);
  }
  __syncthreads();
#pragma unroll
  for (int i = 0; i < 4; i++) {
    const int c = c0 + ty + 8 * i, hw = h0 + tx;
    if (hw < HW) out[((size_t)b * C + c) * HW + hw] = xv[i] + tl[tx][ty + 8 * i];
  }
}

static bool f_attr_done = false;
static int f_nsm = 0;
static const char* f_attrs() {
  if (f_attr_done) return nullptr;
  cudaError_t e = cudaFuncSetAttribute(f_k1, cudaFuncAttributeMaxDynamicSharedMemorySize, 160 * 1024);
  if (e != cudaSuccess) return cudaGetErrorString(e);
  e = cudaFuncSetAttribute(f_k3, cudaFuncAttributeMaxDynamicSharedMemorySize, 160 * 1024);
  if (e != cudaSuccess) return cudaGetErrorString(e);
  int dev = 0;
  cudaGetDevice(&dev);
  cudaDeviceGetAttribute(&f_nsm, cudaDevAttrMultiProcessorCount, dev);
  f_attr_done = true;
  return nullptr;
}
extern "C" int f_occ(int which, size_t smem) {
  if (f_attrs()) return 0;
  int n = 0;
  cudaOccupancyMaxActiveBlocksPerMultiprocessor(&n, which == 1 ? (const void*)f_k1 : (const void*)f_k3, F_NT, smem);
  return n;
}
extern "C" const char* f_k1_launch(K1P p, int RZ, size_t smem, void* st) {
  const char* er = f_attrs();
  if (er) return er;
  cudaLaunchConfig_t cfg = {};
  cfg.gridDim = dim3(8, p.NS, p.B + RZ);
  cfg.blockDim = dim3(F_NT);
  cfg.dynamicSmemBytes = smem;
  cfg.stream = (cudaStream_t)st;
  cudaLaunchAttribute at[1];
  at[0].id = cudaLaunchAttributeClusterDimension;
  at[0].val.clusterDim.x = 8; at[0].val.clusterDim.y = 1; at[0].val.clusterDim.z = 1;
  cfg.attrs = at; cfg.numAttrs = 1;
  cudaError_t e = cudaLaunchKernelEx(&cfg, f_k1, p);
  return e == cudaSuccess ? nullptr : cudaGetErrorString(e);
}
extern "C" const char* f_k3_launch(K3P p, size_t smem, int occ, void* st) {
  const char* er = f_attrs();
  if (er) return er;
  cudaError_t e;
  // the per-image poll needs every CTA of an image resident: a plain launch when the grid is one guaranteed wave
  // (the previous kernel on the stream has drained), else a cooperative launch
  if ((long)occ * f_nsm >= (long)p.NSL * p.B) {
    f_k3<<<dim3(p.NSL, p.B), F_NT, smem, (cudaStream_t)st>>>(p);
    e = cudaGetLastError();
  } else {
    void* args[] = {&p};
    e = cudaLaunchCooperativeKernel((const void*)f_k3, dim3(p.NSL, p.B), dim3(F_NT), args, smem, (cudaStream_t)st);
  }
  return e == cudaSuccess ? nullptr : cudaGetErrorString(e);
}
extern "C" const char* f_k4_launch(const float* y, const float* x, float* out, int B, int C, int HW, void* st) {
  f_k4<<<dim3((HW + 31) / 32, C / 32, B), 256, 0, (cudaStream_t)st>>>(y, x, out, C, HW);
  cudaError_t e = cudaGetLastError();
  return e == cudaSuccess ? nullptr : cudaGetErrorString(e);
}

#define NT 256
#define MAXIMG 5
#define GEMM_SMEM_FLOATS (128 * 132)
#define SMEM_BYTES (GEMM_SMEM_FLOATS * 4 + MAXIMG * 128 * 4 + 64)
#define P1_SLOTS 512

struct Params {
  const float* x; const float* dww; const float* dwb; const float* lnw; const float* lnb;
  const float* w1; const float* b1; const float* gw; const float* gb; const float* w2; const float* b2;
  float* out;
  float* xn; float* part; float* hidden; float* sumsq; float* w1t; float* w2t;
  int B, C, H, W, HW, M, Mp, C4, nch;
  float eps, ln_eps;
  int p1_mode, p1_twr, p1_tiles, p1_lpr, p1_rg, p1_hb, p1_wb;
  int cfg2, cfg3, split3;
};

__device__ __forceinline__ float gelu_f(float v) { return 0.5f * v * (1.f + erff(v * 0.70710678118654752f)); }

__device__ __forceinline__ float warp_sum(float v) {
#pragma unroll
  for (int o = 16; o > 0; o >>= 1) v += __shfl_xor_sync(0xffffffffu, v, o);
  return v;
}

__device__ void transpose_mat(const float* __restrict__ src, int R, int Cc, float* __restrict__ dst, float* tile) {
  const int rt = (R + 31) / 32, ct = (Cc + 31) / 32;
  const int tid = threadIdx.x, j = tid & 31, i0 = tid >> 5;
  for (int t = blockIdx.x; t < rt * ct; t += gridDim.x) {
    const int tr = t / ct, tc = t % ct;
#pragma unroll
    for (int s = 0; s < 4; s++) {
      const int i = i0 + 8 * s, r = tr * 32 + i, c = tc * 32 + j;
      tile[i * 33 + j] = (r < R && c < Cc) ? src[(size_t)r * Cc + c] : 0.f;
    }
    __syncthreads();
#pragma unroll
    for (int s = 0; s < 4; s++) {
      const int i = i0 + 8 * s, c = tc * 32 + i, r = tr * 32 + j;
      if (r < R && c < Cc) dst[(size_t)c * R + r] = tile[j * 33 + i];
    }
    __syncthreads();
  }
}

template <int TW>
__device__ __forceinline__ void conv_patch(const float* __restrict__ xc, const float* wt, int H, int W, int h0, int w0,
                                           float (&acc)[4][TW]) {
#pragma unroll
  for (int r = 0; r < 4; r++)
#pragma unroll
    for (int cc = 0; cc < TW; cc++) acc[r][cc] = 0.f;
#pragma unroll
  for (int i = 0; i < 10; i++) {
    const int hh = h0 - 3 + i;
    const bool rowok = (hh >= 0) && (hh < H);
    float v[TW + 6];
    if (TW == 4) {
      const float* rp = xc + hh * W;
      float4 L = make_float4(0.f, 0.f, 0.f, 0.f), Cq = L, R = L;
      if (rowok) {
        if (w0 >= 4) L = __ldg(reinterpret_cast<const float4*>(rp + w0 - 4));
        Cq = __ldg(reinterpret_cast<const float4*>(rp + w0));
        if (w0 + 4 < W) R = __ldg(reinterpret_cast<const float4*>(rp + w0 + 4));
      }
      v[0] = L.y; v[1] = L.z; v[2] = L.w; v[3] = Cq.x; v[4] = Cq.y; v[5] = Cq.z; v[6] = Cq.w;
      v[7] = R.x; v[8] = R.y; v[9] = R.z;
    } else {
#pragma unroll
      for (int jj = 0; jj < TW + 6; jj++) {
        const int ww = w0 - 3 + jj;
        v[jj] = (rowok && ww >= 0 && ww < W) ? __ldg(xc + hh * W + ww) : 0.f;
      }
    }
#pragma unroll
    for (int r = 0; r < 4; r++) {
      const int ki = i - r;
      if (ki >= 0 && ki < 7) {
#pragma unroll
        for (int cc = 0; cc < TW; cc++)
#pragma unroll
          for (int j = 0; j < 7; j++) acc[r][cc] = fmaf(v[cc + j], wt[ki * 7 + j], acc[r][cc]);
      }
    }
  }
}

template <int MODE, int TWR>
__device__ void phase1_conv(const Params& p, float* smem) {
  const int tid = threadIdx.x, warp = tid >> 5, lane = tid & 31;
  const int C = p.C, H = p.H, W = p.W, HW = p.HW, M = p.M, Mp = p.Mp, nch = p.nch;
  float* st_s = smem;
  float* wsm = smem + 8 * P1_SLOTS * 2;
  const int lpr = p.p1_lpr, rg = p.p1_rg, hb = p.p1_hb, wb = p.p1_wb;
  const int rows_per_tile = 4 * rg;
  const int rpt = lpr * TWR;

  for (int tile = blockIdx.x; tile < p.p1_tiles; tile += gridDim.x) {
    int b = 0, cch = 0, hbi = 0, wbi = 0, tok0 = 0;
    if (MODE == 1) {
      const int per = nch * hb * wb;
      b = tile / per; int rem = tile - b * per;
      cch = rem / (hb * wb); rem -= cch * (hb * wb);
      hbi = rem / wb; wbi = rem - hbi * wb;
    } else {
      cch = tile % nch; tok0 = (tile / nch) * (32 * TWR);
    }
    for (int i = tid; i < 8 * P1_SLOTS * 2; i += NT) st_s[i] = 0.f;
    for (int i = tid; i < 32 * 49; i += NT) wsm[i] = __ldg(p.dww + (size_t)cch * 32 * 49 + i);
    __syncthreads();

    int rgi = 0, cgi = 0, h0 = 0, w0 = 0; bool lane_ok = true;
    if (MODE == 1) {
      rgi = lane / lpr; cgi = lane - rgi * lpr;
      lane_ok = rgi < rg;
      h0 = hbi * rows_per_tile + rgi * 4;
      w0 = (wbi * 32 + cgi) * TWR;
      lane_ok = lane_ok && (w0 < W);
    }
    int tk[MODE == 0 ? TWR : 1], tb[MODE == 0 ? TWR : 1], th[MODE == 0 ? TWR : 1], tw_[MODE == 0 ? TWR : 1];
    bool tv[MODE == 0 ? TWR : 1];
    if (MODE == 0) {
#pragma unroll
      for (int r = 0; r < TWR; r++) {
        const int t = tok0 + r * 32 + lane;
        tv[r] = t < M;
        const int tt = tv[r] ? t : 0;
        const int bb = tt / HW; const int rem = tt - bb * HW; const int hh = rem / W;
        tk[r] = t; tb[r] = bb; th[r] = hh; tw_[r] = rem - hh * W;
      }
    }

#pragma unroll 1
    for (int cl = 0; cl < 4; cl++) {
      const int clc = warp * 4 + cl;
      const int c = cch * 32 + clc;
      float wt[49];
#pragma unroll
      for (int i = 0; i < 49; i++) wt[i] = wsm[clc * 49 + i];
      const float bias = __ldg(p.dwb + c);
      float* xrow = p.xn + (size_t)c * Mp;
      float* sts = st_s + warp * P1_SLOTS * 2;
      if (MODE == 1) {
        if (lane_ok) {
          float acc[4][TWR];
          conv_patch<TWR>(p.x + ((size_t)b * C + c) * HW, wt, H, W, h0, w0, acc);
#pragma unroll
          for (int r = 0; r < 4; r++) {
            const int h = h0 + r;
            if (h < H) {
              const int t = b * HW + h * W + w0;
              const int slot = (rgi * 4 + r) * rpt + cgi * TWR;
              if (TWR == 4) {
                float4 o = make_float4(acc[r][0] + bias, acc[r][1] + bias, acc[r][2] + bias, acc[r][3] + bias);
                *reinterpret_cast<float4*>(xrow + t) = o;
                sts[(slot + 0) * 2] += o.x; sts[(slot + 0) * 2 + 1] += o.x * o.x;
                sts[(slot + 1) * 2] += o.y; sts[(slot + 1) * 2 + 1] += o.y * o.y;
                sts[(slot + 2) * 2] += o.z; sts[(slot + 2) * 2 + 1] += o.z * o.z;
                sts[(slot + 3) * 2] += o.w; sts[(slot + 3) * 2 + 1] += o.w * o.w;
              } else {
#pragma unroll
                for (int cc = 0; cc < TWR; cc++) {
                  const float o = acc[r][cc] + bias;
                  xrow[t + cc] = o;
                  sts[(slot + cc) * 2] += o; sts[(slot + cc) * 2 + 1] += o * o;
                }
              }
            }
          }
        }
      } else {
        float acc[TWR];
#pragma unroll
        for (int r = 0; r < TWR; r++) acc[r] = 0.f;
#pragma unroll
        for (int i = 0; i < 7; i++)
#pragma unroll
          for (int j = 0; j < 7; j++) {
            const float wv = wt[i * 7 + j];
#pragma unroll
            for (int r = 0; r < TWR; r++) {
              const int hh = th[r] - 3 + i, ww = tw_[r] - 3 + j;
              if (tv[r] && hh >= 0 && hh < H && ww >= 0 && ww < W)
                acc[r] = fmaf(__ldg(p.x + ((size_t)tb[r] * C + c) * HW + hh * W + ww), wv, acc[r]);
            }
          }
#pragma unroll
        for (int r = 0; r < TWR; r++) {
          if (tv[r]) {
            const float o = acc[r] + bias;
            xrow[tk[r]] = o;
            const int slot = r * 32 + lane;
            sts[slot * 2] += o; sts[slot * 2 + 1] += o * o;
          }
        }
      }
    }
    __syncthreads();
    const int nslots = (MODE == 1) ? rows_per_tile * rpt : 32 * TWR;
    for (int slot = tid; slot < nslots; slot += NT) {
      int t; bool valid;
      if (MODE == 1) {
        const int row = slot / rpt, col = slot - row * rpt;
        const int h = hbi * rows_per_tile + row, w = wbi * 32 * TWR + col;
        valid = (h < H) && (w < W);
        t = b * HW + h * W + w;
      } else {
        t = tok0 + slot; valid = t < M;
      }
      if (valid) {
        float S1 = 0.f, S2 = 0.f;
#pragma unroll
        for (int ws = 0; ws < 8; ws++) { S1 += st_s[(ws * P1_SLOTS + slot) * 2]; S2 += st_s[(ws * P1_SLOTS + slot) * 2 + 1]; }
        float* pp = p.part + ((size_t)cch * Mp + t) * 2;
        pp[0] = S1; pp[1] = S2;
      }
    }
    __syncthreads();
  }
}

__device__ void phase1(const Params& p, float* smem) {
  const int tid = threadIdx.x;
  for (int i = blockIdx.x * NT + tid; i < p.B * p.C4; i += gridDim.x * NT) p.sumsq[i] = 0.f;
  transpose_mat(p.w1, p.C4, p.C, p.w1t, smem);
  transpose_mat(p.w2, p.C, p.C4, p.w2t, smem);
  if (p.p1_mode == 1) {
    if (p.p1_twr == 4) phase1_conv<1, 4>(p, smem); else phase1_conv<1, 1>(p, smem);
  } else {
    if (p.p1_twr == 4) phase1_conv<0, 4>(p, smem);
    else if (p.p1_twr == 2) phase1_conv<0, 2>(p, smem);
    else phase1_conv<0, 1>(p, smem);
  }
}

template <int BM, int BN, int WM, int WN, int KS, int BKc, int PHASE>
__device__ void gemm_phase(const Params& p, float* smem, float* sq_s, float* inv_s, float* red_s, int split) {
  constexpr int WS = WM * WN;
  constexpr int GT = WS * 32;
  constexpr int TM = BM / (WM * 8), TN = BN / (WN * 4);
  constexpr int AM4 = BM / 4, BN4 = BN / 4;
  constexpr int AQ = (BKc * AM4) / GT, BQ = (BKc * BN4) / GT;
  constexpr int STAGE = BKc * (BM + BN);
  constexpr int CS_LD = BM + 4;
  constexpr int NSTEP = NT / BM, NPT = BN / NSTEP;
  static_assert(WS * KS == 8, "8 warps");
  static_assert((BKc * AM4) % GT == 0 && (BKc * BN4) % GT == 0, "quad mapping");
  static_assert(GT % AM4 == 0, "token quad constant per thread");
  static_assert(KS * 2 * STAGE <= GEMM_SMEM_FLOATS && KS * BN * CS_LD <= GEMM_SMEM_FLOATS, "smem");
  static_assert(TM == 8 || TM == 4, "TM");
  static_assert(TN == 8 || TN == 6, "TN");

  const int tid = threadIdx.x, warp = tid >> 5, lane = tid & 31;
  const int kg = warp / WS, ws = warp - kg * WS, wm = ws / WN, wn = ws - wm * WN;
  const int lm = lane >> 2, ln = lane & 3;
  const int gt = ws * 32 + lane;
  const int M = p.M, Mp = p.Mp, HW = p.HW;
  const int N = (PHASE == 2) ? p.C4 : p.C;
  const int K = (PHASE == 2) ? p.C : p.C4;
  const float* __restrict__ Ag = (PHASE == 2) ? p.xn : p.hidden;
  const float* __restrict__ Bg = (PHASE == 2) ? p.w1t : p.w2t;
  const int mt = (M + BM - 1) / BM, nt = (N + BN - 1) / BN;
  const int ntiles = mt * nt * split, nk = K / BKc;
  float* As = smem + kg * 2 * STAGE;
  float* Bs = As + 2 * BKc * BM;
  const int row_m = wm * (BM / WM) + lm * TM;
  const int col_n = wn * (BN / WN) + ln * TN;
  const int a_m4 = (gt % AM4) * 4;
  const int a_kq = gt / AM4;
  constexpr int KSTEP_A = GT / AM4;

  for (int tile = blockIdx.x; tile < ntiles; tile += gridDim.x) {
    int rem = tile;
    const int ni = rem % nt; rem /= nt;
    const int mi = rem % mt; const int s = rem / mt;
    const int m0 = mi * BM, n0 = ni * BN;
    const int c0 = (int)(((long)nk * s) / split), c1 = (int)(((long)nk * (s + 1)) / split);
    const int b_first = m0 / HW;
    const int nimg = min(MAXIMG, (min(m0 + BM, M) - 1) / HW - b_first + 1);

    float tc0[4], tc1[4]; int timg[4]; bool tval[4];
#pragma unroll
    for (int e = 0; e < 4; e++) {
      const int t = m0 + a_m4 + e;
      tval[e] = t < M;
      const int tt = tval[e] ? t : 0;
      if (PHASE == 2) {
        float S1 = 0.f, S2 = 0.f;
        for (int ch = 0; ch < p.nch; ch++) {
          const float2 pp = __ldg(reinterpret_cast<const float2*>(p.part + ((size_t)ch * Mp + tt) * 2));
          S1 += pp.x; S2 += pp.y;
        }
        const float mean = S1 / (float)p.C;
        float var = S2 / (float)p.C - mean * mean; var = fmaxf(var, 0.f);
        tc0[e] = mean; tc1[e] = rsqrtf(var + p.ln_eps); timg[e] = 0;
      } else {
        timg[e] = tt / HW - b_first; tc0[e] = 0.f; tc1[e] = 0.f;
      }
    }
    if (PHASE == 3) {
      for (int img = 0; img < nimg; img++) {
        const float* sq = p.sumsq + (size_t)(b_first + img) * K;
        float sacc = 0.f;
        for (int k = tid; k < K; k += NT) sacc += sqrtf(sq[k]);
        sacc = warp_sum(sacc);
        if (lane == 0) red_s[warp] = sacc;
        __syncthreads();
        if (tid == 0) {
          float tot = 0.f;
#pragma unroll
          for (int w2 = 0; w2 < 8; w2++) tot += red_s[w2];
          inv_s[img] = 1.f / (tot / (float)K + p.eps);
        }
        __syncthreads();
      }
#pragma unroll
      for (int e = 0; e < 4; e++) tc1[e] = inv_s[timg[e] < MAXIMG ? timg[e] : 0];
    }

    float acc[TM][TN];
#pragma unroll
    for (int i = 0; i < TM; i++)
#pragma unroll
      for (int j = 0; j < TN; j++) acc[i][j] = 0.f;

    float4 ra[AQ], rb[BQ];
    auto load_chunk = [&](int kc) {
      const int k0 = kc * BKc;
#pragma unroll
      for (int q = 0; q < AQ; q++) {
        const int ka = k0 + q * KSTEP_A + a_kq;
        float4 v = __ldg(reinterpret_cast<const float4*>(Ag + (size_t)ka * Mp + m0 + a_m4));
        float vv[4] = {v.x, v.y, v.z, v.w};
        if (PHASE == 2) {
          const float lw = __ldg(p.lnw + ka), lb = __ldg(p.lnb + ka);
#pragma unroll
          for (int e = 0; e < 4; e++) vv[e] = tval[e] ? fmaf((vv[e] - tc0[e]) * tc1[e], lw, lb) : 0.f;
        } else {
          const float gwk = __ldg(p.gw + ka), gbk = __ldg(p.gb + ka);
#pragma unroll
          for (int e = 0; e < 4; e++) {
            const float sq = __ldg(p.sumsq + (size_t)(b_first + timg[e]) * K + ka);
            vv[e] = tval[e] ? fmaf(vv[e], fmaf(gwk * tc1[e], sqrtf(sq), 1.f), gbk) : 0.f;
          }
        }
        ra[q] = make_float4(vv[0], vv[1], vv[2], vv[3]);
      }
#pragma unroll
      for (int q = 0; q < BQ; q++) {
        const int id = q * GT + gt, kb = id / BN4, n4 = (id - kb * BN4) * 4;
        const int col = n0 + n4;
        rb[q] = (col < N) ? __ldg(reinterpret_cast<const float4*>(Bg + (size_t)(k0 + kb) * N + col))
                          : make_float4(0.f, 0.f, 0.f, 0.f);
      }
    };
    auto store_regs = [&](int buf) {
#pragma unroll
      for (int q = 0; q < AQ; q++)
        *reinterpret_cast<float4*>(As + (buf * BKc + q * KSTEP_A + a_kq) * BM + a_m4) = ra[q];
#pragma unroll
      for (int q = 0; q < BQ; q++) {
        const int id = q * GT + gt, kb = id / BN4, n4 = (id - kb * BN4) * 4;
        *reinterpret_cast<float4*>(Bs + (buf * BKc + kb) * BN + n4) = rb[q];
      }
    };
    auto gsync = [&]() {
      if (KS == 1) __syncthreads();
      else if (GT == 32) __syncwarp();
      else asm volatile("bar.sync %0, %1;" ::"r"(kg + 1), "r"(GT));
    };

    int kc = c0 + kg;
    if (kc < c1) { load_chunk(kc); store_regs(0); }
    gsync();
    for (int it = 0; kc < c1; it++, kc += KS) {
      const int buf = it & 1;
      const bool more = (kc + KS) < c1;
      if (more) load_chunk(kc + KS);
      const float* Ab = As + buf * BKc * BM + row_m;
      const float* Bb = Bs + buf * BKc * BN + col_n;
#pragma unroll 4
      for (int k = 0; k < BKc; k++) {
        float a[TM], b[TN];
        {
          const float4 a0 = *reinterpret_cast<const float4*>(Ab + k * BM);
          a[0] = a0.x; a[1] = a0.y; a[2] = a0.z; a[3] = a0.w;
          if (TM == 8) {
            const float4 a1 = *reinterpret_cast<const float4*>(Ab + k * BM + 4);
            a[4] = a1.x; a[5] = a1.y; a[6] = a1.z; a[7] = a1.w;
          }
        }
        if (TN == 8) {
          const float4 b0 = *reinterpret_cast<const float4*>(Bb + k * BN);
          const float4 b1 = *reinterpret_cast<const float4*>(Bb + k * BN + 4);
          b[0] = b0.x; b[1] = b0.y; b[2] = b0.z; b[3] = b0.w; b[4] = b1.x; b[5] = b1.y; b[6] = b1.z; b[7] = b1.w;
        } else {
          const float2 b0 = *reinterpret_cast<const float2*>(Bb + k * BN);
          const float2 b1 = *reinterpret_cast<const float2*>(Bb + k * BN + 2);
          const float2 b2 = *reinterpret_cast<const float2*>(Bb + k * BN + 4);
          b[0] = b0.x; b[1] = b0.y; b[2] = b1.x; b[3] = b1.y; b[4] = b2.x; b[5] = b2.y;
        }
#pragma unroll
        for (int i = 0; i < TM; i++)
#pragma unroll
          for (int j = 0; j < TN; j++) acc[i][j] = fmaf(a[i], b[j], acc[i][j]);
      }
      if (more) store_regs(buf ^ 1);
      gsync();
    }

    __syncthreads();
    float* Cs = smem;
#pragma unroll
    for (int j = 0; j < TN; j++) {
      float* dst = Cs + (kg * BN + col_n + j) * CS_LD + row_m;
      *reinterpret_cast<float4*>(dst) = make_float4(acc[0][j], acc[1][j], acc[2][j], acc[3][j]);
      if (TM == 8) *reinterpret_cast<float4*>(dst + 4) = make_float4(acc[4][j], acc[5][j], acc[6][j], acc[7][j]);
    }
    __syncthreads();
    const int m_l = tid % BM, n_l0 = tid / BM;
    const int t = m0 + m_l;
    const bool tv = t < M;
    const int img = tv ? t / HW : b_first;
    const int hw = t - img * HW;
    if (PHASE == 2) {
      const int img0 = __shfl_sync(0xffffffffu, img, 0);
      const bool uniform = __all_sync(0xffffffffu, (img == img0) && tv);
      const int il0 = img0 - b_first;
#pragma unroll 4
      for (int i = 0; i < NPT; i++) {
        const int n_l = n_l0 + NSTEP * i, col = n0 + n_l;
        float v = 0.f;
#pragma unroll
        for (int g = 0; g < KS; g++) v += Cs[(g * BN + n_l) * CS_LD + m_l];
        const float gl = gelu_f(v + __ldg(p.b1 + col));
        if (tv) p.hidden[(size_t)col * Mp + t] = gl;
        float g2 = tv ? gl * gl : 0.f;
        if (uniform) {
          g2 = warp_sum(g2);
          if (lane == 0) {
            if (il0 < MAXIMG) atomicAdd(&sq_s[il0 * BN + n_l], g2);
            else atomicAdd(p.sumsq + (size_t)img0 * N + col, g2);
          }
        } else if (tv) {
          const int il = img - b_first;
          if (il < MAXIMG) atomicAdd(&sq_s[il * BN + n_l], g2);
          else atomicAdd(p.sumsq + (size_t)img * N + col, g2);
        }
      }
      __syncthreads();
      for (int idx = tid; idx < nimg * BN; idx += NT) {
        const int im = idx / BN, n = idx - im * BN;
        const float v = sq_s[idx]; sq_s[idx] = 0.f;
        atomicAdd(p.sumsq + (size_t)(b_first + im) * N + n0 + n, v);
      }
      __syncthreads();
    } else {
#pragma unroll 4
      for (int i = 0; i < NPT; i++) {
        const int n_l = n_l0 + NSTEP * i, col = n0 + n_l;
        if (col < N && tv) {
          float v = 0.f;
#pragma unroll
          for (int g = 0; g < KS; g++) v += Cs[(g * BN + n_l) * CS_LD + m_l];
          const size_t idx = ((size_t)img * N + col) * HW + hw;
          if (split == 1) p.out[idx] = v + __ldg(p.b2 + col) + __ldg(p.x + idx);
          else atomicAdd(p.out + idx, v);
        }
      }
      __syncthreads();
    }
  }
}

template <int PHASE>
__device__ __forceinline__ void run_gemm(const Params& p, int cfg, int split, float* smem, float* sq_s, float* inv_s, float* red_s) {
  switch (cfg) {
    case 0: gemm_phase<128, 128, 2, 4, 1, 32, PHASE>(p, smem, sq_s, inv_s, red_s, split); break;
    case 1: gemm_phase<128, 96, 2, 4, 1, 32, PHASE>(p, smem, sq_s, inv_s, red_s, split); break;
    case 2: gemm_phase<64, 64, 2, 2, 2, 32, PHASE>(p, smem, sq_s, inv_s, red_s, split); break;
    default: gemm_phase<32, 32, 1, 1, 8, 16, PHASE>(p, smem, sq_s, inv_s, red_s, split); break;
  }
}

__global__ void __launch_bounds__(NT, 2) convnext_fused(Params p) {
  extern __shared__ float4 smem4[];
  float* smem = reinterpret_cast<float*>(smem4);
  float* sq_s = smem + GEMM_SMEM_FLOATS;
  float* inv_s = sq_s + MAXIMG * 128;
  float* red_s = inv_s + 8;
  for (int i = threadIdx.x; i < MAXIMG * 128; i += NT) sq_s[i] = 0.f;
  __syncthreads();
  cg::grid_group grid = cg::this_grid();

  phase1(p, smem);
  grid.sync();
  if (p.split3 > 1) {
    const long total = (long)p.M * p.C;
    for (long i = (long)blockIdx.x * NT + threadIdx.x; i < total; i += (long)gridDim.x * NT) {
      const int c = (int)((i / p.HW) % p.C);
      p.out[i] = p.x[i] + p.b2[c];
    }
  }
  run_gemm<2>(p, p.cfg2, 1, smem, sq_s, inv_s, red_s);
  grid.sync();
  run_gemm<3>(p, p.cfg3, p.split3, smem, sq_s, inv_s, red_s);
}

struct Pick { int cfg; int split; };
static Pick pick_cfg(long M, long N, long K, int grid, bool allow_split) {
  struct CfgD { int id, BM, BN, BK; double eff; };
  const CfgD cfgs[4] = {{0, 128, 128, 32, 0.80}, {1, 128, 96, 32, 0.78}, {2, 64, 64, 32, 0.60}, {3, 32, 32, 16, 0.40}};
  double best = 1e300; Pick r{0, 1};
  for (int ci = 0; ci < 4; ci++) {
    const CfgD& c = cfgs[ci];
    if (c.id == 1 && (N % 96) != 0) continue;
    const long tiles = ((M + c.BM - 1) / c.BM) * ((N + c.BN - 1) / c.BN);
    const long nk = K / c.BK;
    for (int S = 1; S <= (allow_split ? 16 : 1); S *= 2) {
      if (S > 1 && nk / S < 2) break;
      const long total = tiles * S;
      const long waves = (total + grid - 1) / grid;
      double cost = (double)c.BM * c.BN * ((double)K / S) / c.eff * (double)waves;
      if (S > 1) cost *= 1.08 + 0.03 * S;
      if (cost < best) { best = cost; r.cfg = c.id; r.split = S; }
    }
  }
  return r;
}

static const char* launch(Params p, void* stream_v) {
  static int grid_size = 0;
  cudaError_t e;
  if (grid_size == 0) {
    e = cudaFuncSetAttribute(convnext_fused, cudaFuncAttributeMaxDynamicSharedMemorySize, SMEM_BYTES);
    if (e != cudaSuccess) return cudaGetErrorString(e);
    int dev = 0; cudaGetDevice(&dev);
    int sms = 0; cudaDeviceGetAttribute(&sms, cudaDevAttrMultiProcessorCount, dev);
    int bps = 0;
    e = cudaOccupancyMaxActiveBlocksPerMultiprocessor(&bps, convnext_fused, NT, SMEM_BYTES);
    if (e != cudaSuccess) return cudaGetErrorString(e);
    if (bps < 1) bps = 1;
    grid_size = sms * bps;
  }
  const Pick k2 = pick_cfg(p.M, p.C4, p.C, grid_size, false);
  const Pick k3 = pick_cfg(p.M, p.C, p.C4, grid_size, true);
  p.cfg2 = k2.cfg; p.cfg3 = k3.cfg; p.split3 = k3.split;
  {
    const int tw = (p.W % 4 == 0) ? 4 : 1;
    const int cgt = p.W / tw;
    int lpr, rg, wb;
    if (cgt >= 32) { lpr = 32; rg = 1; wb = (cgt + 31) / 32; }
    else { lpr = cgt; rg = 32 / cgt; wb = 1; }
    if (rg > 8) rg = 8;
    const int rows = 4 * rg;
    if (rows <= p.H) {
      p.p1_mode = 1; p.p1_twr = tw; p.p1_lpr = lpr; p.p1_rg = rg; p.p1_wb = wb;
      p.p1_hb = (p.H + rows - 1) / rows;
      p.p1_tiles = p.B * p.nch * p.p1_hb * p.p1_wb;
    } else {
      p.p1_mode = 0; p.p1_lpr = 0; p.p1_rg = 0; p.p1_wb = 0; p.p1_hb = 0;
      int r0 = 4;
      while (r0 > 1 && ((p.M + 32 * r0 - 1) / (32 * r0)) * p.nch < grid_size) r0 >>= 1;
      p.p1_twr = r0;
      p.p1_tiles = ((p.M + 32 * r0 - 1) / (32 * r0)) * p.nch;
    }
  }
  void* args[] = {&p};
  e = cudaLaunchCooperativeKernel((const void*)convnext_fused, dim3(grid_size), dim3(NT), args, (size_t)SMEM_BYTES,
                                  (cudaStream_t)stream_v);
  if (e != cudaSuccess) return cudaGetErrorString(e);
  return nullptr;
}
```

```cpp file=binding.cpp
static const size_t WS_BYTES = 32u << 20;
static cublasLtHandle_t lt() { static cublasLtHandle_t h = nullptr; if (!h) LT_CHECK(cublasLtCreate(&h)); return h; }

struct LtGemm {
  cublasLtMatmulDesc_t op = nullptr;
  cublasLtMatrixLayout_t a = nullptr, b = nullptr, c = nullptr, d = nullptr;
  cublasLtMatmulAlgo_t algo;
  bool bias = false;
};
static cublasLtMatrixLayout_t mk_layout(cudaDataType t, uint64_t rows, uint64_t cols, int64_t ld, int batch, int64_t stride) {
  cublasLtMatrixLayout_t L;
  LT_CHECK(cublasLtMatrixLayoutCreate(&L, t, rows, cols, ld));
  if (batch > 1) {
    int32_t bc = batch;
    LT_CHECK(cublasLtMatrixLayoutSetAttribute(L, CUBLASLT_MATRIX_LAYOUT_BATCH_COUNT, &bc, sizeof(bc)));
    LT_CHECK(cublasLtMatrixLayoutSetAttribute(L, CUBLASLT_MATRIX_LAYOUT_STRIDED_BATCH_OFFSET, &stride, sizeof(stride)));
  }
  return L;
}
static void mk_gemm(LtGemm& g, bool bias, uint32_t align_cd) {
  LT_CHECK(cublasLtMatmulDescCreate(&g.op, CUBLAS_COMPUTE_32F, CUDA_R_32F));
  cublasOperation_t ta = CUBLAS_OP_T, tb = CUBLAS_OP_N;
  LT_CHECK(cublasLtMatmulDescSetAttribute(g.op, CUBLASLT_MATMUL_DESC_TRANSA, &ta, sizeof(ta)));
  LT_CHECK(cublasLtMatmulDescSetAttribute(g.op, CUBLASLT_MATMUL_DESC_TRANSB, &tb, sizeof(tb)));
  g.bias = bias;
  if (bias) {
    cublasLtEpilogue_t ep = CUBLASLT_EPILOGUE_BIAS;
    LT_CHECK(cublasLtMatmulDescSetAttribute(g.op, CUBLASLT_MATMUL_DESC_EPILOGUE, &ep, sizeof(ep)));
    cudaDataType bt = CUDA_R_32F;
    LT_CHECK(cublasLtMatmulDescSetAttribute(g.op, CUBLASLT_MATMUL_DESC_BIAS_DATA_TYPE, &bt, sizeof(bt)));
  }
  cublasLtMatmulPreference_t pref;
  LT_CHECK(cublasLtMatmulPreferenceCreate(&pref));
  size_t ws = WS_BYTES;
  LT_CHECK(cublasLtMatmulPreferenceSetAttribute(pref, CUBLASLT_MATMUL_PREF_MAX_WORKSPACE_BYTES, &ws, sizeof(ws)));
  uint32_t a16 = 16;
  LT_CHECK(cublasLtMatmulPreferenceSetAttribute(pref, CUBLASLT_MATMUL_PREF_MIN_ALIGNMENT_A_BYTES, &a16, sizeof(a16)));
  LT_CHECK(cublasLtMatmulPreferenceSetAttribute(pref, CUBLASLT_MATMUL_PREF_MIN_ALIGNMENT_B_BYTES, &a16, sizeof(a16)));
  LT_CHECK(cublasLtMatmulPreferenceSetAttribute(pref, CUBLASLT_MATMUL_PREF_MIN_ALIGNMENT_C_BYTES, &align_cd, sizeof(align_cd)));
  LT_CHECK(cublasLtMatmulPreferenceSetAttribute(pref, CUBLASLT_MATMUL_PREF_MIN_ALIGNMENT_D_BYTES, &align_cd, sizeof(align_cd)));
  cublasLtMatmulHeuristicResult_t res[8];
  int n = 0;
  LT_CHECK(cublasLtMatmulAlgoGetHeuristic(lt(), g.op, g.a, g.b, g.c, g.d, pref, 8, res, &n));
  cublasLtMatmulPreferenceDestroy(pref);
  if (n <= 0) throw std::runtime_error("cublasLt: no algorithm");
  g.algo = res[0].algo;
}
// pointers (bias included) are set on every call: nothing call-specific is kept in a plan
static void lt_run(LtGemm& g, const void* A, const void* Bm, const void* Cm, void* D, float beta, const void* bias,
                   void* ws, cudaStream_t st) {

  const float alpha = 1.f;

}

// G2 form per shape: NCHW swap-AB batched (x as C, beta 1) for 28x28 images, channels-last + K4 transpose-add else
static int g2_mode(int HW) { return HW == 784 ? 1 : 0; }

struct FPlan { LtGemm g1, g2; };
static std::map<std::tuple<int, int, int, int>, FPlan*> g_plans;  // keyed by shape only
static FPlan* get_plan(int B, int C, int H, int W) {
  auto key = std::make_tuple(B, C, H, W);
  auto it = g_plans.find(key);
  if (it != g_plans.end()) return it->second;
  const int HW = H * W, M = B * HW, C4 = 4 * C, K2 = C4 + 8;
  FPlan* pl = new FPlan();
  // G1: z[M][4C] (col-major 4C x M) = op_T(W1h) (4C x C) * a16 (C x M) + b1
  pl->g1.a = mk_layout(CUDA_R_16F, C, C4, C, 1, 0);
  pl->g1.b = mk_layout(CUDA_R_16F, C, M, C, 1, 0);
  pl->g1.c = mk_layout(CUDA_R_32F, C4, M, C4, 1, 0);
  pl->g1.d = mk_layout(CUDA_R_32F, C4, M, C4, 1, 0);
  mk_gemm(pl->g1, true, 16);
  if (g2_mode(HW) == 0) {  // channels-last y[M][C] = op_T(W2h) (C x K2) * h' (K2 x M)
    pl->g2.a = mk_layout(CUDA_R_16F, K2, C, K2, 1, 0);
    pl->g2.b = mk_layout(CUDA_R_16F, K2, M, K2, 1, 0);
    pl->g2.c = mk_layout(CUDA_R_32F, C, M, C, 1, 0);
    pl->g2.d = mk_layout(CUDA_R_32F, C, M, C, 1, 0);
    mk_gemm(pl->g2, false, 16);
  } else {                 // per image: out_b (HW x C col-major = NCHW) = op_T(h'_b) (HW x K2) * W2h^T (K2 x C) + x_b
    pl->g2.a = mk_layout(CUDA_R_16F, K2, HW, K2, B, (int64_t)HW * K2);
    pl->g2.b = mk_layout(CUDA_R_16F, K2, C, K2, B, 0);
    pl->g2.c = mk_layout(CUDA_R_32F, HW, C, HW, B, (int64_t)C * HW);
    pl->g2.d = mk_layout(CUDA_R_32F, HW, C, HW, B, (int64_t)C * HW);
    mk_gemm(pl->g2, false, (HW % 4 == 0) ? 16 : 4);
  }
  g_plans[key] = pl;
  return pl;
}

static void f_run(const torch::Tensor& x, const torch::Tensor& dww, const torch::Tensor& dwb, const torch::Tensor& lnw,
                  const torch::Tensor& lnb, const torch::Tensor& w1, const torch::Tensor& b1, const torch::Tensor& gw,

                  torch::Tensor& out, cudaStream_t st) {
  static int nsm = 0;
  if (nsm == 0) { int dev = 0; cudaGetDevice(&dev); cudaDeviceGetAttribute(&nsm, cudaDevAttrMultiProcessorCount, dev); }
  const int B = (int)x.size(0), C = (int)x.size(1), H = (int)x.size(2), W = (int)x.size(3);
  const int HW = H * W, M = B * HW, C4 = 4 * C, K2 = C4 + 8, CPC = C / 8;
  int NS, RS;
  if (HW == 784) { NS = 4; RS = 7; } else { NS = 1; RS = H; }
  int KS = 8;  // K3 slice width: about 384 CTAs (B x 4C/KS), 8..256 columns
  while (KS < 256 && (long)B * C4 / (KS * 2) >= 384) KS *= 2;
  const int PS = (NS == 1) ? HW : (RS + 6) * W, T = RS * W;
  const int mode = g2_mode(HW);
  FPlan* pl = get_plan(B, C, H, W);

  const size_t o_a16 = 0;
  const size_t o_w1h = al256(o_a16 + (size_t)M * C * 2);
  const size_t o_w2h = al256(o_w1h + (size_t)C4 * C * 2);
  const size_t o_z = al256(o_w2h + (size_t)C * K2 * 2);
  const size_t o_hp = al256(o_z + (size_t)M * C4 * 4);
  const size_t o_y = al256(o_hp + (size_t)M * K2 * 2);
  const size_t o_ws = al256(o_y + (mode == 0 ? (size_t)M * C * 4 : 0));
  const size_t total = o_ws + WS_BYTES;
  torch::Tensor wsT = torch::empty({(long)total}, x.options().dtype(torch::kUInt8));
  char* wp = (char*)wsT.data_ptr();

  K1P p1;
  p1.x = x.data_ptr<float>(); p1.dww = dww.data_ptr<float>(); p1.dwb = dwb.data_ptr<float>();
  p1.lnw = lnw.data_ptr<float>(); p1.lnb = lnb.data_ptr<float>(); p1.w1 = w1.data_ptr<float>(); p1.w2 = w2.data_ptr<float>();
  p1.gb = gb.data_ptr<float>(); p1.b2 = b2.data_ptr<float>();

  }
}

static void r0_run(const torch::Tensor& x, const torch::Tensor& dwconv_weight, const torch::Tensor& dwconv_bias,
                   const torch::Tensor& layernorm_weight, const torch::Tensor& layernorm_bias, const torch::Tensor& pwconv1_weight,
                   const torch::Tensor& pwconv1_bias, const torch::Tensor& grn_weight, const torch::Tensor& grn_bias,
                   const torch::Tensor& pwconv2_weight, const torch::Tensor& pwconv2_bias, double eps, double layer_norm_eps,
                   torch::Tensor& output, cudaStream_t stream) {
  const int B = (int)x.size(0), C = (int)x.size(1), H = (int)x.size(2), W = (int)x.size(3);
  TORCH_CHECK(C % 32 == 0, "dim must be a multiple of 32");
  TORCH_CHECK(H * W >= 32, "H*W must be >= 32");
  const int HW = H * W, M = B * HW, C4 = 4 * C, nch = C / 32;
  const int Mp = ((M + 127) / 128) * 128;
  const size_t off_xn = 0;
  const size_t off_part = al256(off_xn + (size_t)C * Mp * 4);
  const size_t off_hidden = al256(off_part + (size_t)nch * Mp * 2 * 4);
  const size_t off_sumsq = al256(off_hidden + (size_t)C4 * Mp * 4);
  const size_t off_w1t = al256(off_sumsq + (size_t)B * C4 * 4);
  const size_t off_w2t = al256(off_w1t + (size_t)C * C4 * 4);
  const size_t total = al256(off_w2t + (size_t)C4 * C * 4);
  torch::Tensor ws = torch::empty({(long)total}, x.options().dtype(torch::kUInt8));
  char* wp = (char*)ws.data_ptr();
  R0Params p;
  p.x = x.data_ptr<float>(); p.dww = dwconv_weight.data_ptr<float>(); p.dwb = dwconv_bias.data_ptr<float>();
  p.lnw = layernorm_weight.data_ptr<float>(); p.lnb = layernorm_bias.data_ptr<float>();
  p.w1 = pwconv1_weight.data_ptr<float>(); p.b1 = pwconv1_bias.data_ptr<float>();
  p.gw = grn_weight.data_ptr<float>(); p.gb = grn_bias.data_ptr<float>();
  p.w2 = pwconv2_weight.data_ptr<float>(); p.b2 = pwconv2_bias.data_ptr<float>();
  p.out = output.data_ptr<float>();
  p.xn = (float*)(wp + off_xn); p.part = (float*)(wp + off_part); p.hidden = (float*)(wp + off_hidden);
  p.sumsq = (float*)(wp + off_sumsq); p.w1t = (float*)(wp + off_w1t); p.w2t = (float*)(wp + off_w2t);
  p.B = B; p.C = C; p.H = H; p.W = W; p.HW = HW; p.M = M; p.Mp = Mp; p.C4 = C4; p.nch = nch;
  p.eps = (float)eps; p.ln_eps = (float)layer_norm_eps;
  p.p1_mode = 0; p.p1_twr = 1; p.p1_tiles = 0; p.p1_lpr = 0; p.p1_rg = 0; p.p1_hb = 0; p.p1_wb = 0;
  p.cfg2 = 0; p.cfg3 = 0; p.split3 = 1;
  ck(r0_launch(p, (void*)stream), "r0_launch");
}

void run(torch::Tensor x, torch::Tensor dwconv_weight, torch::Tensor dwconv_bias, torch::Tensor layernorm_weight,
         torch::Tensor layernorm_bias, torch::Tensor pwconv1_weight, torch::Tensor pwconv1_bias, torch::Tensor grn_weight,
         torch::Tensor grn_bias, torch::Tensor pwconv2_weight, torch::Tensor pwconv2_bias, double eps, double layer_norm_eps,
         torch::Tensor output) {
  TORCH_CHECK(x.is_cuda() && x.dtype() == torch::kFloat32 && x.is_contiguous() && x.dim() == 4, "x must be contiguous fp32 NCHW");
  TORCH_CHECK(output.is_contiguous() && output.dtype() == torch::kFloat32, "output must be contiguous fp32");
  TORCH_CHECK(pwconv1_weight.is_contiguous() && pwconv2_weight.is_contiguous() && dwconv_weight.is_contiguous(), "weights must be contiguous");
  const c10::cuda::OptionalCUDAGuard guard(x.device());
  cudaStream_t st = at::cuda::getCurrentCUDAStream(x.device().index()).stream();
  const int B = (int)x.size(0), C = (int)x.size(1), H = (int)x.size(2), W = (int)x.size(3);
  if (f_supported(B, C, H, W))
    f_run(x, dwconv_weight, dwconv_bias, layernorm_weight, layernorm_bias, pwconv1_weight, pwconv1_bias, grn_weight, grn_bias,
          pwconv2_weight, pwconv2_bias, eps, layer_norm_eps, output, st);
  else
    r0_run(x, dwconv_weight, dwconv_bias, layernorm_weight, layernorm_bias, pwconv1_weight, pwconv1_bias, grn_weight, grn_bias,
           pwconv2_weight, pwconv2_bias, eps, layer_norm_eps, output, st);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) { m.def("run", &run, "ConvNeXtV2 block with GRN (lineage F + r0 fallback)"); }
```
