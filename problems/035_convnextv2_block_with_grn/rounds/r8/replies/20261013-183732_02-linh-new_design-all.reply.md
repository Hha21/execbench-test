### Rationale
I built lineage H's image-counter dataflow kernel and tested it in full. The thesis is refuted for this form of the design. In the same container, all five rows ran 58–98% slower than r7-merge. The probe timestamps show why: per-item CTA-time summed over the queue and divided by 148 SMs is already 47, 111.5 and 71.9 µs at 8,768, 32,768 and 16,384. Lineage D runs those whole rows in 34.4, 63.8 and 57.5 µs, so even perfect packing could not win.

The cost is not in claiming or waiting: the dependency wait is about 1.3–1.5 µs median and the gap between items 1.5–3.3 µs. It is in the work each item does on its own CTA, one step after another: a GEMM1 tile takes 11.8–15.9 µs and a GEMM2 tile about 14.6 µs. The tcgen05 engine was correct on the first build. Precision is 58–64% of atol. The candidate is a merge-only kernel: the other 11 rows run the r0 anchor, and the merge should take none of the five rows.

// ConvNeXtV2 block with GRN: one cooperative launch, three phases, fp32 SIMT maths.
//  P1: depthwise 7x7 conv (+bias) -> xn [C][Mp] (K-major), per-(token, 32-channel chunk) LN partial sums;
//      zero GRN sums; transpose weights to [K][N]
//  P2: hidden[4C][Mp] = gelu(LN(xn) @ W1^T + b1); sumsq[b][k] += hidden^2 (per image)
//  P3: out = x + (hidden * (1 + gw*nf) + gb) @ W2^T + b2, written NCHW (split-K with atomics at small sizes)
#include <cuda_runtime.h>
#include <cooperative_groups.h>
namespace cg = cooperative_groups;

#define NT 256
#define MAXIMG 5
#define GEMM_SMEM_FLOATS (128 * 132)
#define SMEM_BYTES (GEMM_SMEM_FLOATS * 4 + MAXIMG * 128 * 4 + 64)
#define P1_SLOTS 512

struct Params {
  const float* x; const float* dww; const float* dwb; const float* lnw; const float* lnb;
  const float* w1; const float* b1; const float* gw; const float* gb; const float* w2; const float* b2;
  float* out;
  float* xn; float* part; float* hidden; float* sumsq; float* w1t; float* w2t;
  int B, C, H, W, HW, M, Mp, C4, nch;
  float eps, ln_eps;
  int p1_mode, p1_twr, p1_tiles, p1_lpr, p1_rg, p1_hb, p1_wb;
  int cfg2, cfg3, split3;
};

__device__ __forceinline__ float gelu_f(float v) { return 0.5f * v * (1.f + erff(v * 0.70710678118654752f)); }

__device__ __forceinline__ float warp_sum(float v) {
#pragma unroll
  for (int o = 16; o > 0; o >>= 1) v += __shfl_xor_sync(0xffffffffu, v, o);
  return v;
}

// ---------------------------------------------------------------- phase 1
__device__ void transpose_mat(const float* __restrict__ src, int R, int Cc, float* __restrict__ dst, float* tile) {
  const int rt = (R + 31) / 32, ct = (Cc + 31) / 32;
  const int tid = threadIdx.x, j = tid & 31, i0 = tid >> 5;
  for (int t = blockIdx.x; t < rt * ct; t += gridDim.x) {
    const int tr = t / ct, tc = t % ct;
#pragma unroll
    for (int s = 0; s < 4; s++) {
      const int i = i0 + 8 * s, r = tr * 32 + i, c = tc * 32 + j;
      tile[i * 33 + j] = (r < R && c < Cc) ? src[(size_t)r * Cc + c] : 0.f;
    }
    __syncthreads();
#pragma unroll
    for (int s = 0; s < 4; s++) {
      const int i = i0 + 8 * s, c = tc * 32 + i, r = tr * 32 + j;
      if (r < R && c < Cc) dst[(size_t)c * R + r] = tile[j * 33 + i];
    }
    __syncthreads();
  }
}

// 4 rows x TW cols output patch of one channel; wt = 49 weights (registers)
template <int TW>
__device__ __forceinline__ void conv_patch(const float* __restrict__ xc, const float* wt, int H, int W, int h0, int w0,
                                           float (&acc)[4][TW]) {
#pragma unroll
  for (int r = 0; r < 4; r++)
#pragma unroll
    for (int cc = 0; cc < TW; cc++) acc[r][cc] = 0.f;
#pragma unroll
  for (int i = 0; i < 10; i++) {
    const int hh = h0 - 3 + i;
    const bool rowok = (hh >= 0) && (hh < H);
    float v[TW + 6];
    if (TW == 4) {
      const float* rp = xc + hh * W;
      float4 L = make_float4(0.f, 0.f, 0.f, 0.f), Cq = L, R = L;
      if (rowok) {
        if (w0 >= 4) L = __ldg(reinterpret_cast<const float4*>(rp + w0 - 4));
        Cq = __ldg(reinterpret_cast<const float4*>(rp + w0));
        if (w0 + 4 < W) R = __ldg(reinterpret_cast<const float4*>(rp + w0 + 4));
      }
      v[0] = L.y; v[1] = L.z; v[2] = L.w; v[3] = Cq.x; v[4] = Cq.y; v[5] = Cq.z; v[6] = Cq.w;
      v[7] = R.x; v[8] = R.y; v[9] = R.z;
    } else {
#pragma unroll
      for (int jj = 0; jj < TW + 6; jj++) {
        const int ww = w0 - 3 + jj;
        v[jj] = (rowok && ww >= 0 && ww < W) ? __ldg(xc + hh * W + ww) : 0.f;
      }
    }
#pragma unroll
    for (int r = 0; r < 4; r++) {
      const int ki = i - r;
      if (ki >= 0 && ki < 7) {
#pragma unroll
        for (int cc = 0; cc < TW; cc++)
#pragma unroll
          for (int j = 0; j < 7; j++) acc[r][cc] = fmaf(v[cc + j], wt[ki * 7 + j], acc[r][cc]);
      }
    }
  }
}

// MODE 1: patch tiles (TWR = TW in {4,1}); MODE 0: flattened token tiles (TWR = tokens per lane in {1,2,4})
template <int MODE, int TWR>
__device__ void phase1_conv(const Params& p, float* smem) {
  const int tid = threadIdx.x, warp = tid >> 5, lane = tid & 31;
  const int C = p.C, H = p.H, W = p.W, HW = p.HW, M = p.M, Mp = p.Mp, nch = p.nch;
  float* st_s = smem;                     // [8][P1_SLOTS][2]
  float* wsm = smem + 8 * P1_SLOTS * 2;   // [32][49]
  const int lpr = p.p1_lpr, rg = p.p1_rg, hb = p.p1_hb, wb = p.p1_wb;
  const int rows_per_tile = 4 * rg;
  const int rpt = lpr * TWR;  // slots per row (mode 1)

  for (int tile = blockIdx.x; tile < p.p1_tiles; tile += gridDim.x) {
    int b = 0, cch = 0, hbi = 0, wbi = 0, tok0 = 0;
    if (MODE == 1) {
      const int per = nch * hb * wb;
      b = tile / per; int rem = tile - b * per;
      cch = rem / (hb * wb); rem -= cch * (hb * wb);
      hbi = rem / wb; wbi = rem - hbi * wb;
    } else {
      cch = tile % nch; tok0 = (tile / nch) * (32 * TWR);
    }
    for (int i = tid; i < 8 * P1_SLOTS * 2; i += NT) st_s[i] = 0.f;
    for (int i = tid; i < 32 * 49; i += NT) wsm[i] = __ldg(p.dww + (size_t)cch * 32 * 49 + i);
    __syncthreads();

    int rgi = 0, cgi = 0, h0 = 0, w0 = 0; bool lane_ok = true;
    if (MODE == 1) {
      rgi = lane / lpr; cgi = lane - rgi * lpr;
      lane_ok = rgi < rg;
      h0 = hbi * rows_per_tile + rgi * 4;
      w0 = (wbi * 32 + cgi) * TWR;
      lane_ok = lane_ok && (w0 < W);
    }
    int tk[MODE == 0 ? TWR : 1], tb[MODE == 0 ? TWR : 1], th[MODE == 0 ? TWR : 1], tw_[MODE == 0 ? TWR : 1];
    bool tv[MODE == 0 ? TWR : 1];
    if (MODE == 0) {
#pragma unroll
      for (int r = 0; r < TWR; r++) {
        const int t = tok0 + r * 32 + lane;
        tv[r] = t < M;
        const int tt = tv[r] ? t : 0;
        const int bb = tt / HW; const int rem = tt - bb * HW; const int hh = rem / W;
        tk[r] = t; tb[r] = bb; th[r] = hh; tw_[r] = rem - hh * W;
      }
    }

#pragma unroll 1
    for (int cl = 0; cl < 4; cl++) {
      const int clc = warp * 4 + cl;
      const int c = cch * 32 + clc;
      float wt[49];
#pragma unroll
      for (int i = 0; i < 49; i++) wt[i] = wsm[clc * 49 + i];
      const float bias = __ldg(p.dwb + c);
      float* xrow = p.xn + (size_t)c * Mp;
      float* sts = st_s + warp * P1_SLOTS * 2;
      if (MODE == 1) {
        if (lane_ok) {
          float acc[4][TWR];
          conv_patch<TWR>(p.x + ((size_t)b * C + c) * HW, wt, H, W, h0, w0, acc);
#pragma unroll
          for (int r = 0; r < 4; r++) {
            const int h = h0 + r;
            if (h < H) {
              const int t = b * HW + h * W + w0;
              const int slot = (rgi * 4 + r) * rpt + cgi * TWR;
              if (TWR == 4) {
                float4 o = make_float4(acc[r][0] + bias, acc[r][1] + bias, acc[r][2] + bias, acc[r][3] + bias);
                *reinterpret_cast<float4*>(xrow + t) = o;
                sts[(slot + 0) * 2] += o.x; sts[(slot + 0) * 2 + 1] += o.x * o.x;
                sts[(slot + 1) * 2] += o.y; sts[(slot + 1) * 2 + 1] += o.y * o.y;
                sts[(slot + 2) * 2] += o.z; sts[(slot + 2) * 2 + 1] += o.z * o.z;
                sts[(slot + 3) * 2] += o.w; sts[(slot + 3) * 2 + 1] += o.w * o.w;
              } else {
#pragma unroll
                for (int cc = 0; cc < TWR; cc++) {
                  const float o = acc[r][cc] + bias;
                  xrow[t + cc] = o;
                  sts[(slot + cc) * 2] += o; sts[(slot + cc) * 2 + 1] += o * o;
                }
              }
            }
          }
        }
      } else {
        float acc[TWR];
#pragma unroll
        for (int r = 0; r < TWR; r++) acc[r] = 0.f;
#pragma unroll
        for (int i = 0; i < 7; i++)
#pragma unroll
          for (int j = 0; j < 7; j++) {
            const float wv = wt[i * 7 + j];
#pragma unroll
            for (int r = 0; r < TWR; r++) {
              const int hh = th[r] - 3 + i, ww = tw_[r] - 3 + j;
              if (tv[r] && hh >= 0 && hh < H && ww >= 0 && ww < W)
                acc[r] = fmaf(__ldg(p.x + ((size_t)tb[r] * C + c) * HW + hh * W + ww), wv, acc[r]);
            }
          }
#pragma unroll
        for (int r = 0; r < TWR; r++) {
          if (tv[r]) {
            const float o = acc[r] + bias;
            xrow[tk[r]] = o;
            const int slot = r * 32 + lane;
            sts[slot * 2] += o; sts[slot * 2 + 1] += o * o;
          }
        }
      }
    }
    __syncthreads();
    const int nslots = (MODE == 1) ? rows_per_tile * rpt : 32 * TWR;
    for (int slot = tid; slot < nslots; slot += NT) {
      int t; bool valid;
      if (MODE == 1) {
        const int row = slot / rpt, col = slot - row * rpt;
        const int h = hbi * rows_per_tile + row, w = wbi * 32 * TWR + col;
        valid = (h < H) && (w < W);
        t = b * HW + h * W + w;
      } else {
        t = tok0 + slot; valid = t < M;
      }
      if (valid) {
        float S1 = 0.f, S2 = 0.f;
#pragma unroll
        for (int ws = 0; ws < 8; ws++) { S1 += st_s[(ws * P1_SLOTS + slot) * 2]; S2 += st_s[(ws * P1_SLOTS + slot) * 2 + 1]; }
        float* pp = p.part + ((size_t)cch * Mp + t) * 2;
        pp[0] = S1; pp[1] = S2;
      }
    }
    __syncthreads();
  }
}

__device__ void phase1(const Params& p, float* smem) {
  const int tid = threadIdx.x;
  for (int i = blockIdx.x * NT + tid; i < p.B * p.C4; i += gridDim.x * NT) p.sumsq[i] = 0.f;
  transpose_mat(p.w1, p.C4, p.C, p.w1t, smem);
  transpose_mat(p.w2, p.C, p.C4, p.w2t, smem);
  if (p.p1_mode == 1) {
    if (p.p1_twr == 4) phase1_conv<1, 4>(p, smem); else phase1_conv<1, 1>(p, smem);
  } else {
    if (p.p1_twr == 4) phase1_conv<0, 4>(p, smem);
    else if (p.p1_twr == 2) phase1_conv<0, 2>(p, smem);
    else phase1_conv<0, 1>(p, smem);
  }
}

// ---------------------------------------------------------------- GEMM phases
// A = act [K][Mp] (K-major), B = Wt [K][N]. 8 warps = WM x WN spatial warps x KS K-groups. Lane grid 8 x 4.
template <int BM, int BN, int WM, int WN, int KS, int BKc, int PHASE>
__device__ void gemm_phase(const Params& p, float* smem, float* sq_s, float* inv_s, float* red_s, int split) {
  constexpr int WS = WM * WN;
  constexpr int GT = WS * 32;
  constexpr int TM = BM / (WM * 8), TN = BN / (WN * 4);
  constexpr int AM4 = BM / 4, BN4 = BN / 4;
  constexpr int AQ = (BKc * AM4) / GT, BQ = (BKc * BN4) / GT;
  constexpr int STAGE = BKc * (BM + BN);
  constexpr int CS_LD = BM + 4;
  constexpr int NSTEP = NT / BM, NPT = BN / NSTEP;
  static_assert(WS * KS == 8, "8 warps");
  static_assert((BKc * AM4) % GT == 0 && (BKc * BN4) % GT == 0, "quad mapping");
  static_assert(GT % AM4 == 0, "token quad constant per thread");
  static_assert(KS * 2 * STAGE <= GEMM_SMEM_FLOATS && KS * BN * CS_LD <= GEMM_SMEM_FLOATS, "smem");
  static_assert(TM == 8 || TM == 4, "TM");
  static_assert(TN == 8 || TN == 6, "TN");

  const int tid = threadIdx.x, warp = tid >> 5, lane = tid & 31;
  const int kg = warp / WS, ws = warp - kg * WS, wm = ws / WN, wn = ws - wm * WN;
  const int lm = lane >> 2, ln = lane & 3;
  const int gt = ws * 32 + lane;
  const int M = p.M, Mp = p.Mp, HW = p.HW;
  const int N = (PHASE == 2) ? p.C4 : p.C;
  const int K = (PHASE == 2) ? p.C : p.C4;
  const float* __restrict__ Ag = (PHASE == 2) ? p.xn : p.hidden;
  const float* __restrict__ Bg = (PHASE == 2) ? p.w1t : p.w2t;
  const int mt = (M + BM - 1) / BM, nt = (N + BN - 1) / BN;
  const int ntiles = mt * nt * split, nk = K / BKc;
  float* As = smem + kg * 2 * STAGE;
  float* Bs = As + 2 * BKc * BM;
  const int row_m = wm * (BM / WM) + lm * TM;
  const int col_n = wn * (BN / WN) + ln * TN;
  const int a_m4 = (gt % AM4) * 4;
  const int a_kq = gt / AM4;
  constexpr int KSTEP_A = GT / AM4;

  for (int tile = blockIdx.x; tile < ntiles; tile += gridDim.x) {
    int rem = tile;
    const int ni = rem % nt; rem /= nt;
    const int mi = rem % mt; const int s = rem / mt;
    const int m0 = mi * BM, n0 = ni * BN;
    const int c0 = (int)(((long)nk * s) / split), c1 = (int)(((long)nk * (s + 1)) / split);
    const int b_first = m0 / HW;
    const int nimg = min(MAXIMG, (min(m0 + BM, M) - 1) / HW - b_first + 1);

    // per-thread token quad constants
    float tc0[4], tc1[4]; int timg[4]; bool tval[4];
#pragma unroll
    for (int e = 0; e < 4; e++) {
      const int t = m0 + a_m4 + e;
      tval[e] = t < M;
      const int tt = tval[e] ? t : 0;
      if (PHASE == 2) {
        float S1 = 0.f, S2 = 0.f;
        for (int ch = 0; ch < p.nch; ch++) {
          const float2 pp = __ldg(reinterpret_cast<const float2*>(p.part + ((size_t)ch * Mp + tt) * 2));
          S1 += pp.x; S2 += pp.y;
        }
        const float mean = S1 / (float)p.C;
        float var = S2 / (float)p.C - mean * mean; var = fmaxf(var, 0.f);
        tc0[e] = mean; tc1[e] = rsqrtf(var + p.ln_eps); timg[e] = 0;
      } else {
        timg[e] = tt / HW - b_first; tc0[e] = 0.f; tc1[e] = 0.f;
      }
    }
    if (PHASE == 3) {
      for (int img = 0; img < nimg; img++) {
        const float* sq = p.sumsq + (size_t)(b_first + img) * K;
        float sacc = 0.f;
        for (int k = tid; k < K; k += NT) sacc += sqrtf(sq[k]);
        sacc = warp_sum(sacc);
        if (lane == 0) red_s[warp] = sacc;
        __syncthreads();
        if (tid == 0) {
          float tot = 0.f;
#pragma unroll
          for (int w2 = 0; w2 < 8; w2++) tot += red_s[w2];
          inv_s[img] = 1.f / (tot / (float)K + p.eps);
        }
        __syncthreads();
      }
#pragma unroll
      for (int e = 0; e < 4; e++) tc1[e] = inv_s[timg[e] < MAXIMG ? timg[e] : 0];
    }

    float acc[TM][TN];
#pragma unroll
    for (int i = 0; i < TM; i++)
#pragma unroll
      for (int j = 0; j < TN; j++) acc[i][j] = 0.f;

    float4 ra[AQ], rb[BQ];
    auto load_chunk = [&](int kc) {
      const int k0 = kc * BKc;
#pragma unroll
      for (int q = 0; q < AQ; q++) {
        const int ka = k0 + q * KSTEP_A + a_kq;
        float4 v = __ldg(reinterpret_cast<const float4*>(Ag + (size_t)ka * Mp + m0 + a_m4));
        float vv[4] = {v.x, v.y, v.z, v.w};
        if (PHASE == 2) {
          const float lw = __ldg(p.lnw + ka), lb = __ldg(p.lnb + ka);
#pragma unroll
          for (int e = 0; e < 4; e++) vv[e] = tval[e] ? fmaf((vv[e] - tc0[e]) * tc1[e], lw, lb) : 0.f;
        } else {
          const float gwk = __ldg(p.gw + ka), gbk = __ldg(p.gb + ka);
#pragma unroll
          for (int e = 0; e < 4; e++) {
            const float sq = __ldg(p.sumsq + (size_t)(b_first + timg[e]) * K + ka);
            vv[e] = tval[e] ? fmaf(vv[e], fmaf(gwk * tc1[e], sqrtf(sq), 1.f), gbk) : 0.f;
          }
        }
        ra[q] = make_float4(vv[0], vv[1], vv[2], vv[3]);
      }
#pragma unroll
      for (int q = 0; q < BQ; q++) {
        const int id = q * GT + gt, kb = id / BN4, n4 = (id - kb * BN4) * 4;
        const int col = n0 + n4;
        rb[q] = (col < N) ? __ldg(reinterpret_cast<const float4*>(Bg + (size_t)(k0 + kb) * N + col))
                          : make_float4(0.f, 0.f, 0.f, 0.f);
      }
    };
    auto store_regs = [&](int buf) {
#pragma unroll
      for (int q = 0; q < AQ; q++)
        *reinterpret_cast<float4*>(As + (buf * BKc + q * KSTEP_A + a_kq) * BM + a_m4) = ra[q];
#pragma unroll
      for (int q = 0; q < BQ; q++) {
        const int id = q * GT + gt, kb = id / BN4, n4 = (id - kb * BN4) * 4;
        *reinterpret_cast<float4*>(Bs + (buf * BKc + kb) * BN + n4) = rb[q];
      }
    };
    auto gsync = [&]() {
      if (KS == 1) __syncthreads();
      else if (GT == 32) __syncwarp();
      else asm volatile("bar.sync %0, %1;" ::"r"(kg + 1), "r"(GT));
    };

    int kc = c0 + kg;
    if (kc < c1) { load_chunk(kc); store_regs(0); }
    gsync();
    for (int it = 0; kc < c1; it++, kc += KS) {
      const int buf = it & 1;
      const bool more = (kc + KS) < c1;
      if (more) load_chunk(kc + KS);
      const float* Ab = As + buf * BKc * BM + row_m;
      const float* Bb = Bs + buf * BKc * BN + col_n;
#pragma unroll 4
      for (int k = 0; k < BKc; k++) {
        float a[TM], b[TN];
        {
          const float4 a0 = *reinterpret_cast<const float4*>(Ab + k * BM);
          a[0] = a0.x; a[1] = a0.y; a[2] = a0.z; a[3] = a0.w;
          if (TM == 8) {
            const float4 a1 = *reinterpret_cast<const float4*>(Ab + k * BM + 4);
            a[4] = a1.x; a[5] = a1.y; a[6] = a1.z; a[7] = a1.w;
          }
        }
        if (TN == 8) {
          const float4 b0 = *reinterpret_cast<const float4*>(Bb + k * BN);
          const float4 b1 = *reinterpret_cast<const float4*>(Bb + k * BN + 4);
          b[0] = b0.x; b[1] = b0.y; b[2] = b0.z; b[3] = b0.w; b[4] = b1.x; b[5] = b1.y; b[6] = b1.z; b[7] = b1.w;
        } else {
          const float2 b0 = *reinterpret_cast<const float2*>(Bb + k * BN);
          const float2 b1 = *reinterpret_cast<const float2*>(Bb + k * BN + 2);
          const float2 b2 = *reinterpret_cast<const float2*>(Bb + k * BN + 4);
          b[0] = b0.x; b[1] = b0.y; b[2] = b1.x; b[3] = b1.y; b[4] = b2.x; b[5] = b2.y;
        }
#pragma unroll
        for (int i = 0; i < TM; i++)
#pragma unroll
          for (int j = 0; j < TN; j++) acc[i][j] = fmaf(a[i], b[j], acc[i][j]);
      }
      if (more) store_regs(buf ^ 1);
      gsync();
    }

    // ---- epilogue: partial tiles -> smem [KS][BN][CS_LD], then one token per thread
    __syncthreads();
    float* Cs = smem;
#pragma unroll
    for (int j = 0; j < TN; j++) {

#pragma unroll

          }

          if (il < MAXIMG) atomicAdd(&sq_s[il * BN + n_l], g2);
          else atomicAdd(p.sumsq + (size_t)img * N + col, g2);
        }
      }
      __syncthreads();
      for (int idx = tid; idx < nimg * BN; idx += NT) {

        atomicAdd(p.sumsq + (size_t)(b_first + im) * N + n0 + n, v);
      }
      __syncthreads();
    } else {

          else atomicAdd(p.out + idx, v);
        }
      }
      __syncthreads();
    }
  }

struct Pick { int cfg; int split; };
static Pick pick_cfg(long M, long N, long K, int grid, bool allow_split) {
  struct CfgD { int id, BM, BN, BK; double eff; };
  const CfgD cfgs[4] = {{0, 128, 128, 32, 0.80}, {1, 128, 96, 32, 0.78}, {2, 64, 64, 32, 0.60}, {3, 32, 32, 16, 0.40}};
  double best = 1e300; Pick r{0, 1};
  for (int ci = 0; ci < 4; ci++) {
    const CfgD& c = cfgs[ci];
    if (c.id == 1 && (N % 96) != 0) continue;
    const long tiles = ((M + c.BM - 1) / c.BM) * ((N + c.BN - 1) / c.BN);
    const long nk = K / c.BK;
    for (int S = 1; S <= (allow_split ? 16 : 1); S *= 2) {
      if (S > 1 && nk / S < 2) break;
      const long total = tiles * S;
      const long waves = (total + grid - 1) / grid;
      double cost = (double)c.BM * c.BN * ((double)K / S) / c.eff * (double)waves;
      if (S > 1) cost *= 1.08 + 0.03 * S;
      if (cost < best) { best = cost; r.cfg = c.id; r.split = S; }
    }
  }
  return r;
}

        }
      }
      __syncthreads();
      if (tid < nr) { p.u1[r0 + tid] = s_red[tid]; p.c1[r0 + tid] = s_red[32 + tid] + __ldg(p.b1 + r0 + tid); }
      fence_proxy_smem(); __syncthreads();
      if (tid == 0) red_release(p.ctr + CT_W1, 1u);
    } else if (item < p.o_g1) {
      // ---- conv unit: PG planes of one image -> d16 [c][b*HWP + t], LN partial sums per token
      const int jj = item - p.o_conv;
      const int PG = p.PG, NG = p.NG;
      const int b = jj / NG, g = jj - b * NG, c0 = g * PG;
      float* xs = (float*)ring; float* wsm = (float*)(ring + 32768); float* dbuf = (float*)(ring + 65536); float* red = (float*)(ring + 98304);
      if (tid == 0) {
        const uint32_t bx = PG * HW * 4, bw = PG * 49 * 4;
        mbar_expect_tx(bar_conv, bx + bw);
        bulk_g2s(smem_u32(xs), p.x + ((size_t)b * CC + c0) * HW, bx, bar_conv);
        bulk_g2s(smem_u32(wsm), p.dww + (size_t)c0 * 49, bw, bar_conv);
      }
      mbar_wait(bar_conv, convph); convph ^= 1;
      for (int wi = tid; wi < PG * WW; wi += NTH) {
        const int pl = wi / WW, y = wi - pl * WW;
        const float bias = __ldg(p.dwb + c0 + pl);
        float acc[WW];
#pragma unroll
        for (int xx = 0; xx < WW; xx++) acc[xx] = bias;
        const float* xp = xs + pl * HW; const float* wp = wsm + pl * 49;
#pragma unroll
        for (int i = 0; i < 7; i++) {
          const int yy = y + i - 3;
          if (yy >= 0 && yy < WW) {
            float v[WW];
#pragma unroll
            for (int xx = 0; xx < WW; xx++) v[xx] = xp[yy * WW + xx];
#pragma unroll
            for (int j = 0; j < 7; j++) {
              const float wv = wp[i * 7 + j];
#pragma unroll
              for (int xx = 0; xx < WW; xx++) { const int xi = xx + j - 3; if (xi >= 0 && xi < WW) acc[xx] = fmaf(v[xi], wv, acc[xx]); }
            }
          }
        }
#pragma unroll
        for (int xx = 0; xx < WW; xx++) dbuf[pl * HW + y * WW + xx] = acc[xx];
      }
      __syncthreads();
      constexpr int SUB = NTH / HW;
      if (tid < SUB * HW) {
        const int t = tid % HW, s = tid / HW;
        float S1 = 0.f, S2 = 0.f;
        for (int pl = s; pl < PG; pl += SUB) { const float v = dbuf[pl * HW + t]; S1 += v; S2 = fmaf(v, v, S2); }
        red[s * HW + t] = S1; red[(SUB + s) * HW + t] = S2;
      }
      for (int idx = tid; idx < PG * HW; idx += NTH) { const int pl = idx / HW, t = idx - pl * HW; p.d16[(size_t)(c0 + pl) * p.LDT + b * HWP + t] = __float2half_rn(dbuf[idx]); }
      __syncthreads();
      if (tid < HW) {
        float S1 = 0.f, S2 = 0.f;
#pragma unroll
        for (int s = 0; s < SUB; s++) { S1 += red[s * HW + tid]; S2 += red[(SUB + s) * HW + tid]; }
        float* pp = p.part + ((size_t)(b * NG + g) * 2) * HWP;
        pp[tid] = S1; pp[HWP + tid] = S2;
      }
      fence_proxy_smem(); __syncthreads();
      if (tid == 0) red_release(p.ctr + CT_CONV + b, 1u);
    } else if (item >= p.o_w2 && item < p.o_g2) {
      // ---- W2 unit: W2h = fp16(W2), b2' = W2.gb + b2, out = x + b2'
      constexpr int R4 = C4 / 4;
      const int u = item - p.o_w2;
      const int rpu = (CC + p.nW - 1) / p.nW, r0 = u * rpu, r1 = min(CC, r0 + rpu);
      const int nr = max(r1 - r0, 0);
      float* s_red = (float*)ring;
      if (tid < 32) s_red[tid] = 0.f;
      __syncthreads();
      const int nf = nr * R4;
      const float4* src4 = (const float4*)(p.w2 + (size_t)r0 * C4);
      constexpr int JB = 8;
      for (int base = 0; base < nf; base += NTH * JB) {
        float4 wv[JB], gv[JB];
#pragma unroll
        for (int j = 0; j < JB; j++) {
          const int f = base + tid + NTH * j;
          if (f < nf) { wv[j] = __ldg(src4 + f); gv[j] = __ldg((const float4*)p.gb + (f % R4)); }
        }
#pragma unroll
        for (int j = 0; j < JB; j++) {
          const int f = base + tid + NTH * j;
          if (f < nf) {
            const int rl = f / R4;
            float dot = wv[j].x * gv[j].x + wv[j].y * gv[j].y + wv[j].z * gv[j].z + wv[j].w * gv[j].w;
            uint2 st; st.x = h2u(__floats2half2_rn(wv[j].x, wv[j].y)); st.y = h2u(__floats2half2_rn(wv[j].z, wv[j].w));
            *(uint2*)(p.w2h + (size_t)r0 * C4 + 4 * (size_t)f) = st;
            dot = warp_sum(dot);
            if (lane == 0) atomicAdd(&s_red[rl], dot);
          }
        }
      }
      __syncthreads();
      if (tid < nr) s_red[tid] += __ldg(p.b2 + r0 + tid);
      __syncthreads();
      const int seg = nr * HW, tot = p.B * seg;
#pragma unroll 8
      for (int i = tid; i < tot; i += NTH) {
        const int bb = i / seg, rem = i - bb * seg;
        const size_t idx = ((size_t)bb * CC + r0) * HW + rem;
        p.out[idx] = __ldg(p.x + idx) + s_red[rem / HW];
      }
      fence_proxy_smem(); __syncthreads();
      if (tid == 0) red_release(p.ctr + CT_W2, 1u);
    } else if (item < p.o_w2) {
      // ---- GEMM1 tile: 128 hidden rows x IPT1 whole images
      const int jj = item - p.o_g1;
      const int IPT = p.IPT1, N1 = p.N1;
      const int grp = jj / MT1, mt = jj - grp * MT1;
      const int b0 = grp * IPT, k0 = mt * 128;
      if (tid == 0) {
        if (!w1ok) { wait_ctr(p.ctr + CT_W1, (unsigned)p.nW); w1ok = true; }
        for (int i = 0; i < IPT; i++) { const int b = b0 + i; if (!((seen_conv >> b) & 1u)) { wait_ctr(p.ctr + CT_CONV + b, (unsigned)p.NG); seen_conv |= 1u << b; } }
        fence_proxy_async();
      }
      __syncthreads();
      STAMP(1)
      if (warp == 0) {
        for (int kb = 0; kb < NKB1; kb++) {
          const uint32_t it = kit + kb, s = it % NST, ph = (it / NST) & 1u;
          mbar_wait(bar_empty + 8 * s, ph ^ 1u);
          if (elect_one()) {
            const uint32_t st = ringa + s * STAGE;
            mbar_expect_tx(bar_full + 8 * s, 16384u + (uint32_t)p.NCH1 * 8192u);
            tma2d(st, &p.tmW1, kb * 64, k0, bar_full + 8 * s);
            for (int c = 0; c < p.NCH1; c++) tma2d(st + 16384 + c * 8192, &p.tmD, b0 * HWP + c * 64, kb * 64, bar_full + 8 * s);
          }
          __syncwarp();
        }
      } else if (warp == 1) {
        const uint32_t idesc = make_idesc(N1);
        for (int kb = 0; kb < NKB1; kb++) {
          const uint32_t it = kit + kb, s = it % NST, ph = (it / NST) & 1u;
          mbar_wait(bar_full + 8 * s, ph);
          tc_fence_after();
          if (elect_one()) {
            const uint32_t st = ringa + s * STAGE;
#pragma unroll
            for (int kk = 0; kk < 4; kk++) mma_f16(tmem, sdesc(st + kk * 32, 16, 1024), sdesc(st + 16384 + kk * 2048, 8192, 1024), idesc, (kb | kk) ? 1u : 0u);
            mma_commit(bar_empty + 8 * s);
            if (kb == NKB1 - 1) mma_commit(bar_acc);
          }
          __syncwarp();
        }
      } else {
        // prestage: per-token LN statistics (one round of loads), per-row fold constants
        const int i0 = tid - 64;
        const int NT1 = ((N1 + 31) >> 5) << 5;
        for (int t = i0; t < NT1; t += NTH - 64) {
          const int img = b0 + (t >> LOGHWP), tl = t & (HWP - 1);
          float mu = 0.f, rs = 0.f;
          if (tl < HW) {
            const float* pp = p.part + (size_t)img * p.NG * 2 * HWP + tl;
            float S1 = 0.f, S2 = 0.f;
            constexpr int NGMAX = CC / 32;
            float a1[NGMAX], a2[NGMAX];
#pragma unroll
            for (int g = 0; g < NGMAX; g++) { a1[g] = 0.f; a2[g] = 0.f; if (g < p.NG) { a1[g] = __ldcg(pp + (size_t)g * 2 * HWP); a2[g] = __ldcg(pp + (size_t)g * 2 * HWP + HWP); } }
#pragma unroll
            for (int g = 0; g < NGMAX; g++) { S1 += a1[g]; S2 += a2[g]; }
            mu = S1 * (1.f / CC);
            const float var = fmaxf(S2 * (1.f / CC) - mu * mu, 0.f);
            rs = rsqrtf(var + p.ln_eps);
          }
          s_mu[t] = mu; s_rs[t] = rs;
        }
        for (int r = i0; r < 128; r += NTH - 64) { s_u[r] = __ldcg(p.u1 + k0 + r); s_c1[r] = __ldcg(p.c1 + k0 + r); s_gw[r] = __ldg(p.gw + k0 + r); }
        for (int r = i0; r < 128 * IPT; r += NTH - 64) s_ss[r] = 0.f;
      }
      kit += NKB1;
      __syncthreads();
      mbar_wait(bar_acc, accph); accph ^= 1u;
      tc_fence_after();
      STAMP(4)
      // pass 1: TMEM -> LN fold -> exact GELU -> fp16 staging + per-(image,row) sum of squares
      const int q = warp & 3, hf = warp >> 2;
      const int r = q * 32 + lane;
      const float uu = s_u[r], cb = s_c1[r];
      const int NC32 = (N1 + 31) >> 5;
      const int RS = NC32 * 64 + 16;
      uint8_t* stg = ring;
      for (int c = hf; c < NC32; c += 2) {
        uint32_t v[32];
        tmem_ld32w(tmem + ((uint32_t)(q * 32) << 16) + (uint32_t)(c * 32), v);
        const int t0 = c * 32;
        float ss = 0.f;
        uint32_t pk[16];
#pragma unroll
        for (int e = 0; e < 32; e += 2) {
          float hv[2];
#pragma unroll
          for (int f = 0; f < 2; f++) {
            const int t = t0 + e + f;
            const float a = __uint_as_float(v[e + f]);
            const float z = fmaf(s_rs[t], a - s_mu[t] * uu, cb);
            float h = gelu_erf(z);
            h = ((t & (HWP - 1)) < HW) ? h : 0.f;
            ss = fmaf(h, h, ss);
            hv[f] = h;
          }
          pk[e >> 1] = h2u(__floats2half2_rn(hv[0], hv[1]));
        }
        atomicAdd(&s_ss[(t0 >> LOGHWP) * 128 + r], ss);
        uint4* dst = (uint4*)(stg + r * RS + c * 64);
        dst[0] = make_uint4(pk[0], pk[1], pk[2], pk[3]); dst[1] = make_uint4(pk[4], pk[5], pk[6], pk[7]);
        dst[2] = make_uint4(pk[8], pk[9], pk[10], pk[11]); dst[3] = make_uint4(pk[12], pk[13], pk[14], pk[15]);
      }
      tc_fence_before(); __syncthreads();
      STAMP(5)
      // pass 2: gf per (image, row), q = gw * gf, per-image sum of gf
      for (int id = tid; id < 128 * IPT; id += NTH) {
        const int i = id >> 7, rr = id & 127;
        const float gf = sqrtf(s_ss[id]);
        s_q[id] = s_gw[rr] * gf;
        const float sg = warp_sum(gf);
        if (lane == 0) atomicAdd(p.gsum + b0 + i, sg);
      }
      __syncthreads();
      // pass 3: coalesced stores of h16 and g16 = fp16(gw*gf*h)
      const int total = 128 * IPT * NCK;
      for (int id = tid; id < total; id += NTH) {
        const int j8 = id % NCK; const int rest = id / NCK; const int i = rest % IPT, rr = rest / IPT;
        const uint4 hv4 = *(const uint4*)(stg + rr * RS + (i * HWP + j8 * 8) * 2);
        const size_t off = (size_t)(k0 + rr) * p.LDT + (size_t)(b0 + i) * HWP + j8 * 8;
        *(uint4*)(p.h16 + off) = hv4;
        const float qv = s_q[i * 128 + rr];
        uint4 gv4; const uint32_t* hs = (const uint32_t*)&hv4; uint32_t* gs = (uint32_t*)&gv4;
#pragma unroll
        for (int e = 0; e < 4; e++) { const float2 f = __half22float2(u2h(hs[e])); gs[e] = h2u(__floats2half2_rn(f.x * qv, f.y * qv)); }
        *(uint4*)(p.g16 + off) = gv4;
      }
      fence_proxy_smem(); __syncthreads();
      if (tid == 0) { for (int i = 0; i < IPT; i++) red_release(p.ctr + CT_G1 + b0 + i, 1u); }
    } else {
      // ---- GEMM2 tile: 128 output channels x tokens x split-K slice; accumulators for h and g
      const int jj = item - p.o_g2;
      const int per = p.NTS * MT2 * p.KS2;
      const int grp = jj / per;
      int rem = jj - grp * per;
      const int ks = rem % p.KS2; rem /= p.KS2;
      const int mt2 = rem % MT2, ts = rem / MT2;
      const int b0 = grp * IPT2, tok0 = ts * 128;
      const int nkb = (C4 / 64) / p.KS2, kb0 = ks * nkb;
      const int N2 = (WW == 7) ? 128 : (ts == 0 ? 128 : 80);
      if (tid == 0) {
        if (!w2ok) { wait_ctr(p.ctr + CT_W2, (unsigned)p.nW); w2ok = true; }
        for (int i = 0; i < IPT2; i++) { const int b = b0 + i; if (!((seen_g1 >> b) & 1u)) { wait_ctr(p.ctr + CT_G1 + b, (unsigned)MT1); seen_g1 |= 1u << b; } }
        fence_proxy_async();
      }
      __syncthreads();
      STAMP(1)
      if (warp == 0) {
        for (int kb = 0; kb < nkb; kb++) {
          const uint32_t it = kit + kb, s = it % NST, ph = (it / NST) & 1u;
          mbar_wait(bar_empty + 8 * s, ph ^ 1u);
          if (elect_one()) {
            const uint32_t st = ringa + s * STAGE;
            const int kr = (kb0 + kb) * 64;
            mbar_expect_tx(bar_full + 8 * s, 49152u);
            tma2d(st, &p.tmW2, kr, mt2 * 128, bar_full + 8 * s);
#pragma unroll
            for (int c = 0; c < 2; c++) {
              tma2d(st + 16384 + c * 8192, &p.tmH, b0 * HWP + tok0 + c * 64, kr, bar_full + 8 * s);
              tma2d(st + 32768 + c * 8192, &p.tmG, b0 * HWP + tok0 + c * 64, kr, bar_full + 8 * s);
            }
          }
          __syncwarp();
        }
      } else if (warp == 1) {
        const uint32_t idesc = make_idesc(N2);
        for (int kb = 0; kb < nkb; kb++) {
          const uint32_t it = kit + kb, s = it % NST, ph = (it / NST) & 1u;
          mbar_wait(bar_full + 8 * s, ph);
          tc_fence_after();
          if (elect_one()) {
            const uint32_t st = ringa + s * STAGE;
#pragma unroll
            for (int kk = 0; kk < 4; kk++) {
              const uint64_t ad = sdesc(st + kk * 32, 16, 1024);
              const uint32_t accf = (kb | kk) ? 1u : 0u;
              mma_f16(tmem, ad, sdesc(st + 16384 + kk * 2048, 8192, 1024), idesc, accf);
              mma_f16(tmem + 256, ad, sdesc(st + 32768 + kk * 2048, 8192, 1024), idesc, accf);
            }
            mma_commit(bar_empty + 8 * s);
            if (kb == nkb - 1) mma_commit(bar_acc);
          }
          __syncwarp();
        }
      } else {
        if (tid >= 64 && tid < 64 + IPT2) s_inv[tid - 64] = 1.f / (__ldcg(p.gsum + b0 + tid - 64) * (1.f / C4) + p.eps);
      }
      kit += nkb;
      __syncthreads();
      mbar_wait(bar_acc, accph); accph ^= 1u;
      tc_fence_after();
      STAMP(4)
      // out += D_h + inv_b * D_g, transposed through a per-warp smem tile so each red.add row segment is contiguous
      const int q = warp & 3, hf = warp >> 2;
      float* buf = (float*)ring + warp * 32 * 17;
      const int NC16 = N2 >> 4;
      for (int c = hf; c < NC16; c += 2) {
        uint32_t v[32];
        const uint32_t ta = tmem + ((uint32_t)(q * 32) << 16) + (uint32_t)(c * 16);
        tmem_ld16x2w(ta, ta + 256, v);
        const int gt = tok0 + c * 16;
        const int ii = gt >> LOGHWP, tl0 = gt & (HWP - 1);
        const float inv = s_inv[ii];
#pragma unroll
        for (int e = 0; e < 16; e++) buf[lane * 17 + e] = __uint_as_float(v[e]) + inv * __uint_as_float(v[16 + e]);
        __syncwarp();
        const int col = lane & 15, tl = tl0 + col;
        if (tl < HW) {
          float* ob = p.out + ((size_t)(b0 + ii) * CC + mt2 * 128 + q * 32) * HW + tl;
#pragma unroll 4
          for (int i2 = 0; i2 < 16; i2++) { const int rr = 2 * i2 + (lane >> 4); atomicAdd(ob + (size_t)rr * HW, buf[rr * 17 + col]); }
        }
        __syncwarp();
      }
      tc_fence_before(); fence_proxy_smem(); __syncthreads();
    }
    STAMP(2)
    if (tid == 0) s_misc[1] = (uint32_t)nxt;
    __syncthreads();
    item = (int)s_misc[1];
  }
  tc_fence_before(); __syncthreads();
  if (warp == 0) { tc_fence_after(); asm volatile("tcgen05.dealloc.cta_group::1.sync.aligned.b32 %0, 512;" ::"r"(tmem) : "memory"); }
  if (tid == 0) {
    __threadfence();
    const unsigned old = atomicAdd(p.ctr + CT_DEP, 1u);
    if (old == gridDim.x - 1) {  // every other CTA has departed: reset the counters for the next call
      __threadfence();
      p.ctr[CT_HEAD] = 0u; p.ctr[CT_W1] = 0u; p.ctr[CT_W2] = 0u;
      for (int b = 0; b < p.B; b++) { p.ctr[CT_CONV + b] = 0u; p.ctr[CT_G1 + b] = 0u; }
      __threadfence();
      p.ctr[CT_DEP] = 0u;
    }
  }
}

// ---------------------------------------------------------------- host
struct HArgs { const float *x, *dww, *dwb, *lnw, *lnb, *w1, *b1, *gw, *gb, *w2, *b2; float* out; int B, C, H, W; float eps, ln_eps; void* stamps; };

static inline size_t al256(size_t v) { return (v + 255) & ~(size_t)255; }
struct WsLayout { size_t w1h, w2h, u1, c1, gsum, part, d16, h16, g16, total; };
static WsLayout ws_layout(int B, int C, int W) {
  const size_t HWP = (W == 7) ? 64 : 256, LDT = (size_t)B * HWP, C4 = 4 * (size_t)C;
  WsLayout L; size_t o = 0;
  L.w1h = o; o = al256(o + C4 * C * 2); L.w2h = o; o = al256(o + C4 * C * 2);
  L.u1 = o; o = al256(o + C4 * 4); L.c1 = o; o = al256(o + C4 * 4); L.gsum = o; o = al256(o + 64 * 4);
  L.part = o; o = al256(o + (size_t)B * (C / 32) * 2 * HWP * 4);
  L.d16 = o; o = al256(o + (size_t)C * LDT * 2); L.h16 = o; o = al256(o + C4 * LDT * 2); L.g16 = o; o = al256(o + C4 * LDT * 2);
  L.total = o; return L;
}
extern "C" size_t h_ws_bytes(int B, int C, int H, int W) { (void)H; return ws_layout(B, C, W).total; }

static bool enc2d(CUtensorMap* m, const void* ptr, uint64_t inner, uint64_t outer, uint32_t box_in, uint32_t box_out) {
  cuuint64_t dims[2] = {(cuuint64_t)inner, (cuuint64_t)outer}; cuuint64_t strides[1] = {(cuuint64_t)(inner * 2)};
  cuuint32_t box[2] = {box_in, box_out}; cuuint32_t es[2] = {1, 1};
  return cuTensorMapEncodeTiled(m, CU_TENSOR_MAP_DATA_TYPE_FLOAT16, 2, const_cast<void*>(ptr), dims, strides, box, es, CU_TENSOR_MAP_INTERLEAVE_NONE, CU_TENSOR_MAP_SWIZZLE_128B, CU_TENSOR_MAP_L2_PROMOTION_L2_256B, CU_TENSOR_MAP_FLOAT_OOB_FILL_NONE) == CUDA_SUCCESS;
}

extern "C" const char* h_launch(const HArgs* a, void* ws, void* stream) {
  using namespace hk;
  static int sms = 0; static unsigned* ctr = nullptr; cudaError_t e;
  if (sms == 0) {  // shape-independent: attributes set once to the maximum any shape needs
    int dev = 0; cudaGetDevice(&dev); cudaDeviceGetAttribute(&sms, cudaDevAttrMultiProcessorCount, dev);
    e = cudaFuncSetAttribute(h_kernel<768, 7>, cudaFuncAttributeMaxDynamicSharedMemorySize, SMEM_TOTAL); if (e != cudaSuccess) return cudaGetErrorString(e);
    e = cudaFuncSetAttribute(h_kernel<384, 14>, cudaFuncAttributeMaxDynamicSharedMemorySize, SMEM_TOTAL); if (e != cudaSuccess) return cudaGetErrorString(e);
    void* cp = nullptr; e = cudaGetSymbolAddress(&cp, g_hctr); if (e != cudaSuccess) return cudaGetErrorString(e); ctr = (unsigned*)cp;
  }
  const int B = a->B, C = a->C, W = a->W, C4 = 4 * C;
  const int HWP = (W == 7) ? 64 : 256, LDT = B * HWP;
  const WsLayout L = ws_layout(B, C, W);
  char* w = (char*)ws;
  HP hp; memset(&hp, 0, sizeof(hp));
  hp.x = a->x; hp.dww = a->dww; hp.dwb = a->dwb; hp.lnw = a->lnw; hp.lnb = a->lnb; hp.w1 = a->w1; hp.b1 = a->b1;
  hp.gw = a->gw; hp.gb = a->gb; hp.w2 = a->w2; hp.b2 = a->b2; hp.out = a->out;

// r0 anchor (fp32 SIMT cooperative kernel): every row the lineage-H kernel does not take
struct Params {
  const float* x; const float* dww; const float* dwb; const float* lnw; const float* lnb;
  const float* w1; const float* b1; const float* gw; const float* gb; const float* w2; const float* b2;

};
extern "C" const char* convnext_launch(Params p, void* stream);

// lineage H: image-counter dataflow tcgen05 kernel for B >= 8 at C=768 7x7 and C=384 14x14
struct HArgs { const float *x, *dww, *dwb, *lnw, *lnb, *w1, *b1, *gw, *gb, *w2, *b2; float* out; int B, C, H, W; float eps, ln_eps; void* stamps; };
extern "C" size_t h_ws_bytes(int B, int C, int H, int W);
extern "C" const char* h_launch(const HArgs* a, void* ws, void* stream);

static inline size_t align256(size_t v) { return (v + 255) & ~(size_t)255; }

void run(torch::Tensor x, torch::Tensor dwconv_weight, torch::Tensor dwconv_bias, torch::Tensor layernorm_weight,
         torch::Tensor layernorm_bias, torch::Tensor pwconv1_weight, torch::Tensor pwconv1_bias, torch::Tensor grn_weight,
         torch::Tensor grn_bias, torch::Tensor pwconv2_weight, torch::Tensor pwconv2_bias, double eps, double layer_norm_eps,
         torch::Tensor output) {
  TORCH_CHECK(x.is_cuda() && x.dtype() == torch::kFloat32 && x.is_contiguous(), "x must be contiguous fp32 cuda");
  TORCH_CHECK(output.is_contiguous() && output.dtype() == torch::kFloat32, "output must be contiguous fp32");
  TORCH_CHECK(x.dim() == 4, "x must be NCHW");
  const c10::cuda::OptionalCUDAGuard guard(x.device());
  const int B = (int)x.size(0), C = (int)x.size(1), H = (int)x.size(2), W = (int)x.size(3);
  cudaStream_t stream = at::cuda::getCurrentCUDAStream(x.device().index()).stream();

  const bool hpath = B >= 8 && B <= 32 && (B % 2) == 0 &&
                     ((C == 768 && H == 7 && W == 7) || (C == 384 && H == 14 && W == 14));
  if (hpath) {
    HArgs a;
    a.x = x.data_ptr<float>(); a.dww = dwconv_weight.data_ptr<float>(); a.dwb = dwconv_bias.data_ptr<float>();
    a.lnw = layernorm_weight.data_ptr<float>(); a.lnb = layernorm_bias.data_ptr<float>();
    a.w1 = pwconv1_weight.data_ptr<float>(); a.b1 = pwconv1_bias.data_ptr<float>();
    a.gw = grn_weight.data_ptr<float>(); a.gb = grn_bias.data_ptr<float>();
    a.w2 = pwconv2_weight.data_ptr<float>(); a.b2 = pwconv2_bias.data_ptr<float>();
    a.out = output.data_ptr<float>();
    a.B = B; a.C = C; a.H = H; a.W = W; a.eps = (float)eps; a.ln_eps = (float)layer_norm_eps; a.stamps = nullptr;
    torch::Tensor ws = torch::empty({(long)h_ws_bytes(B, C, H, W)}, x.options().dtype(torch::kUInt8));
    const char* err = h_launch(&a, ws.data_ptr(), (void*)stream);
    if (err != nullptr) throw std::runtime_error(std::string("h_launch failed: ") + err);
    return;
  }

  TORCH_CHECK(C % 32 == 0, "dim must be a multiple of 32");
  TORCH_CHECK(H * W >= 32, "H*W must be >= 32");
  const int HW = H * W, M = B * HW, C4 = 4 * C, nch = C / 32;
  const int Mp = ((M + 127) / 128) * 128;
  const size_t off_xn = 0;
  const size_t off_part = align256(off_xn + (size_t)C * Mp * 4);
  const size_t off_hidden = align256(off_part + (size_t)nch * Mp * 2 * 4);
  const size_t off_sumsq = align256(off_hidden + (size_t)C4 * Mp * 4);
  const size_t off_w1t = align256(off_sumsq + (size_t)B * C4 * 4);
  const size_t off_w2t = align256(off_w1t + (size_t)C * C4 * 4);
  const size_t total = align256(off_w2t + (size_t)C4 * C * 4);
  torch::Tensor ws = torch::empty({(long)total}, x.options().dtype(torch::kUInt8));
  char* wp = (char*)ws.data_ptr();

  Params p;
  p.x = x.data_ptr<float>(); p.dww = dwconv_weight.data_ptr<float>(); p.dwb = dwconv_bias.data_ptr<float>();
  p.lnw = layernorm_weight.data_ptr<float>(); p.lnb = layernorm_bias.data_ptr<float>();
  p.w1 = pwconv1_weight.data_ptr<float>(); p.b1 = pwconv1_bias.data_ptr<float>();
  p.gw = grn_weight.data_ptr<float>(); p.gb = grn_bias.data_ptr<float>();
  p.w2 = pwconv2_weight.data_ptr<float>(); p.b2 = pwconv2_bias.data_ptr<float>();
  p.out = output.data_ptr<float>();
  p.xn = (float*)(wp + off_xn); p.part = (float*)(wp + off_part); p.hidden = (float*)(wp + off_hidden);
  p.sumsq = (float*)(wp + off_sumsq); p.w1t = (float*)(wp + off_w1t); p.w2t = (float*)(wp + off_w2t);
  p.B = B; p.C = C; p.H = H; p.W = W; p.HW = HW; p.M = M; p.Mp = Mp; p.C4 = C4; p.nch = nch;
  p.eps = (float)eps; p.ln_eps = (float)layer_norm_eps;
  p.p1_mode = 0; p.p1_twr = 1; p.p1_tiles = 0; p.p1_lpr = 0; p.p1_rg = 0; p.p1_hb = 0; p.p1_wb = 0;
  p.cfg2 = 0; p.cfg3 = 0; p.split3 = 1;
  const char* err = convnext_launch(p, (void*)stream);
  if (err != nullptr) throw std::runtime_error(std::string("convnext_launch failed: ") + err);
}
