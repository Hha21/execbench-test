def _ok(res, what):
    """cuda-python calls return (err,) or (err, value); raise on error, else return the value."""

def compile_cubin(src, opts=()):
    """CUDA C++ source -> cubin bytes for sm_100a with NVRTC. Raises with the compile log on error."""

### Rationale
I tested the bulk-pipeline idea and it does not pay off. Plain 256-bit LDG/STG beats every bulk-copy configuration I tried, at every size. The idea assumed that the TMA engine moves bytes at least as fast as LDG/STG while costing fewer SM instructions. Two probes say it does not:
- **Bulk stores are the bottleneck.** Writing 403 MB took 69 µs with bulk stores against 54.6 µs with STG.256, about 27% slower. A pure copy using bulk loads and bulk stores was 126–131 µs at 805 MB against 116 µs for a plain LDG/STG copy.
- **Bulk loads are close but still slower,** about 3% at 805 MB, and they add about 0.3–0.4 µs of fixed cost at 12.6 MB.

The best version keeps bulk loads (a 4-stage ring of 8 KB tiles, two CTAs per SM above 2048 tokens) and writes outputs with plain STG.256 stores marked evict_last. It passes all 16 workloads. On the rented B200 it is slower than the current best r6 in every band: S +18.8%, M +7.1%, L +1.6%. The emulator predicts a portal score of 0.58, against 0.61 for the current best. Do not submit it for score; it is returned as the record of a negative result.

}
__device__ __forceinline__ void bulk_ld(uint32_t dst, const void* src, uint32_t bytes, uint32_t b) {
    asm volatile("mbarrier.arrive.expect_tx.shared::cta.b64 _, [%0], %1;" :: "r"(b), "r"(bytes) : "memory");
    asm volatile("cp.async.bulk.shared::cluster.global.mbarrier::complete_tx::bytes [%0], [%1], %2, [%3];"
                 :: "r"(dst), "l"(src), "r"(bytes), "r"(b) : "memory");
}

__global__ void __launch_bounds__(THREADS)
qk_rms_bulk(const float* __restrict__ q, const float* __restrict__ k,
            const float* __restrict__ wq, const float* __restrict__ wk,
            float* __restrict__ qo, float* __restrict__ ko, int n_tiles, float eps) {
    __shared__ __align__(128) float sm[STAGES * TILE_FLOATS];   // 32 KB
    __shared__ __align__(8) uint64_t bar[STAGES];
    const int G = gridDim.x, b = blockIdx.x, tot = 2 * n_tiles;
    const int my_n = b < tot ? (tot - 1 - b) / G + 1 : 0;
    const int tid = threadIdx.x;
    auto tsrc = [&](int j) -> const float* {
        const int t = b + j * G;
        return t < n_tiles ? q + (size_t)t * TILE_FLOATS : k + (size_t)(t - n_tiles) * TILE_FLOATS;
    };
    if (tid == 0) {
        for (int s = 0; s < STAGES; ++s) mb_init(saddr(&bar[s]));
        asm volatile("fence.mbarrier_init.release.cluster;" ::: "memory");
        for (int j = 0; j < STAGES && j < my_n; ++j) bulk_ld(saddr(sm + j * TILE_FLOATS), tsrc(j), TILE_BYTES, saddr(&bar[j]));
    }
    __syncthreads();
    const int r = tid >> 4, c = (tid & 15) * 8;
    const uint64_t pol = pol_evict_last();
    for (int j = 0; j < my_n; ++j) {
        const int s = j % STAGES;
        const int t = b + j * G;
        const bool is_k = t >= n_tiles;
        const int lt = is_k ? t - n_tiles : t;
        const float* w = is_k ? wk : wq;
        float* y = (is_k ? ko : qo) + (size_t)lt * TILE_FLOATS;
        const int head = ((lt * ROWS) % H) + r;
        float wv[8];
        ld8_nc(w + (size_t)head * D + c, wv);
        mb_wait(saddr(&bar[s]), (j / STAGES) & 1);
        const float* p = sm + s * TILE_FLOATS + r * D + c;
        const float4 a = *reinterpret_cast<const float4*>(p);
        const float4 a2 = *reinterpret_cast<const float4*>(p + 4);
        const float xv[8] = {a.x, a.y, a.z, a.w, a2.x, a2.y, a2.z, a2.w};
        __syncthreads();                       // every thread has read stage s
        if (tid == 0 && j + STAGES < my_n) {
            asm volatile("fence.proxy.async.shared::cta;" ::: "memory");
            bulk_ld(saddr(sm + s * TILE_FLOATS), tsrc(j + STAGES), TILE_BYTES, saddr(&bar[s]));
        }
        float ss = 0.0f;
#pragma unroll
        for (int i = 0; i < 8; ++i) ss = fmaf(xv[i], xv[i], ss);
        ss += __shfl_xor_sync(0xffffffffu, ss, 8);
        ss += __shfl_xor_sync(0xffffffffu, ss, 4);
        ss += __shfl_xor_sync(0xffffffffu, ss, 2);
        ss += __shfl_xor_sync(0xffffffffu, ss, 1);
        const float inv = rsqrtf(ss * (1.0f / (float)D) + eps);
        float yv[8];
#pragma unroll
        for (int i = 0; i < 8; ++i) yv[i] = (xv[i] * inv) * wv[i];
        st8_h(y + r * D + c, yv, pol);
    }
}
}  // namespace

extern "C" int r10_launch(const float* q, const float* k, const float* wq, const float* wk, float* qo, float* ko,
                          long long n_rows, float eps, cudaStream_t stream) {
    static int n_sm = 0;
    if (n_sm == 0) {
        int dev = 0;
        cudaGetDevice(&dev);
        cudaDeviceGetAttribute(&n_sm, cudaDevAttrMultiProcessorCount, dev);
        if (n_sm <= 0) n_sm = 148;
    }
    const long long n_tiles = n_rows / ROWS;
    const long long tokens = n_rows / H;
    const int cps = tokens <= 2048 ? 4 : 2;     // shape-only choice of CTAs per SM
    long long G = (long long)n_sm * cps;
    if (G > 2 * n_tiles) G = 2 * n_tiles;
    qk_rms_bulk<<<(unsigned)G, THREADS, 0, stream>>>(q, k, wq, wk, qo, ko, (int)n_tiles, eps);
    return (int)cudaGetLastError();
}

static void check_tensor(const torch::Tensor& t, const char* name) {
    TORCH_CHECK(t.is_cuda() && t.scalar_type() == torch::kFloat32 && t.is_contiguous(), name,
                " must be a contiguous fp32 CUDA tensor");
    TORCH_CHECK((reinterpret_cast<uintptr_t>(t.data_ptr()) & 31) == 0, name, " must be 32-byte aligned");
}

void run(torch::Tensor query, torch::Tensor key, torch::Tensor weight_q, torch::Tensor weight_k,
         double eps, torch::Tensor query_norm, torch::Tensor key_norm) {
    check_tensor(query, "query"); check_tensor(key, "key");
    check_tensor(weight_q, "weight_q"); check_tensor(weight_k, "weight_k");
    check_tensor(query_norm, "query_norm"); check_tensor(key_norm, "key_norm");
    TORCH_CHECK(weight_q.numel() == 48 * 128 && weight_k.numel() == 48 * 128, "weights must be [48,128]");
    TORCH_CHECK(query.numel() % (48 * 128) == 0 && key.numel() == query.numel() &&
                query_norm.numel() == query.numel() && key_norm.numel() == query.numel(), "shape mismatch");
    const int64_t n_rows = query.numel() / 128;
    if (n_rows == 0) return;
    TORCH_CHECK(n_rows / 16 * 2 < (int64_t)0x7fffffff, "too large");
    const c10::cuda::CUDAGuard guard(query.device());
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    const int err = r10_launch(query.data_ptr<float>(), key.data_ptr<float>(), weight_q.data_ptr<float>(),
                               weight_k.data_ptr<float>(), query_norm.data_ptr<float>(), key_norm.data_ptr<float>(),
                               (long long)n_rows, (float)eps, stream);
    TORCH_CHECK(err == 0, "launch failed: ", cudaGetErrorString((cudaError_t)err));
}

    m.def("run", &run, "Per-head RMSNorm of Q and K, 1-D TMA load ring");
