### Rationale

Fresh-eyes measurements on the rented B200 say the textbook kernel is already at the practical floor: one 512-B row per 16 threads with 256-bit loads/stores streams at 6.8 TB/s, and the only lever that moved anything was L2 cache policy. The harness's write-flush leaves L2 full of dirty lines, so every line we allocate forces a write-back; marking loads `L2::evict_first` and output stores `L2::evict_last` recovers 3–24 % depending on size. Looping rows per CTA, more bytes in flight, other CTA sizes, L1 hints, streaming stores, per-shape CTA balancing, Triton (identical speed) and torch.compile (5× slower, failed to recompile) were all dead ends or neutral. The result passes all 16 workloads and is level with the current best (predicted 0.611 ± 0.006 vs 0.6125), which evidently already uses equivalent hints. I submit the simplest tested version.

```cuda file=kernel.cu
// Per-head RMSNorm of Q and K (head_dim = 128, fp32).
// One 512-byte row per 16 threads: each thread loads 8 floats with one 256-bit load,
// reduces the sum of squares over the half-warp with 4 shuffles, scales and stores 256 bits.
// Loads carry an L2 evict_first hint (streamed once); output stores carry L2 evict_last
// (keeps our output lines resident instead of forcing them out behind the inputs).
#include <cuda_runtime.h>

struct __align__(32) f8 { float4 a, b; };

__device__ __forceinline__ f8 ld_stream(const f8* p) {
    f8 v;
    asm volatile("ld.global.L2::evict_first.v8.f32 {%0,%1,%2,%3,%4,%5,%6,%7}, [%8];"
                 : "=f"(v.a.x), "=f"(v.a.y), "=f"(v.a.z), "=f"(v.a.w),
                   "=f"(v.b.x), "=f"(v.b.y), "=f"(v.b.z), "=f"(v.b.w)
                 : "l"(p));
    return v;
}

__device__ __forceinline__ void st_keep(f8* p, const f8& v) {
    asm volatile("st.global.L2::evict_last.v8.f32 [%0], {%1,%2,%3,%4,%5,%6,%7,%8};"
                 :: "l"(p), "f"(v.a.x), "f"(v.a.y), "f"(v.a.z), "f"(v.a.w),
                    "f"(v.b.x), "f"(v.b.y), "f"(v.b.z), "f"(v.b.w)
                 : "memory");
}

__global__ void __launch_bounds__(256)
rmsnorm_qk_kernel(const float* __restrict__ q, const float* __restrict__ k,
                  const float* __restrict__ wq, const float* __restrict__ wk,
                  float* __restrict__ oq, float* __restrict__ ok,
                  float eps, long long nrows)
{
    long long gt  = (long long)blockIdx.x * blockDim.x + threadIdx.x;
    long long row = gt >> 4;              // 16 threads per row
    const int lane = threadIdx.x & 15;
    const float* x; const float* w; float* o;
    if (row < nrows) { x = q; w = wq; o = oq; }
    else { row -= nrows; if (row >= nrows) return; x = k; w = wk; o = ok; }
    const int head = (int)(row % 48);

    f8 v  = ld_stream(reinterpret_cast<const f8*>(x + row * 128) + lane);
    f8 ww = *(reinterpret_cast<const f8*>(w + head * 128) + lane);

    float s = v.a.x*v.a.x + v.a.y*v.a.y + v.a.z*v.a.z + v.a.w*v.a.w
            + v.b.x*v.b.x + v.b.y*v.b.y + v.b.z*v.b.z + v.b.w*v.b.w;
    s += __shfl_xor_sync(0xffffffffu, s, 1);
    s += __shfl_xor_sync(0xffffffffu, s, 2);
    s += __shfl_xor_sync(0xffffffffu, s, 4);
    s += __shfl_xor_sync(0xffffffffu, s, 8);
    const float r = rsqrtf(s * (1.0f / 128.0f) + eps);

    f8 out;
    out.a.x = v.a.x * r * ww.a.x; out.a.y = v.a.y * r * ww.a.y;
    out.a.z = v.a.z * r * ww.a.z; out.a.w = v.a.w * r * ww.a.w;
    out.b.x = v.b.x * r * ww.b.x; out.b.y = v.b.y * r * ww.b.y;
    out.b.z = v.b.z * r * ww.b.z; out.b.w = v.b.w * r * ww.b.w;
    st_keep(reinterpret_cast<f8*>(o + row * 128) + lane, out);
}

extern "C" cudaError_t launch_rmsnorm_qk(const float* q, const float* k,
                                         const float* wq, const float* wk,
                                         float* oq, float* ok, float eps,
                                         long long nrows, cudaStream_t stream)
{
    if (nrows <= 0) return cudaSuccess;
    long long total_threads = 2LL * nrows * 16;
    unsigned grid = (unsigned)((total_threads + 255) / 256);
    rmsnorm_qk_kernel<<<grid, 256, 0, stream>>>(q, k, wq, wk, oq, ok, eps, nrows);
    return cudaGetLastError();
}
```

         double eps, torch::Tensor query_norm, torch::Tensor key_norm)

                                        weight_q.data_ptr<float>(), weight_k.data_ptr<float>(),
                                        query_norm.data_ptr<float>(), key_norm.data_ptr<float>(),

}
