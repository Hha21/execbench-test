What I measured first, as the harness times a call:

- **Too many CTAs cost a lot:** an empty kernel with 7,032 CTAs costs 4.5 µs. Even so, a single launch of small, low-register CTAs beat every persistent or fat-CTA layout I tried.

```cuda file=kernel.cu
#include <cuda_runtime.h>

// Per-head RMSNorm for Q and K. One row (128 fp32) per half-warp: 16 threads x 8 floats (256-bit access).
// Loads: L2::evict_first (read once). Stores: L2::evict_last, so this call's outputs are preferred over the
// flush's dirty lines when the L2 picks victims. Both hints apply only to this call's own buffers.
// One-shot grid of small CTAs (31 regs, 16 CTAs/SM): beat persistent loops on every size in probes.

struct f8 { float v[8]; };

__device__ __forceinline__ f8 ld_x(const float* p) {
    f8 r;
    asm volatile("ld.global.nc.L1::no_allocate.L2::evict_first.v8.f32 {%0,%1,%2,%3,%4,%5,%6,%7}, [%8];"
                 : "=f"(r.v[0]), "=f"(r.v[1]), "=f"(r.v[2]), "=f"(r.v[3]),
                   "=f"(r.v[4]), "=f"(r.v[5]), "=f"(r.v[6]), "=f"(r.v[7])
                 : "l"(p));
    return r;
}

__device__ __forceinline__ f8 ld_w(const float* p) {
    f8 r;
    asm volatile("ld.global.nc.v8.f32 {%0,%1,%2,%3,%4,%5,%6,%7}, [%8];"
                 : "=f"(r.v[0]), "=f"(r.v[1]), "=f"(r.v[2]), "=f"(r.v[3]),
                   "=f"(r.v[4]), "=f"(r.v[5]), "=f"(r.v[6]), "=f"(r.v[7])
                 : "l"(p));
    return r;
}

__device__ __forceinline__ void st_y(float* p, const f8& r) {
    asm volatile("st.global.L2::evict_last.v8.f32 [%0], {%1,%2,%3,%4,%5,%6,%7,%8};"
                 :: "l"(p), "f"(r.v[0]), "f"(r.v[1]), "f"(r.v[2]), "f"(r.v[3]),
                    "f"(r.v[4]), "f"(r.v[5]), "f"(r.v[6]), "f"(r.v[7])
                 : "memory");
}

constexpr int THREADS = 128;
constexpr int ROWS_PER_CTA = THREADS / 16;

// INTERLEAVE: even CTAs take Q rows, odd CTAs the matching K rows (better at small sizes);
// otherwise CTAs walk all Q rows then all K rows.
template <bool INTERLEAVE>
__global__ void __launch_bounds__(THREADS)
qk_rmsnorm_kernel(const float* __restrict__ q, const float* __restrict__ k,
                  const float* __restrict__ wq, const float* __restrict__ wk, float eps,
                  float* __restrict__ qo, float* __restrict__ ko, int rows) {
    const int l = threadIdx.x & 15;
    const int h = threadIdx.x >> 4;
    bool isq;
    int r;
    if (INTERLEAVE) {
        isq = (blockIdx.x & 1) == 0;
        r = (blockIdx.x >> 1) * ROWS_PER_CTA + h;
        if (r >= rows) return;
    } else {
        const int g = blockIdx.x * ROWS_PER_CTA + h;
        if (g >= 2 * rows) return;
        isq = g < rows;
        r = isq ? g : g - rows;
    }
    const f8 x = ld_x((isq ? q : k) + (long)r * 128 + l * 8);
    const f8 w = ld_w((isq ? wq : wk) + (r % 48) * 128 + l * 8);
    float s = 0.f;
#pragma unroll
    for (int j = 0; j < 8; j++) s = fmaf(x.v[j], x.v[j], s);
    s += __shfl_xor_sync(0xffffffffu, s, 8);
    s += __shfl_xor_sync(0xffffffffu, s, 4);
    s += __shfl_xor_sync(0xffffffffu, s, 2);
    s += __shfl_xor_sync(0xffffffffu, s, 1);
    const float inv = rsqrtf(s * (1.f / 128.f) + eps);
    f8 y;
#pragma unroll
    for (int j = 0; j < 8; j++) y.v[j] = x.v[j] * inv * w.v[j];
    st_y((isq ? qo : ko) + (long)r * 128 + l * 8, y);
}

// rows = B*S*48 rows per tensor (always a multiple of 48, so of ROWS_PER_CTA).
extern "C" void launch_qk_rmsnorm(const float* q, const float* k, const float* wq, const float* wk,
                                  float eps, float* qo, float* ko, int rows, cudaStream_t stream) {
    if (rows <= 0) return;
    const int per_tensor = (rows + ROWS_PER_CTA - 1) / ROWS_PER_CTA;
    if (rows <= 256 * 48) {
        qk_rmsnorm_kernel<true><<<2 * per_tensor, THREADS, 0, stream>>>(q, k, wq, wk, eps, qo, ko, rows);
    } else {
        const int grid = (2 * rows + ROWS_PER_CTA - 1) / ROWS_PER_CTA;
        qk_rmsnorm_kernel<false><<<grid, THREADS, 0, stream>>>(q, k, wq, wk, eps, qo, ko, rows);
    }
}
```
