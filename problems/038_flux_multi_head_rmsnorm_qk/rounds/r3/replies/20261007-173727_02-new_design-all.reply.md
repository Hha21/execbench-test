### Rationale
This fills the empty `lang:cutedsl` niche. It is a CuTe DSL port of the B200 winner, the one-shot fused design, with one change: each thread moves **256 bits**. A half-warp of 16 lanes × 8 floats covers one 128-float row, and the row sum is a 4-step butterfly. Triton never emits 256-bit loads.

- **Tile:** 256 threads per CTA and 16 rows per tile, so the head pattern is fixed and no masks are needed.
- **Fusion:** Q tiles and K tiles share one launch.
- **Compilation:** the kernel is compiled once with a dynamic row count, so every workload uses the same binary.
- **Fallback:** if the 256-bit copy atom is rejected at compile time, the same kernel is recompiled with 128-bit atoms. That variant uses 32 lanes per row and 8 rows per CTA, so the candidate is still correct.

The target is the L/M bands: half the LSU instructions per byte, with 64 KB per SM in flight. The S band should stay about the same as the parent. Nothing here has been compiled or run; the API choices are flagged as risks in the card.

```python file=kernel.py
"""r3-cute-ldg256-os-r16: CuTe DSL kernel for SOL-ExecBench #38 (per-head RMSNorm of Q and K, fp32).

View every tensor as [R, 128] with R = B*S*48. One launch covers both streams: CTAs [0, n_tiles) do Q tiles and
CTAs [n_tiles, 2*n_tiles) do K tiles. A tile is ROWS consecutive rows (ROWS divides 48, so there are no masks), and
its weights are tile % (48 // ROWS) of the [48,128] weight matrix.

Per thread: one contiguous VEC-float segment of one row (VEC=8, i.e. 256-bit, on sm_100a). Weights are loaded in
the same way. The row sum of squares is a butterfly over the TPR = 128/VEC lanes that share the row. Then
y = (x * rsqrt(mean + eps)) * w, written with the same vector width.

The kernel is compiled once, at the first call, with a dynamic row count, so one binary serves every workload
shape. If the 256-bit copy atom fails to compile, it is recompiled with 128-bit atoms (VEC=4, 8 rows per CTA).
That choice depends only on the toolchain, never on data. All maths is fp32. The kernel is launched on the current
PyTorch stream.
"""

import torch
import cutlass
import cutlass.cute as cute
from cutlass.cute.runtime import from_dlpack
import cuda.bindings.driver as cuda

D = 128
H = 48
THREADS = 256

_frag_like = getattr(cute, "make_fragment_like", None) or getattr(cute, "make_rmem_tensor_like")


@cute.jit
def _row_tile(gX: cute.Tensor, gW: cute.Tensor, gY: cute.Tensor, tile, tidx, eps,
              thr_layout: cute.Layout, val_layout: cute.Layout,
              vec: cutlass.Constexpr, n_shfl: cutlass.Constexpr, groups: cutlass.Constexpr):
    atom = cute.make_copy_atom(cute.nvgpu.CopyUniversalOp(), gX.element_type, num_bits_per_copy=32 * vec)
    tiled_copy = cute.make_tiled_copy_tv(atom, thr_layout, val_layout)
    thr_copy = tiled_copy.get_slice(tidx)

    blkX = gX[((None, None), tile)]
    blkW = gW[((None, None), tile % groups)]
    blkY = gY[((None, None), tile)]

    tXgX = thr_copy.partition_S(blkX)
    tWgW = thr_copy.partition_S(blkW)
    tYgY = thr_copy.partition_D(blkY)

    rX = _frag_like(tXgX)
    rW = _frag_like(tWgW)
    # issue both loads before any dependent work
    cute.copy(atom, tXgX, rX)
    cute.copy(atom, tWgW, rW)

    x = rX.load()
    ss = (x * x).reduce(cute.ReductionOp.ADD, init_val=0.0, reduction_profile=0)
    for i in cutlass.range_constexpr(n_shfl):          # lanes of one row are contiguous: offsets < TPR
        ss = ss + cute.arch.shuffle_sync_bfly(ss, offset=1 << i)
    r = cute.math.rsqrt(ss * (1.0 / 128.0) + eps, fastmath=True)   # * (1/128) is exact

    y = (x * r) * rW.load()
    rY = _frag_like(tYgY)
    rY.store(y)
    cute.copy(atom, rY, tYgY)


class _QKRMSNorm:
    def __init__(self, vec):
        self.vec = vec                     # floats per thread per copy (8 -> 256-bit, 4 -> 128-bit)
        self.tpr = D // vec                # threads per row
        self.rows = THREADS // self.tpr    # rows per CTA tile (16 or 8; both divide 48)
        self.groups = H // self.rows       # distinct weight tiles
        self.n_shfl = (self.tpr - 1).bit_length()

    @cute.jit
    def __call__(self, mQ: cute.Tensor, mK: cute.Tensor, mWq: cute.Tensor, mWk: cute.Tensor,
                 mQo: cute.Tensor, mKo: cute.Tensor, eps: cutlass.Float32, stream: cuda.CUstream):
        thr_layout = cute.make_ordered_layout((self.rows, self.tpr), order=(1, 0))
        val_layout = cute.make_ordered_layout((1, self.vec), order=(1, 0))
        tiler_mn, tv_layout = cute.make_layout_tv(thr_layout, val_layout)   # tiler = (rows, 128)

        gQ = cute.zipped_divide(mQ, tiler_mn)
        gK = cute.zipped_divide(mK, tiler_mn)
        gQo = cute.zipped_divide(mQo, tiler_mn)
        gKo = cute.zipped_divide(mKo, tiler_mn)
        gWq = cute.zipped_divide(mWq, tiler_mn)
        gWk = cute.zipped_divide(mWk, tiler_mn)

        n_tiles = cute.size(gQ, mode=[1])
        self.kernel(gQ, gK, gWq, gWk, gQo, gKo, eps, thr_layout, val_layout).launch(
            grid=[2 * n_tiles, 1, 1],
            block=[cute.size(tv_layout, mode=[0]), 1, 1],
            stream=stream,
        )

    @cute.kernel
    def kernel(self, gQ: cute.Tensor, gK: cute.Tensor, gWq: cute.Tensor, gWk: cute.Tensor,
               gQo: cute.Tensor, gKo: cute.Tensor, eps: cutlass.Float32,
               thr_layout: cute.Layout, val_layout: cute.Layout):
        tidx, _, _ = cute.arch.thread_idx()
        bidx, _, _ = cute.arch.block_idx()
        n_tiles = cute.size(gQ, mode=[1])
        if bidx < n_tiles:       # uniform per CTA
            _row_tile(gQ, gWq, gQo, bidx, tidx, eps, thr_layout, val_layout,
                      self.vec, self.n_shfl, self.groups)
        else:
            _row_tile(gK, gWk, gKo, bidx - n_tiles, tidx, eps, thr_layout, val_layout,
                      self.vec, self.n_shfl, self.groups)


_COMPILED = None   # (compiled kernel, vec); compiled-kernel cache only, no tensors or data kept


def _wrap_rows(t, align):
    # dynamic row count, static 128-float row stride (needed so vector copies are provably aligned)
    return from_dlpack(t, assumed_align=align).mark_compact_shape_dynamic(
        mode=0, stride_order=t.dim_order(), divisibility=16)


def _wrap_w(t, align):
    return from_dlpack(t, assumed_align=align)   # [48,128], fully static layout


def _args(vec, q, k, wq, wk, qo, ko):
    align = 4 * vec
    return (_wrap_rows(q, align), _wrap_rows(k, align), _wrap_w(wq, align), _wrap_w(wk, align),
            _wrap_rows(qo, align), _wrap_rows(ko, align))


def _get(tensors, eps, stream):
    global _COMPILED
    if _COMPILED is None:
        last_err = None
        for vec in (8, 4):      # 256-bit first; 128-bit only if the toolchain rejects 256-bit atoms
            try:
                c = cute.compile(_QKRMSNorm(vec), *_args(vec, *tensors), eps, stream)
                _COMPILED = (c, vec)
                break
            except Exception as e:   # compile-time only; independent of data
                last_err = e
        if _COMPILED is None:
            raise last_err
    return _COMPILED


def run(query, key, weight_q, weight_k, eps, query_norm, key_norm):
    q = query.view(-1, D)
    k = key.view(-1, D)
    qo = query_norm.view(-1, D)
    ko = key_norm.view(-1, D)
    tensors = (q, k, weight_q, weight_k, qo, ko)
    eps = float(eps)
    stream = cuda.CUstream(torch.cuda.current_stream().cuda_stream)
    compiled, vec = _get(tensors, eps, stream)
    compiled(*_args(vec, *tensors), eps, stream)
```
