### Rationale

The new kernel serves only B·S ≤ 296 tokens on B200: workloads 128, 131, 256 and 256. Every larger size, including 586, goes to the parent's code, which is copied verbatim and keeps its exact configuration.

The new kernel has a fixed grid of 2·SMs (296 on B200) CTAs of 384 threads. Each CTA works on half-tokens (24 rows each), and each CTA issues all of its 256-bit loads before any arithmetic: 2 rows per thread up to 148 tokens, 4 rows per thread up to 296 tokens.

Because the stride (148 half-tokens) is even, each thread's head is fixed. It loads its weight segment once, and the CTA count and warp count fall well below the parent's 768 CTAs and 6,144 warps.

My model of the parent's roughly 3 µs fixed cost is:

- dispatching about 6k warps (about 0.5–1 µs);
- the first DRAM round trip (about 1 µs);
- the store drain and end-of-grid flush (about 0.5 µs);
- write-backs of the harness's dirty L2 lines, which make the bandwidth term look like about 6 TB/s.

This candidate attacks the warp dispatch and the per-row overhead. The write-back term is left untouched, so it stays isolated.

```python file=kernel.py
"""r4-cute-ldg256-sw2-disp: size dispatcher for SOL-ExecBench #38 (per-head RMSNorm of Q and K, fp32).

Large path (unchanged from parent r3-cute-ldg256-os-r16): one-shot grid, 16-row tiles, 256-bit copies.

Small path (new, spec:S): a fixed grid of 2*GS CTAs (GS = SM count rounded down to even; 148 on B200).
CTAs [0, GS) do Q and CTAs [GS, 2*GS) do K. A unit is half a token (24 rows, 24 x 128 floats), and each
384-thread CTA covers it with 16 threads per row and 8 floats (256 bit) per thread. CTA cs handles the units
cs + j*GS for j < NT (NT = 2 or 4, constexpr). It issues all of its loads up front, then reduces and stores.
GS is even, so the unit parity (which half of the 48 heads) is fixed per CTA. Each thread therefore loads its
weight segment once.

The dispatch depends only on shapes and the device's SM count. All maths is fp32. Everything is launched on the
current PyTorch stream as exactly one kernel per call. Only compiled kernels are cached.
"""

import torch
import cutlass
import cutlass.cute as cute
from cutlass.cute.runtime import from_dlpack
import cuda.bindings.driver as cuda

D = 128
H = 48
THREADS = 256
S_ROWS = 24          # rows per small-path unit (half token); divides 48

_frag_like = getattr(cute, "make_fragment_like", None) or getattr(cute, "make_rmem_tensor_like")


# ----------------------------------------------------------------------------------------------------------
# Parent kernel (verbatim from r3-cute-ldg256-os-r16)
# ----------------------------------------------------------------------------------------------------------
@cute.jit
def _row_tile(gX: cute.Tensor, gW: cute.Tensor, gY: cute.Tensor, tile, tidx, eps,
              thr_layout: cute.Layout, val_layout: cute.Layout,
              vec: cutlass.Constexpr, n_shfl: cutlass.Constexpr, groups: cutlass.Constexpr):
    atom = cute.make_copy_atom(cute.nvgpu.CopyUniversalOp(), gX.element_type, num_bits_per_copy=32 * vec)
    tiled_copy = cute.make_tiled_copy_tv(atom, thr_layout, val_layout)
    thr_copy = tiled_copy.get_slice(tidx)

    blkX = gX[((None, None), tile)]
    blkW = gW[((None, None), tile % groups)]
    blkY = gY[((None, None), tile)]

    tXgX = thr_copy.partition_S(blkX)
    tWgW = thr_copy.partition_S(blkW)
    tYgY = thr_copy.partition_D(blkY)

    rX = _frag_like(tXgX)
    rW = _frag_like(tWgW)
    cute.copy(atom, tXgX, rX)
    cute.copy(atom, tWgW, rW)

    x = rX.load()
    ss = (x * x).reduce(cute.ReductionOp.ADD, init_val=0.0, reduction_profile=0)
    for i in cutlass.range_constexpr(n_shfl):
        ss = ss + cute.arch.shuffle_sync_bfly(ss, offset=1 << i)
    r = cute.math.rsqrt(ss * (1.0 / 128.0) + eps, fastmath=True)

    y = (x * r) * rW.load()
    rY = _frag_like(tYgY)
    rY.store(y)
    cute.copy(atom, rY, tYgY)


class _QKRMSNorm:
    def __init__(self, vec):
        self.vec = vec
        self.tpr = D // vec
        self.rows = THREADS // self.tpr
        self.groups = H // self.rows
        self.n_shfl = (self.tpr - 1).bit_length()

    @cute.jit
    def __call__(self, mQ: cute.Tensor, mK: cute.Tensor, mWq: cute.Tensor, mWk: cute.Tensor,
                 mQo: cute.Tensor, mKo: cute.Tensor, eps: cutlass.Float32, stream: cuda.CUstream):
        thr_layout = cute.make_ordered_layout((self.rows, self.tpr), order=(1, 0))
        val_layout = cute.make_ordered_layout((1, self.vec), order=(1, 0))
        tiler_mn, tv_layout = cute.make_layout_tv(thr_layout, val_layout)

        gQ = cute.zipped_divide(mQ, tiler_mn)
        gK = cute.zipped_divide(mK, tiler_mn)
        gQo = cute.zipped_divide(mQo, tiler_mn)
        gKo = cute.zipped_divide(mKo, tiler_mn)
        gWq = cute.zipped_divide(mWq, tiler_mn)
        gWk = cute.zipped_divide(mWk, tiler_mn)

        n_tiles = cute.size(gQ, mode=[1])
        self.kernel(gQ, gK, gWq, gWk, gQo, gKo, eps, thr_layout, val_layout).launch(
            grid=[2 * n_tiles, 1, 1],
            block=[cute.size(tv_layout, mode=[0]), 1, 1],
            stream=stream,
        )

    @cute.kernel
    def kernel(self, gQ: cute.Tensor, gK: cute.Tensor, gWq: cute.Tensor, gWk: cute.Tensor,
               gQo: cute.Tensor, gKo: cute.Tensor, eps: cutlass.Float32,
               thr_layout: cute.Layout, val_layout: cute.Layout):
        tidx, _, _ = cute.arch.thread_idx()
        bidx, _, _ = cute.arch.block_idx()
        n_tiles = cute.size(gQ, mode=[1])
        if bidx < n_tiles:
            _row_tile(gQ, gWq, gQo, bidx, tidx, eps, thr_layout, val_layout,
                      self.vec, self.n_shfl, self.groups)
        else:
            _row_tile(gK, gWk, gKo, bidx - n_tiles, tidx, eps, thr_layout, val_layout,
                      self.vec, self.n_shfl, self.groups)


# ----------------------------------------------------------------------------------------------------------
# Small-input kernel (new)
# ----------------------------------------------------------------------------------------------------------
@cute.jit
def _multi_unit(gX: cute.Tensor, gW: cute.Tensor, gY: cute.Tensor, cs, tidx, eps, n_units,
                thr_layout: cute.Layout, val_layout: cute.Layout,
                vec: cutlass.Constexpr, n_shfl: cutlass.Constexpr,
                nt: cutlass.Constexpr, gs: cutlass.Constexpr):
    atom = cute.make_copy_atom(cute.nvgpu.CopyUniversalOp(), gX.element_type, num_bits_per_copy=32 * vec)
    tiled_copy = cute.make_tiled_copy_tv(atom, thr_layout, val_layout)
    thr_copy = tiled_copy.get_slice(tidx)

    # gs is even, so every unit of this CTA has parity cs % 2: same 24 heads, one weight load per thread
    tWgW = thr_copy.partition_S(gW[((None, None), cs % 2)])
    rW = _frag_like(tWgW)
    tX0 = thr_copy.partition_S(gX[((None, None), 0)])
    rXs = []
    for j in cutlass.range_constexpr(nt):
        rXs.append(_frag_like(tX0))

    # issue every load before any dependent work
    cute.copy(atom, tWgW, rW)
    for j in cutlass.range_constexpr(nt):
        u = cs + j * gs
        if u < n_units:                       # uniform per CTA
            cute.copy(atom, thr_copy.partition_S(gX[((None, None), u)]), rXs[j])

    w = rW.load()
    for j in cutlass.range_constexpr(nt):
        x = rXs[j].load()
        ss = (x * x).reduce(cute.ReductionOp.ADD, init_val=0.0, reduction_profile=0)
        for i in cutlass.range_constexpr(n_shfl):   # the 16 lanes of a row are contiguous
            ss = ss + cute.arch.shuffle_sync_bfly(ss, offset=1 << i)
        r = cute.math.rsqrt(ss * (1.0 / 128.0) + eps, fastmath=True)   # * (1/128) is exact
        rXs[j].store((x * r) * w)
        u = cs + j * gs
        if u < n_units:                       # invalid units computed garbage; never stored
            cute.copy(atom, rXs[j], thr_copy.partition_D(gY[((None, None), u)]))


class _QKRMSNormS:
    def __init__(self, vec, nt, gs):
        self.vec = vec
        self.tpr = D // vec
        self.rows = S_ROWS
        self.nt = nt
        self.gs = gs
        self.n_shfl = (self.tpr - 1).bit_length()

    @cute.jit
    def __call__(self, mQ: cute.Tensor, mK: cute.Tensor, mWq: cute.Tensor, mWk: cute.Tensor,
                 mQo: cute.Tensor, mKo: cute.Tensor, eps: cutlass.Float32, stream: cuda.CUstream):
        thr_layout = cute.make_ordered_layout((self.rows, self.tpr), order=(1, 0))
        val_layout = cute.make_ordered_layout((1, self.vec), order=(1, 0))
        tiler_mn, tv_layout = cute.make_layout_tv(thr_layout, val_layout)   # tiler = (24, 128)

        gQ = cute.zipped_divide(mQ, tiler_mn)
        gK = cute.zipped_divide(mK, tiler_mn)
        gQo = cute.zipped_divide(mQo, tiler_mn)
        gKo = cute.zipped_divide(mKo, tiler_mn)
        gWq = cute.zipped_divide(mWq, tiler_mn)   # 2 tiles: heads 0-23, 24-47
        gWk = cute.zipped_divide(mWk, tiler_mn)

        self.kernel(gQ, gK, gWq, gWk, gQo, gKo, eps, thr_layout, val_layout).launch(
            grid=[2 * self.gs, 1, 1],
            block=[cute.size(tv_layout, mode=[0]), 1, 1],
            stream=stream,
        )

    @cute.kernel
    def kernel(self, gQ: cute.Tensor, gK: cute.Tensor, gWq: cute.Tensor, gWk: cute.Tensor,
               gQo: cute.Tensor, gKo: cute.Tensor, eps: cutlass.Float32,
               thr_layout: cute.Layout, val_layout: cute.Layout):
        tidx, _, _ = cute.arch.thread_idx()
        bidx, _, _ = cute.arch.block_idx()
        n_units = cute.size(gQ, mode=[1])
        if bidx < self.gs:
            _multi_unit(gQ, gWq, gQo, bidx, tidx, eps, n_units, thr_layout, val_layout,
                        self.vec, self.n_shfl, self.nt, self.gs)
        else:
            _multi_unit(gK, gWk, gKo, bidx - self.gs, tidx, eps, n_units, thr_layout, val_layout,
                        self.vec, self.n_shfl, self.nt, self.gs)


# ----------------------------------------------------------------------------------------------------------
# Host side: compiled-kernel caches only (no tensors, no data)
# ----------------------------------------------------------------------------------------------------------
_COMPILED = None      # parent: (compiled, vec)
_COMPILED_S = {}      # small path: nt -> (compiled, vec)
_GS = None            # SM count rounded down to even (device property)


def _wrap_rows(t, align):
    return from_dlpack(t, assumed_align=align).mark_compact_shape_dynamic(
        mode=0, stride_order=t.dim_order(), divisibility=16)


def _wrap_w(t, align):
    return from_dlpack(t, assumed_align=align)


def _args(vec, q, k, wq, wk, qo, ko):
    align = 4 * vec
    return (_wrap_rows(q, align), _wrap_rows(k, align), _wrap_w(wq, align), _wrap_w(wk, align),
            _wrap_rows(qo, align), _wrap_rows(ko, align))


def _get(tensors, eps, stream):
    global _COMPILED
    if _COMPILED is None:
        last_err = None
        for vec in (8, 4):
            try:
                c = cute.compile(_QKRMSNorm(vec), *_args(vec, *tensors), eps, stream)
                _COMPILED = (c, vec)
                break
            except Exception as e:   # compile-time only; independent of data
                last_err = e
        if _COMPILED is None:
            raise last_err
    return _COMPILED


def _get_s(nt, gs, tensors, eps, stream):
    ent = _COMPILED_S.get(nt)
    if ent is None:
        last_err = None
        for vec in (8, 4):   # 256-bit first; 128-bit (768-thread CTAs) only if the toolchain rejects it
            try:
                c = cute.compile(_QKRMSNormS(vec, nt, gs), *_args(vec, *tensors), eps, stream)
                ent = (c, vec)
                break
            except Exception as e:   # compile-time only; independent of data
                last_err = e
        if ent is None:
            raise last_err
        _COMPILED_S[nt] = ent
    return ent


def _get_gs():
    global _GS
    if _GS is None:
        n = torch.cuda.get_device_properties(torch.cuda.current_device()).multi_processor_count
        _GS = max(2, n - (n % 2))
    return _GS


def run(query, key, weight_q, weight_k, eps, query_norm, key_norm):
    q = query.view(-1, D)
    k = key.view(-1, D)
    qo = query_norm.view(-1, D)
    ko = key_norm.view(-1, D)
    tensors = (q, k, weight_q, weight_k, qo, ko)
    eps = float(eps)
    stream = cuda.CUstream(torch.cuda.current_stream().cuda_stream)

    n_units = q.shape[0] // S_ROWS          # half-tokens per stream (rows are a multiple of 48)
    gs = _get_gs()
    if n_units <= 4 * gs:                   # shape-only dispatch: B*S <= 2*gs tokens (296 on B200)
        nt = 2 if n_units <= 2 * gs else 4
        compiled, vec = _get_s(nt, gs, tensors, eps, stream)
    else:
        compiled, vec = _get(tensors, eps, stream)
    compiled(*_args(vec, *tensors), eps, stream)
```
