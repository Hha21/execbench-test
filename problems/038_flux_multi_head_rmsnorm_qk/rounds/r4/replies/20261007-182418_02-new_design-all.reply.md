### Rationale
This candidate fills `tile:token`. Each CTA normalises all 48 heads of one token for one stream (Q or K), so the weight tile is the whole `[48,128]` matrix. Each thread's weight address depends only on `threadIdx` and the stream, with no `% 48` arithmetic.

The useful change is bytes in flight. The parent holds one 256-bit load per thread, which is 64 KB per SM. Here 384 threads per CTA each own two rows (head h and h+24), so each thread issues two 256-bit DRAM loads up front. `__launch_bounds__(384,4)` caps registers at 40, which keeps 4 CTAs per SM (1,536 threads) and gives about 96 KB of DRAM reads in flight per SM, 1.5× the parent. That should lift M/L bandwidth.

The cost is coarser granularity on S. At 128 tokens there are 256 CTAs, so 40 of the 148 SMs get only one CTA. The S band may lose a few percent.

The code is CUDA C++ with an sm_100 `v8.f32` path and a `float4` fallback, so it also runs on CSF3.

```cuda file=kernel.cu
// r4-cuda-ldg256-token-r2: SOL-ExecBench #38 (per-head RMSNorm of Q and K, fp32), tile:token design.
//
// View each tensor as [n_tok, 48, 128]. One CTA handles one token of one stream:
//   blockIdx.x <  n_tok : Q token blockIdx.x
//   blockIdx.x >= n_tok : K token blockIdx.x - n_tok
// 384 threads = 24 half-warps. Half-warp hw (0..23) owns heads hw and hw+24, and each lane holds 8 contiguous floats
// of both rows (two 256-bit loads on sm_100). The weight tile is the whole [48,128] matrix and the same for every
// CTA of a stream, so a thread's weight address depends only on threadIdx and the stream (L1/L2 resident).
// Row sum of squares: in-thread, then butterfly xor 8,4,2,1 inside the half-warp.
// y = (x * rsqrt(sum * (1/128) + eps)) * w, all fp32. One launch per call on the current PyTorch stream.

#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda_runtime.h>
#include <cstdint>

namespace {

constexpr int D = 128;
constexpr int H = 48;
constexpr int VEC = 8;           // floats per thread per row
constexpr int HALF_H = H / 2;    // 24: second row of each half-warp is head + 24
constexpr int THREADS = 384;     // 24 half-warps x 16 lanes
constexpr int MIN_BLOCKS = 4;    // 4 x 384 threads -> <= 42 regs -> 40

__device__ __forceinline__ void ld8_stream(float (&v)[8], const float* p) {
#if defined(__CUDA_ARCH__) && (__CUDA_ARCH__ >= 1000)
    asm volatile("ld.global.L1::no_allocate.v8.f32 {%0,%1,%2,%3,%4,%5,%6,%7}, [%8];"
                 : "=f"(v[0]), "=f"(v[1]), "=f"(v[2]), "=f"(v[3]),
                   "=f"(v[4]), "=f"(v[5]), "=f"(v[6]), "=f"(v[7])
                 : "l"(p));
#else
    const float4* p4 = reinterpret_cast<const float4*>(p);
    float4 a = p4[0];
    float4 b = p4[1];
    v[0] = a.x; v[1] = a.y; v[2] = a.z; v[3] = a.w;
    v[4] = b.x; v[5] = b.y; v[6] = b.z; v[7] = b.w;
#endif
}

__device__ __forceinline__ void ld8_weight(float (&v)[8], const float* p) {
#if defined(__CUDA_ARCH__) && (__CUDA_ARCH__ >= 1000)
    asm volatile("ld.global.v8.f32 {%0,%1,%2,%3,%4,%5,%6,%7}, [%8];"
                 : "=f"(v[0]), "=f"(v[1]), "=f"(v[2]), "=f"(v[3]),
                   "=f"(v[4]), "=f"(v[5]), "=f"(v[6]), "=f"(v[7])
                 : "l"(p));
#else
    const float4* p4 = reinterpret_cast<const float4*>(p);
    float4 a = __ldg(p4);
    float4 b = __ldg(p4 + 1);
    v[0] = a.x; v[1] = a.y; v[2] = a.z; v[3] = a.w;
    v[4] = b.x; v[5] = b.y; v[6] = b.z; v[7] = b.w;
#endif
}

__device__ __forceinline__ void st8(float* p, const float (&v)[8]) {
#if defined(__CUDA_ARCH__) && (__CUDA_ARCH__ >= 1000)
    asm volatile("st.global.v8.f32 [%0], {%1,%2,%3,%4,%5,%6,%7,%8};"
                 :: "l"(p), "f"(v[0]), "f"(v[1]), "f"(v[2]), "f"(v[3]),
                    "f"(v[4]), "f"(v[5]), "f"(v[6]), "f"(v[7])
                 : "memory");
#else
    float4* p4 = reinterpret_cast<float4*>(p);
    p4[0] = make_float4(v[0], v[1], v[2], v[3]);
    p4[1] = make_float4(v[4], v[5], v[6], v[7]);
#endif
}

__global__ void __launch_bounds__(THREADS, MIN_BLOCKS)
qk_rms_token_kernel(const float* __restrict__ q, const float* __restrict__ k,
                    const float* __restrict__ wq, const float* __restrict__ wk,
                    float* __restrict__ qo, float* __restrict__ ko,
                    int n_tok, float eps) {
    const int b = blockIdx.x;
    const bool is_k = b >= n_tok;                 // uniform per CTA
    const int tok = is_k ? (b - n_tok) : b;
    const float* __restrict__ x = is_k ? k : q;
    const float* __restrict__ w = is_k ? wk : wq;
    float* __restrict__ y = is_k ? ko : qo;

    const int t = threadIdx.x;
    const int head = t >> 4;                      // 0..23
    const int col = (t & 15) * VEC;               // 0..120
    const int off0 = head * D + col;              // offset inside the [48,128] token tile
    const int off1 = off0 + HALF_H * D;           // head + 24

    const size_t base = (size_t)tok * (size_t)(H * D);

    float a[8], c[8], wa[8], wc[8];
    // All loads issued before any dependent work: 2 DRAM streams + 2 L1/L2 weight hits.
    ld8_stream(a, x + base + off0);
    ld8_stream(c, x + base + off1);
    ld8_weight(wa, w + off0);
    ld8_weight(wc, w + off1);

    float s0a = a[0] * a[0], s0b = a[1] * a[1];
    float s1a = c[0] * c[0], s1b = c[1] * c[1];
#pragma unroll
    for (int i = 2; i < 8; i += 2) {
        s0a = fmaf(a[i], a[i], s0a);
        s0b = fmaf(a[i + 1], a[i + 1], s0b);
        s1a = fmaf(c[i], c[i], s1a);
        s1b = fmaf(c[i + 1], c[i + 1], s1b);
    }
    float s0 = s0a + s0b;
    float s1 = s1a + s1b;
#pragma unroll
    for (int o = 8; o >= 1; o >>= 1) {            // xor < 16: stays inside the half-warp (one row)
        s0 += __shfl_xor_sync(0xffffffffu, s0, o);
        s1 += __shfl_xor_sync(0xffffffffu, s1, o);
    }
    const float r0 = rsqrtf(s0 * (1.0f / 128.0f) + eps);   // * (1/128) is exact
    const float r1 = rsqrtf(s1 * (1.0f / 128.0f) + eps);

#pragma unroll
    for (int i = 0; i < 8; ++i) {
        a[i] = (a[i] * r0) * wa[i];
        c[i] = (c[i] * r1) * wc[i];
    }
    st8(y + base + off0, a);
    st8(y + base + off1, c);
}

}  // namespace

void run(torch::Tensor query, torch::Tensor key, torch::Tensor weight_q, torch::Tensor weight_k,
         double eps, torch::Tensor query_norm, torch::Tensor key_norm) {
    TORCH_CHECK(query.is_cuda() && key.is_cuda() && query_norm.is_cuda() && key_norm.is_cuda(), "CUDA tensors required");
    TORCH_CHECK(query.scalar_type() == torch::kFloat32 && key.scalar_type() == torch::kFloat32 &&
                weight_q.scalar_type() == torch::kFloat32 && weight_k.scalar_type() == torch::kFloat32 &&
                query_norm.scalar_type() == torch::kFloat32 && key_norm.scalar_type() == torch::kFloat32,
                "fp32 tensors required");
    TORCH_CHECK(query.is_contiguous() && key.is_contiguous() && weight_q.is_contiguous() && weight_k.is_contiguous() &&
                query_norm.is_contiguous() && key_norm.is_contiguous(), "contiguous tensors required");
    TORCH_CHECK(weight_q.numel() == H * D && weight_k.numel() == H * D, "weights must be [48,128]");
    const int64_t numel = query.numel();
    TORCH_CHECK(numel % (H * D) == 0 && key.numel() == numel && query_norm.numel() == numel &&
                key_norm.numel() == numel, "shape mismatch");
    const auto aligned = [](const torch::Tensor& t) {
        return (reinterpret_cast<uintptr_t>(t.data_ptr()) & 31u) == 0;
    };
    TORCH_CHECK(aligned(query) && aligned(key) && aligned(weight_q) && aligned(weight_k) &&
                aligned(query_norm) && aligned(key_norm), "32-byte alignment required");

    const int n_tok = static_cast<int>(numel / (H * D));
    if (n_tok == 0) return;
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    dim3 grid(2 * n_tok), block(THREADS);
    qk_rms_token_kernel<<<grid, block, 0, stream>>>(
        query.data_ptr<float>(), key.data_ptr<float>(),
        weight_q.data_ptr<float>(), weight_k.data_ptr<float>(),
        query_norm.data_ptr<float>(), key_norm.data_ptr<float>(),
        n_tok, static_cast<float>(eps));
    C10_CUDA_KERNEL_LAUNCH_CHECK();
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("run", &run, "FLUX multi-head RMSNorm Q/K (tile:token, fp32)");
}
```
