### Rationale
This is the first kernel in the `lang:gluon` niche. It passes all 16 workloads on A100, but it does not beat the parent anywhere. On A100 it is 40–50% slower than r5 in every band, so I would not submit it for a score.

- **Plain loads did not compile.** In this Gluon build, `gl.load` returns a tensor with no layout. Both `gl.sum` and `gl.convert_layout` then fail to compile ("expected distributed_type").
- **Working path:** each thread copies its data into shared memory with Ampere-style async copies (`cp.async`), waits, and reads it back into registers, the pattern FlashInfer and quack use.
- **4-byte copies:** a shared layout with `vec=1` produces 4-byte copies. `vec=8` gives 128-bit copies (`LDGSTS.128`). The sm_100a build uses 30 registers, 16 KB shared memory and fits 8 CTAs per SM.
- **The slowdown is not the hints or L1.** Two controls gave identical times: `.cg` copies that bypass L1, and stores with no cache hint. The cause is therefore the cp.async → shared memory → register path itself, or how Gluon lowers it on sm_80.
- **No B200 gain expected:** about +15–40%, low confidence.

```python file=kernel.py
# r6-gluon-cpasync-os-r16w8-stel: per-head RMSNorm of Q and K (fp32) in Gluon, one fused one-shot launch.
# 16 rows per 256-thread CTA; each thread owns 8 contiguous floats of one row (a half-warp per row).
# Loads go global -> shared through 128-bit cp.async (LDGSTS.128): in this Gluon build gl.load returns a
# tensor without a layout, so the explicit async-copy path is used instead. Output stores carry L2 evict_last.
# All maths fp32; gl.rsqrt -> rsqrt.approx; 1/128 is an exact multiply.
import torch
from triton.experimental import gluon
from triton.experimental.gluon import language as gl
from triton.experimental.gluon.language.nvidia.ampere import async_copy as cp

D = 128
H = 48


@gluon.jit
def _tile(x_ptr, w_ptr, y_ptr, tile, eps, ROWS: gl.constexpr, D: gl.constexpr, H: gl.constexpr,
          layout: gl.constexpr, slayout: gl.constexpr):
    GROUPS: gl.constexpr = H // ROWS
    rows = gl.arange(0, ROWS, layout=gl.SliceLayout(1, layout))
    cols = gl.arange(0, D, layout=gl.SliceLayout(0, layout))
    c2 = gl.expand_dims(cols, 0)
    offs = gl.expand_dims(tile * ROWS + rows, 1) * D + c2
    woffs = gl.expand_dims((tile % GROUPS) * ROWS + rows, 1) * D + c2
    xs = gl.allocate_shared_memory(gl.float32, [ROWS, D], slayout)
    ws = gl.allocate_shared_memory(gl.float32, [ROWS, D], slayout)
    cp.async_copy_global_to_shared(xs, x_ptr + offs)
    cp.async_copy_global_to_shared(ws, w_ptr + woffs)
    cp.commit_group()
    cp.wait_group(0)
    # each thread reads back exactly the bytes it copied (same layout), so no CTA barrier is needed
    x = xs.load(layout)
    w = ws.load(layout)
    s = gl.sum(x * x, axis=1)
    inv = gl.rsqrt(s * (1.0 / D) + eps)
    y = (x * gl.expand_dims(inv, 1)) * w
    gl.store(y_ptr + offs, y, eviction_policy="evict_last")


@gluon.jit
def _qk_rms(q_ptr, k_ptr, wq_ptr, wk_ptr, qo_ptr, ko_ptr, n_tiles, eps,
            ROWS: gl.constexpr, D: gl.constexpr, H: gl.constexpr, layout: gl.constexpr,
            slayout: gl.constexpr):
    pid = gl.program_id(0)
    if pid < n_tiles:
        _tile(q_ptr, wq_ptr, qo_ptr, pid, eps, ROWS, D, H, layout, slayout)
    else:
        _tile(k_ptr, wk_ptr, ko_ptr, pid - n_tiles, eps, ROWS, D, H, layout, slayout)


_ROWS = 16
_WARPS = 8
_LAYOUT = gl.BlockedLayout(size_per_thread=[1, 8], threads_per_warp=[2, 16],
                           warps_per_cta=[_WARPS, 1], order=[1, 0])
_SLAYOUT = gl.SwizzledSharedLayout(vec=8, per_phase=1, max_phase=1, order=[1, 0])


def run(query, key, weight_q, weight_k, eps, query_norm, key_norm):
    n_rows = query.numel() // D
    n_tiles = n_rows // _ROWS
    if n_tiles == 0:
        return
    _qk_rms[(2 * n_tiles,)](query, key, weight_q, weight_k, query_norm, key_norm,
                            n_tiles, float(eps), ROWS=_ROWS, D=D, H=H, layout=_LAYOUT,
                            slayout=_SLAYOUT, num_warps=_WARPS)
```
