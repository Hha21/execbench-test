I found no way to cut the small-input fixed cost by anything like 1 µs. The candidate below changes only the 256-token sizes, and its expected portal gain is about 1%, which is inside portal noise. It passes all 16 workloads on the rented B200. Against the parent on that GPU, S is −1.9% and M −0.8%; M and L run the same code as the parent, so their change is noise. Projected portal score is 0.6118, against 0.6099 for the parent.

### Rationale
Probes on the rented B200 (harness timing) suggest the parent's small-input fixed cost is mostly outside the kernel's control:
- **Launch span:** an empty kernel already takes 1.45–1.65 µs, whatever the grid size.
- **First DRAM access:** a cold load costs about 0.6 µs over the empty kernel, and each further dependent load about 0.45 µs. Address translation adds nothing measurable.
- **Dirty L2 after the flush:** reading only 6.3 MB costs 2.4 µs over empty, about 2.6 TB/s. That fits every read allocation evicting one of the flush's dirty zero lines, which doubles DRAM traffic while reading. This is inferred from the timings; the clean-flush comparison I tried was too noisy to confirm it.

At 12.6 MB the parent is already at that limit. CTA shapes, rows per thread, balanced mapping, discard, the 256-byte prefetch-size hint, no-L1-allocate loads and dropping the weight loads all came within ±1% of the parent, or were worse. The one repeatable gain is the L2 prefetch one wave ahead: −2% at 256 tokens (where the grid exceeds one resident wave), but +1.7% at 586 tokens. The dispatch therefore uses it only up to 300 tokens.

```cuda file=kernel.cu
// r8-s-pf256-disp: per-head RMSNorm of Q and K (fp32), one fused one-shot launch, shape-only dispatch.
// All paths: 16 rows per 256-thread CTA, half-warp per row, 256-bit ld/st (sm_100a), stores L2 evict_last.
// MODE 0 (B*S > 600): parent r6 large path exactly (default-policy loads).
// MODE 1 (300 < B*S <= 600): parent r6 S path exactly (x L2 evict_first, w ld.global.nc).
// MODE 2 (B*S <= 300): MODE 1 + one cp.async.bulk.prefetch.L2 per CTA of the tile one resident wave ahead
//         (only does anything when the grid exceeds one resident wave, i.e. the 256-token sizes).
#include <cuda_runtime.h>
#include <cstdint>

namespace {
constexpr int D = 128;
constexpr int ROWS = 16;
constexpr int THREADS = 256;
constexpr int GROUPS = 48 / ROWS;
constexpr uint64_t POL_EVICT_FIRST = 0x12F0000000000000ULL;   // CUTLASS CacheHintSm90::EVICT_FIRST

__device__ __forceinline__ uint64_t pol_evict_last() {
    uint64_t p;
    asm volatile("createpolicy.fractional.L2::evict_last.b64 %0, 1.0;" : "=l"(p));
    return p;
}

#if defined(__CUDA_ARCH__) && (__CUDA_ARCH__ >= 1000)
__device__ __forceinline__ void ld8(const float* p, float (&v)[8]) {
    asm volatile("ld.global.v8.f32 {%0,%1,%2,%3,%4,%5,%6,%7}, [%8];"
                 : "=f"(v[0]), "=f"(v[1]), "=f"(v[2]), "=f"(v[3]),
                   "=f"(v[4]), "=f"(v[5]), "=f"(v[6]), "=f"(v[7]) : "l"(p));
}
__device__ __forceinline__ void ld8_nc(const float* p, float (&v)[8]) {
    asm volatile("ld.global.nc.v8.f32 {%0,%1,%2,%3,%4,%5,%6,%7}, [%8];"
                 : "=f"(v[0]), "=f"(v[1]), "=f"(v[2]), "=f"(v[3]),
                   "=f"(v[4]), "=f"(v[5]), "=f"(v[6]), "=f"(v[7]) : "l"(p));
}
__device__ __forceinline__ void ld8_h(const float* p, float (&v)[8], uint64_t pol) {

                   "=f"(v[4]), "=f"(v[5]), "=f"(v[6]), "=f"(v[7]) : "l"(p), "l"(pol));
}

}

__device__ __forceinline__ void l2_prefetch(const void* p, unsigned bytes) {
#if defined(__CUDA_ARCH__) && (__CUDA_ARCH__ >= 900)
    asm volatile("cp.async.bulk.prefetch.L2.global [%0], %1;" :: "l"(p), "r"(bytes) : "memory");

template <int MODE>
__global__ void __launch_bounds__(THREADS, 8)
qk_rms_kernel(const float* __restrict__ q, const float* __restrict__ k,
              const float* __restrict__ wq, const float* __restrict__ wk,
              float* __restrict__ qo, float* __restrict__ ko, int n_tiles, int pf_dist, float eps) {
    const int bid = blockIdx.x;
    const bool is_k = bid >= n_tiles;
    const int tile = is_k ? bid - n_tiles : bid;
    const float* __restrict__ x = is_k ? k : q;
    const float* __restrict__ w = is_k ? wk : wq;
    float* __restrict__ y = is_k ? ko : qo;

    const int r = threadIdx.x >> 4;
    const int c = (threadIdx.x & 15) * 8;
    const size_t row = (size_t)tile * ROWS + r;
    const int head = (tile % GROUPS) * ROWS + r;   // == row % 48

    float xv[8], wv[8], yv[8];
    if (MODE >= 1) {                                // both loads issued before any use
        ld8_h(x + row * D + c, xv, POL_EVICT_FIRST);
        ld8_nc(w + (size_t)head * D + c, wv);
    } else {
        ld8(x + row * D + c, xv);
        ld8(w + (size_t)head * D + c, wv);
    }
    if (MODE == 2 && threadIdx.x == 0) {            // warm L2 for the CTA one resident wave later
        const int pb = bid + pf_dist;
        if (pb < 2 * n_tiles) {
            const bool pk = pb >= n_tiles;
            const int pt = pk ? pb - n_tiles : pb;
            l2_prefetch((pk ? k : q) + (size_t)pt * ROWS * D, ROWS * D * 4);
        }
    }
    const uint64_t pol_out = pol_evict_last();

    float s = 0.0f;
#pragma unroll
    for (int i = 0; i < 8; ++i) s = fmaf(xv[i], xv[i], s);
    s += __shfl_xor_sync(0xffffffffu, s, 8);
    s += __shfl_xor_sync(0xffffffffu, s, 4);
    s += __shfl_xor_sync(0xffffffffu, s, 2);
    s += __shfl_xor_sync(0xffffffffu, s, 1);
    const float inv = rsqrtf(s * (1.0f / (float)D) + eps);   // * (1/128) is exact
#pragma unroll
    for (int i = 0; i < 8; ++i) yv[i] = (xv[i] * inv) * wv[i];
    st8_h(y + row * D + c, yv, pol_out);
}
}  // namespace

extern "C" int qk_rms_launch(const float* q, const float* k, const float* wq, const float* wk,
                             float* qo, float* ko, long long n_rows, int mode, int num_sms,
                             float eps, cudaStream_t stream) {
    const long long n_tiles = n_rows / ROWS;        // exact: n_rows is a multiple of 48
    const dim3 grid((unsigned)(2 * n_tiles));
    const int pf_dist = num_sms * 8;                // one resident wave ahead
    if (mode == 2)
        qk_rms_kernel<2><<<grid, THREADS, 0, stream>>>(q, k, wq, wk, qo, ko, (int)n_tiles, pf_dist, eps);
    else if (mode == 1)
        qk_rms_kernel<1><<<grid, THREADS, 0, stream>>>(q, k, wq, wk, qo, ko, (int)n_tiles, pf_dist, eps);
    else
        qk_rms_kernel<0><<<grid, THREADS, 0, stream>>>(q, k, wq, wk, qo, ko, (int)n_tiles, pf_dist, eps);
    return (int)cudaGetLastError();
}
```

```cpp file=binding.cpp
// r8-s-pf256-disp: PyTorch glue. Shape-only dispatch:
//   B*S <= 300 -> mode 2 (EF x loads, nc weights, L2 bulk prefetch one wave ahead)
//   B*S <= 600 -> mode 1 (parent r6 S path)
//   otherwise  -> mode 0 (parent r6 / r5 large path)
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <c10/cuda/CUDAGuard.h>
#include <cuda_runtime.h>
#include <cstdint>

extern "C" int qk_rms_launch(const float* q, const float* k, const float* wq, const float* wk,
                             float* qo, float* ko, long long n_rows, int mode, int num_sms,
                             float eps, cudaStream_t stream);

namespace {
constexpr int64_t D = 128, H = 48;
constexpr int64_t PF_MAX_TOKENS = 300;
constexpr int64_t S_MAX_TOKENS = 600;
int g_num_sms = 0;   // device property cache (not data-dependent)

void check_tensor(const torch::Tensor& t, const char* name) {
    TORCH_CHECK(t.is_cuda() && t.scalar_type() == torch::kFloat32 && t.is_contiguous(), name,
                " must be a contiguous fp32 CUDA tensor");
    TORCH_CHECK((reinterpret_cast<uintptr_t>(t.data_ptr()) & 31) == 0, name, " must be 32-byte aligned");
}
}  // namespace

void run(torch::Tensor query, torch::Tensor key, torch::Tensor weight_q, torch::Tensor weight_k,
         double eps, torch::Tensor query_norm, torch::Tensor key_norm) {
    check_tensor(query, "query"); check_tensor(key, "key");
    check_tensor(weight_q, "weight_q"); check_tensor(weight_k, "weight_k");
    check_tensor(query_norm, "query_norm"); check_tensor(key_norm, "key_norm");
    TORCH_CHECK(weight_q.numel() == H * D && weight_k.numel() == H * D, "weights must be [48,128]");
    TORCH_CHECK(query.numel() % (H * D) == 0 && key.numel() == query.numel() &&
                query_norm.numel() == query.numel() && key_norm.numel() == query.numel(), "shape mismatch");
    const int64_t n_rows = query.numel() / D;
    if (n_rows == 0) return;
    const int64_t tokens = n_rows / H;
    TORCH_CHECK(2 * (n_rows / 16) <= (int64_t)0x7fffffff, "grid too large");
    const c10::cuda::CUDAGuard guard(query.device());
    if (g_num_sms == 0) {
        cudaDeviceProp prop;
        C10_CUDA_CHECK(cudaGetDeviceProperties(&prop, query.device().index()));
        g_num_sms = prop.multiProcessorCount;
    }
    const int mode = tokens <= PF_MAX_TOKENS ? 2 : (tokens <= S_MAX_TOKENS ? 1 : 0);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    const int err = qk_rms_launch(query.data_ptr<float>(), key.data_ptr<float>(),
                                  weight_q.data_ptr<float>(), weight_k.data_ptr<float>(),
                                  query_norm.data_ptr<float>(), key_norm.data_ptr<float>(),
                                  (long long)n_rows, mode, g_num_sms, (float)eps, stream);
    TORCH_CHECK(err == 0, "qk_rms kernel launch failed: ", cudaGetErrorString((cudaError_t)err));
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("run", &run, "QK RMSNorm fp32; S: EF x, nc w, L2 prefetch for B*S<=300; M/L: parent r6");
}
```
